// Procedural PBR Micro-Detail, Edge-Wear, Caustics & Fur Textures
function makeCanvasTex(size, drawFn, repeat = 1) {
  const c = document.createElement('canvas');
  c.width = c.height = size;
  const ctx = c.getContext('2d');
  drawFn(ctx, size);
  const tex = new THREE.CanvasTexture(c);
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  tex.repeat.set(repeat, repeat);
  tex.anisotropy = 4;
  return tex;
}

const microNormalTex = makeCanvasTex(256, (ctx, s) => {
  const img = ctx.createImageData(s, s);
  for (let i = 0; i < img.data.length; i += 4) {
    const n1 = Math.sin(i * 0.017) * Math.cos(i * 0.031);
    const n2 = (Math.random() - 0.5) * 0.6;
    img.data[i] = Math.floor(128 + (n1 * 22 + n2 * 18));
    img.data[i + 1] = Math.floor(128 + (n1 * 18 - n2 * 22));
    img.data[i + 2] = 245;
    img.data[i + 3] = 255;
  }
  ctx.putImageData(img, 0, 0);
}, 3);

// Realistic Shader Injector: AO, SSS, Edge Wear, Parallax Depth, Wetness & Wind
R.enhanceMaterial = function(mat, opts = {}) {
  if (!mat || R.enhancedMaterials.has(mat)) return mat;
  if (!(mat.isMeshStandardMaterial || mat.isMeshPhysicalMaterial)) return mat;
  R.enhancedMaterials.add(mat);

  if (!mat.normalMap) {
    mat.normalMap = microNormalTex;
    mat.normalScale = new THREE.Vector2(opts.microDepth || 0.35, opts.microDepth || 0.35);
  }
  if (mat.roughness !== undefined && mat.roughness < 0.08) mat.roughness = 0.12;

  const u = {
    uTime: { value: 0 },
    uWind: { value: new THREE.Vector3(0.4, 0, 0.2) },
    uWindAmp: { value: opts.windAmp || 0.0 },
    uSSS: { value: opts.sss || 0.0 },
    uSSSColor: { value: new THREE.Color(opts.sssColor || '#ff6b4a') },
    uEdgeWear: { value: opts.edgeWear !== undefined ? opts.edgeWear : 0.38 },
    uWetness: { value: opts.wetness || 0.0 },
    uParallax: { value: opts.parallax || 0.015 },
    uAOStrength: { value: opts.ao || 0.45 }
  };
  mat.userData.realismUniforms = u;
  R.uniformsList.push(u);

  const prevCompile = mat.onBeforeCompile;
  mat.onBeforeCompile = (shader) => {
    if (prevCompile) prevCompile(shader);
    Object.assign(shader.uniforms, u);

    shader.vertexShader = shader.vertexShader
      .replace('#include <common>', `
        #include <common>
        uniform float uTime;
        uniform vec3 uWind;
        uniform float uWindAmp;
        varying vec3 vWorldPosR;
        varying vec3 vWorldNormR;
      `)
      .replace('#include <begin_vertex>', `
        #include <begin_vertex>
        if (uWindAmp > 0.001) {
          float sway = sin(uTime * 2.6 + position.y * 5.0 + position.x * 3.0) * uWindAmp * max(0.0, position.y + 0.3);
          transformed.x += uWind.x * sway;
          transformed.z += uWind.z * sway;
        }
      `)
      .replace('#include <worldpos_vertex>', `
        #include <worldpos_vertex>
        vec4 wpR = modelMatrix * vec4(transformed, 1.0);
        vWorldPosR = wpR.xyz;
        vWorldNormR = normalize(mat3(modelMatrix) * objectNormal);
      `);

    shader.fragmentShader = shader.fragmentShader
      .replace('#include <common>', `
        #include <common>
        uniform float uTime;
        uniform float uSSS;
        uniform vec3 uSSSColor;
        uniform float uEdgeWear;
        uniform float uWetness;
        uniform float uParallax;
        uniform float uAOStrength;
        varying vec3 vWorldPosR;
        varying vec3 vWorldNormR;
        float hashR(vec3 p) {
          p = fract(p * 0.3183099 + 0.1);
          p *= 17.0;
          return fract(p.x * p.y * p.z * (p.x + p.y + p.z));
        }
      `)
      .replace('#include <roughnessmap_fragment>', `
        #include <roughnessmap_fragment>
        // Surface wetness lowers roughness and creates droplet sheen
        float wetMask = clamp(uWetness, 0.0, 1.0);
        float droplets = smoothstep(0.72, 0.95, sin(vWorldPosR.x * 48.0) * cos(vWorldPosR.z * 48.0 + uTime * 1.2));
        roughnessFactor = mix(roughnessFactor, 0.06, wetMask * (0.75 + 0.25 * droplets));
      `)
      .replace('#include <dithering_fragment>', `
        vec3 viewDirR = normalize(cameraPosition - vWorldPosR);
        float ndv = clamp(abs(dot(normalize(vWorldNormR), viewDirR)), 0.0, 1.0);
        float fresnelR = pow(1.0 - ndv, 3.0);

        // Contact & Crevice Ambient Occlusion
        float floorAO = smoothstep(0.0, 0.28, vWorldPosR.y);
        float creviceAO = mix(1.0 - uAOStrength * 0.45, 1.0, floorAO);
        gl_FragColor.rgb *= creviceAO;

        // Parallax micro-relief shading
        float pNoise = hashR(floor((vWorldPosR + viewDirR * uParallax) * 32.0));
        gl_FragColor.rgb *= (0.94 + pNoise * 0.12);

        // Edge Wear: exposed lighter/metallic edge scratches
        if (uEdgeWear > 0.01) {
          float scratch = smoothstep(0.62, 0.92, hashR(floor(vWorldPosR * 45.0)));
          float wear = fresnelR * uEdgeWear * (0.5 + 0.5 * scratch);
          gl_FragColor.rgb = mix(gl_FragColor.rgb, gl_FragColor.rgb * 1.35 + vec3(0.14), wear * 0.55);
        }

        // Subsurface Scattering (skin, wax, leaves, organic)
        if (uSSS > 0.01) {
          float sssTerm = pow(1.0 - ndv, 2.0) * uSSS;
          gl_FragColor.rgb += uSSSColor * sssTerm * 0.42;
        }

        // Surface Wetness darkening + wet specular rim
        if (uWetness > 0.01) {
          gl_FragColor.rgb *= mix(1.0, 0.68, clamp(uWetness, 0.0, 1.0));
          gl_FragColor.rgb += vec3(0.28, 0.34, 0.42) * fresnelR * clamp(uWetness, 0.0, 1.0) * 0.65;
        }
        #include <dithering_fragment>
      `);
  };
  mat.needsUpdate = true;
  return mat;
};

// Multi-layer Fur & Hair Shell Generator for realistic organic surfaces
R.addFurShells = function(mesh, opts = {}) {
  if (!mesh || !mesh.geometry || mesh.userData._hasFur) return;
  mesh.userData._hasFur = true;
  const layers = opts.layers || 7;
  const length = opts.length || 0.028;
  const baseColor = (mesh.material && mesh.material.color) ? mesh.material.color.clone() : new THREE.Color('#6b4f3a');
  const shellGroup = new THREE.Group();
  shellGroup.name = 'realism_fur_shells';

  for (let i = 1; i <= layers; i++) {
    const t = i / layers;
    const mat = new THREE.MeshStandardMaterial({
      color: baseColor.clone().multiplyScalar(0.55 + t * 0.55),
      alphaMap: furAlphaTex,
      alphaTest: 0.25 + t * 0.55,
      roughness: 0.88,
      metalness: 0.02,
      side: THREE.DoubleSide
    });
    R.enhanceMaterial(mat, { windAmp: 0.06 * t, sss: 0.25, sssColor: '#d9a066', edgeWear: 0.05 });
    const shell = new THREE.Mesh(mesh.geometry, mat);
    const s = 1 + length * t;
    shell.scale.set(s, s, s);
    shell.castShadow = false;
    shell.receiveShadow = true;
    shellGroup.add(shell);
  }
  mesh.add(shellGroup);
};

      v: new THREE.Vector3((Math.random() - 0.5) * 2.4, 1.0 + Math.random() * 2.2, (Math.random() - 0.5) * 2.4),
      r: new THREE.Euler(Math.random() * 6, Math.random() * 6, Math.random() * 6),
      s: 0.5 + Math.random() * 1.1
    });
  }

  APP.onTick('debris_' + Math.random().toString(36).slice(2, 7), 3.2, (dt, t) => {
    const fade = Math.max(0, 1 - t / 3.2);
    for (let i = 0; i < count; i++) {
      const o = pts[i];
      o.v.y -= 9.8 * dt;
      o.p.addScaledVector(o.v, dt);
      if (o.p.y < 0.015) {
        o.p.y = 0.015;
        o.v.y *= -0.35;
        o.v.x *= 0.72;
        o.v.z *= 0.72;
      }
      o.r.x += dt * 4.5;
      o.r.y += dt * 3.2;
      dummy.position.copy(o.p);
      dummy.rotation.copy(o.r);
      dummy.scale.setScalar(o.s * fade);
      dummy.updateMatrix();
      inst.setMatrixAt(i, dummy.matrix);
    }
    inst.instanceMatrix.needsUpdate = true;
  }, () => {
    APP.scene.remove(inst);
    geo.dispose();
    mat.dispose();
  });
};

// Auto-enhance every entity (including AI mode creations)
R.upgradeEntity = function(ent) {
  if (!ent || !ent.root || ent.root.userData._realismUpgraded) return;
  ent.root.userData._realismUpgraded = true;
  const label = (ent.name || '').toLowerCase();
  const isFur = /(fur|hair|cat|dog|wolf|bear|lion|fox|plush|carpet|wool|شعر|فرو|قط|كلب|دب)/.test(label);
  const isOrganic = /(skin|leaf|plant|tree|flower|wax|candle|fruit|human|hand|face|ورق|شمع|نبات|بشرة)/.test(label);
  const isMetal = /(metal|sword|armor|car|robot|iron|steel|gold|bronze|معدن|حديد|سيف|روبوت)/.test(label);
  const isGlass = /(glass|water|ice|crystal|diamond|زجاج|ماء|جليد|كريستال)/.test(label);

  ent.root.traverse((obj) => {
    if (!obj.isMesh || R.enhancedMeshes.has(obj)) return;
    R.enhancedMeshes.add(obj);
    obj.castShadow = true;
    obj.receiveShadow = true;

    const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
    mats.forEach((m) => {
      R.enhanceMaterial(m, {
        sss: isOrganic ? 0.65 : (isFur ? 0.25 : 0.0),
        sssColor: /(leaf|plant|tree|ورق|نبات)/.test(label) ? '#55cc44' : '#ff7755',
        edgeWear: isMetal ? 0.72 : (isOrganic ? 0.12 : 0.35),
        windAmp: (isOrganic || isFur) ? 0.045 : 0.0,
        wetness: isGlass ? 0.35 : R.config.globalWetness,
        parallax: isMetal ? 0.01 : 0.022
      });
    });

    if (isFur && !obj.userData._hasFur) {
      R.addFurShells(obj, { layers: 6, length: 0.024 });
    }
  });

  // Contact Ambient Occlusion Decal under grounded objects
  const box = APP.computeBox ? APP.computeBox(ent.root) : new THREE.Box3().setFromObject(ent.root);
  const size = box.getSize(new THREE.Vector3());
  if (size.x > 0.05 && size.z > 0.05 && size.x < 8) {
    const aoGeo = new THREE.PlaneGeometry(size.x * 1.35, size.z * 1.35);
    const aoMat = new THREE.MeshBasicMaterial({
      map: aoDecalTex, transparent: true, depthWrite: false, opacity: 0.65
    });
    const aoMesh = new THREE.Mesh(aoGeo, aoMat);
    aoMesh.rotation.x = -Math.PI / 2;
    aoMesh.position.set(0, -size.y * 0.48 + 0.004, 0);
    aoMesh.name = 'realism_contact_ao';
    ent.root.add(aoMesh);
  }

  // Caustics projector under glass/water objects
  if (isGlass && R.config.causticsEnabled) {
    const cGeo = new THREE.PlaneGeometry(Math.max(0.6, size.x * 1.8), Math.max(0.6, size.z * 1.8));
    const cMat = new THREE.MeshBasicMaterial({
      map: causticsTex, transparent: true, blending: THREE.AdditiveBlending,
      depthWrite: false, opacity: 0.42, color: '#99eeff'
    });
    const cMesh = new THREE.Mesh(cGeo, cMat);
    cMesh.rotation.x = -Math.PI / 2;
    cMesh.position.set(0, -size.y * 0.48 + 0.008, 0);
    cMesh.userData._isCaustic = true;
    ent.root.add(cMesh);
  }

  if (APP.updateReflection) APP.updateReflection(ent.root.position);
};
