const sleepMs = ms => new Promise(r => setTimeout(r, ms));
function isRecoverableApiError(status, msg = '') {
  return status === 503 || status === 429 || status === 404 || status === 400 ||
    /high demand|overload|quota|rate limit|unavailable|not found|deprecated|decommissioned/i.test(String(msg));
}

function lowercaseSchemaTypes(schema) {
  if (!schema || typeof schema !== 'object') return schema;
  if (Array.isArray(schema)) return schema.map(lowercaseSchemaTypes);
  const out = {};
  for (const [k, v] of Object.entries(schema)) out[k] = (k === 'type' && typeof v === 'string') ? v.toLowerCase() : lowercaseSchemaTypes(v);
  return out;
}
function repairJsonString(str) {
  let out = '', inStr = false, esc = false;
  for (let i = 0; i < str.length; i++) {
    const ch = str[i], code = ch.charCodeAt(0);
    if (!inStr) { if (ch === '"') inStr = true; out += ch; }
    else if (esc) { out += ('"\\/bfnrt'.includes(ch) || (ch === 'u' && /^[0-9a-fA-F]{4}$/.test(str.slice(i + 1, i + 5)))) ? ch : ('\\' + ch); esc = false; }
    else if (ch === '\\') { out += ch; esc = true; }
    else if (ch === '"') { inStr = false; out += ch; }
    else if (ch === '\n') out += '\\n';
    else if (ch === '\r') out += '\\r';
    else if (ch === '\t') out += '\\t';
    else if (code >= 32) out += ch;
  }
  return out;
}
function parseJsonSafe(raw) {
  const cleaned = String(raw || '').trim().replace(/^```json\s*/i, '').replace(/^```\s*/i, '').replace(/```\s*$/i, '').trim();
  try { return JSON.parse(cleaned); }
  catch {
    const first = cleaned.indexOf('{'), last = cleaned.lastIndexOf('}');
    const sub = (first >= 0 && last > first) ? cleaned.slice(first, last + 1) : cleaned;
    return JSON.parse(repairJsonString(sub));
  }
}

async function discoverGeminiModels() {
  if (!aiState.geminiKey) return [];
  if (aiState.discoveredGeminiModels) return aiState.discoveredGeminiModels;
  try {
    const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models?key=${encodeURIComponent(aiState.geminiKey)}`);
    if (r.ok) {
      const d = await r.json();
      const list = (d.models || []).filter(m => (m.supportedGenerationMethods || []).includes('generateContent')).map(m => m.name.replace(/^models\//, '')).filter(n => !/embedding|tts|imagen|aqa|robotics/i.test(n));
      aiState.discoveredGeminiModels = list;
      return list;
    }
  } catch {}
  return ['gemini-4-hydrogen', 'gemini-3.8-flash', 'gemini-3.6-flash', 'gemini-3.5-flash-lite'];
}

function extractGeminiInteractionsText(d) {
  if (!d) return '';
  if (typeof d.output_text === 'string' && d.output_text) return d.output_text;
  if (Array.isArray(d.outputs)) {
    const t = d.outputs.filter(o => typeof o.text === 'string').map(o => o.text).join('\n');
    if (t) return t;
  }
  if (Array.isArray(d.steps)) {
    const t = d.steps.map(s => s?.model_output?.content?.text || '').filter(Boolean).join('\n');
    if (t) return t;
  }
  const cParts = d.candidates?.[0]?.content?.parts || [];
  return cParts.filter(p => typeof p.text === 'string' && !p.thought).map(p => p.text).join('');
}

async function callGeminiDirect(mdl, systemPrompt, userPrompt, thinkLevel = 'high') {
  const safeThink = ['low', 'medium', 'high', 'xhigh'].includes(thinkLevel) ? thinkLevel : 'high';
  // Gemini 4 Hydrogen & 3.8 Flash work natively on /v1beta/interactions!
  if (/^gemini-(4|3\.8)/i.test(mdl)) {
    try {
      const r = await fetch('https://generativelanguage.googleapis.com/v1beta/interactions', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'x-goog-api-key': aiState.geminiKey },
        body: JSON.stringify({
          model: mdl,
          system_instruction: systemPrompt,
          input: userPrompt,
          generation_config: { thinking_level: mdl === 'gemini-4-hydrogen' ? safeThink : (safeThink === 'xhigh' ? 'high' : safeThink), max_output_tokens: 8192 }
        })
      });
      const d = await r.json().catch(() => ({}));
      if (r.ok) {
        const t = extractGeminiInteractionsText(d);
        if (t) return t;
      }
      if (r.status === 401 || r.status === 403) throw new Error(`مفتاح Gemini مرفوض (${r.status}): ${d.error?.message || ''}`);
    } catch (e) { if (/مرفوض/.test(e.message)) throw e; }
  }

  let useThink = true;
  for (let att = 0; att < 2; att++) {
    const body = { systemInstruction: { parts: [{ text: systemPrompt }] }, contents: [{ role: 'user', parts: [{ text: userPrompt }] }] };
    if (useThink && /gemini-(3|4)/i.test(mdl)) body.generationConfig = { thinkingConfig: { thinkingLevel: safeThink === 'xhigh' ? 'high' : safeThink } };
    const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(mdl)}:generateContent`, {
      method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': aiState.geminiKey }, body: JSON.stringify(body)
    });
    const d = await r.json().catch(() => ({}));
    if (r.ok) {
      const parts = d.candidates?.[0]?.content?.parts || [];
      const t = parts.filter(p => typeof p.text === 'string' && !p.thought).map(p => p.text).join('') || parts.map(p => p.text || '').join('');
      if (t) return t;
    }
    if (r.status === 400 && useThink && att === 0) { useThink = false; continue; }
    if ((r.status === 503 || r.status === 429) && att === 0) { await sleepMs(500); continue; }
    const e = new Error(`Gemini (${mdl}) ${r.status}: ${d.error?.message || r.statusText}`); e.status = r.status; throw e;
  }
  throw new Error(`رد فارغ من ${mdl}`);
}

async function getSmartProviderChain(preferredModel) {
  saveCreds();
  const chain = [];
  const add = m => { if (m && !chain.includes(m)) chain.push(m); };
  const hasGemini = aiState.geminiKey.startsWith('AIza');
  const hasOpenAi = aiState.openAiKey.startsWith('sk-');
  const hasClaude = aiState.claudeKey.startsWith('sk-ant-');
  const hasGroq = (aiState.groqKey || '').startsWith('gsk_');

  if (preferredModel?.startsWith('openai:') && !hasOpenAi) throw new Error('أدخل مفتاح ChatGPT (sk-...) في الإعدادات ⚙️');
  if (preferredModel?.startsWith('claude:') && !hasClaude) throw new Error('أدخل مفتاح Claude Haiku (sk-ant-...) في الإعدادات ⚙️');
  if (preferredModel?.startsWith('groq:') && !hasGroq) throw new Error('أدخل مفتاح Groq (gsk_...) في الإعدادات ⚙️');
  if (!hasGemini && !hasOpenAi && !hasClaude && !hasGroq) {
    if ($('aiPanel')) $('aiPanel').classList.add('open');
    if ($('aiSettingsBox')) $('aiSettingsBox').style.display = 'flex';
    throw new Error('أدخل مفتاح Gemini المجاني (AIza...) أو ChatGPT أو Claude Haiku في ⚙️');
  }

  if (preferredModel) add(preferredModel);
  if (hasGemini) ['gemini-4-hydrogen', 'gemini-3.8-flash', 'gemini-3.6-flash', 'gemini-3.5-flash-lite'].forEach(add);
  if (hasOpenAi) ['openai:gpt-6.2-astra', 'openai:gpt-5.4', 'openai:gpt-5-mini'].forEach(add);
  if (hasClaude) ['claude:claude-haiku-5-5', 'claude:claude-haiku-4-5', 'claude:claude-sonnet-5-5'].forEach(add);
  if (hasGroq) ['groq:openai/gpt-oss-120b', 'groq:openai/gpt-oss-20b'].forEach(add);
  return chain;
}

async function callAnyProviderRaw(systemPrompt, userPrompt, preferredModel) {
  const chain = await getSmartProviderChain(preferredModel);
  let lastErr = null;
  for (let i = 0; i < chain.length; i++) {
    const mdl = chain[i];
    try {
      if (i > 0) appendLog(`⚡ تعذر ${chain[i - 1]} — التحويل إلى ${mdl}`, 'bot');
      if (mdl.startsWith('claude:')) {
        const cModel = mdl.slice(7);
        const r = await fetch('https://api.anthropic.com/v1/messages', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'x-api-key': aiState.claudeKey, 'anthropic-version': '2023-06-01', 'anthropic-dangerous-direct-browser-access': 'true' },
          body: JSON.stringify({ model: cModel, max_tokens: 4096, system: systemPrompt, messages: [{ role: 'user', content: userPrompt }] })
        });
        const d = await r.json().catch(() => ({}));
        if (!r.ok) { const e = new Error(`Claude (${cModel}) ${r.status}: ${d.error?.message || r.statusText}`); e.status = r.status; throw e; }
        const t = (d.content || []).filter(b => b.type === 'text').map(b => b.text).join('');
        if (t) return { text: t, modelUsed: mdl };
        throw new Error('رد فارغ من Claude');
      }
      if (mdl.startsWith('openai:') || mdl.startsWith('groq:')) {
        const isOpenAi = mdl.startsWith('openai:');
        const ep = isOpenAi ? 'https://api.openai.com/v1/chat/completions' : 'https://api.groq.com/openai/v1/chat/completions';
        const key = isOpenAi ? aiState.openAiKey : aiState.groqKey;
        const r = await fetch(ep, {
          method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + key },
          body: JSON.stringify({ model: mdl.slice(mdl.indexOf(':') + 1), messages: [{ role: 'system', content: systemPrompt }, { role: 'user', content: userPrompt }] })
        });
        const d = await r.json().catch(() => ({}));
        if (!r.ok) { const e = new Error(`${mdl} (${r.status}): ${d.error?.message || r.statusText}`); e.status = r.status; throw e; }
        const t = d.choices?.[0]?.message?.content || '';
        if (t) return { text: t, modelUsed: mdl };
        throw new Error('رد فارغ من ' + mdl);
      }
      const t = await callGeminiDirect(mdl, systemPrompt, userPrompt, $('aiThink')?.value || 'high');
      return { text: t, modelUsed: mdl };
    } catch (err) {
      lastErr = err;
      if (!isRecoverableApiError(err.status, err.message) && i === chain.length - 1) throw err;
    }
  }
  throw lastErr || new Error('تعذر الاتصال بالنماذج');
}
