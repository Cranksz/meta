function executeToolbarItem(item) {
  if (item.code && item.code.trim()) {
    try { new Function('APP', `return (async () => {\n${item.code}\n})();`)(APP).catch(e => say('❌ ' + e.message)); }
    catch (e) { say('❌ ' + e.message); }
    return;
  }
  if (item.command) APP.matchAndRun(item.command);
  else if (item.ab) APP.runAbility(item.ab, item.ab.matches[0] || item.ab.id);
}

function getEffectiveToolbarItems() {
  const cfg = getToolbarConfig();
  const hidden = new Set((cfg.hiddenIds || []).map(s => String(s).toLowerCase()));
  const renamed = cfg.renamed || {};
  const items = [];

  for (const baseBtn of CORE_BASIC_BUTTONS) {
    if (hidden.has(baseBtn.id.toLowerCase())) continue;
    const rn = renamed[baseBtn.id];
    items.push({ ...baseBtn, isCustom: false, label: rn?.label || baseBtn.label, sub: rn?.sub !== undefined ? rn.sub : baseBtn.sub });
  }
  for (const btn of (cfg.customButtons || [])) {
    if (hidden.has(btn.id.toLowerCase())) continue;
    const rn = renamed[btn.id];
    items.push({
      id: btn.id, isCustom: true,
      label: rn?.label || btn.label || btn.id,
      sub: rn?.sub !== undefined ? rn.sub : (btn.sub || 'Custom'),
      accent: btn.accent || '#00e5ff', bg: btn.bg || '#082026',
      command: btn.command || '', code: btn.code || ''
    });
  }
  for (const ab of (APP.parsedAbilities || [])) {
    if (!ab.pinned || ab.autorun || ab.fallback) continue;
    if (hidden.has(ab.id.toLowerCase())) continue;
    const rn = renamed[ab.id];
    const arHint = (ab.matches.find(m => /[\u0600-\u06FF]/.test(m)) || '').replace(/&/g, ' ');
    items.push({
      id: ab.id, isCustom: false, ab,
      label: rn?.label || arHint || ab.id.replace(/_/g, ' '),
      sub: rn?.sub !== undefined ? rn.sub : (arHint ? ab.id.replace(/_/g, ' ') : ''),
      accent: '#00ff7b', bg: '#0b1a13'
    });
  }
  return items;
}

function renderAbilityChips() {
  const bar = $('abilityChips'); if (!bar) return;
  bar.innerHTML = '';
  for (const item of getEffectiveToolbarItems()) {
    const b = document.createElement('button');
    b.textContent = item.label;
    if (item.isCustom) { b.style.borderColor = item.accent || '#00e5ff'; b.style.background = item.bg || '#09242a'; }
    b.onclick = () => executeToolbarItem(item);
    bar.appendChild(b);
  }
}

function makeBtnTexture(label, accent = '#00ff7b', bg = '#0b1912', sub = '') {
  const c = document.createElement('canvas'); c.width = 256; c.height = 96; const g = c.getContext('2d');
  g.fillStyle = bg; g.fillRect(0, 0, 256, 96); g.strokeStyle = accent; g.lineWidth = 4; g.strokeRect(3, 3, 250, 90);
  g.fillStyle = '#e8fff2'; g.font = 'bold 25px system-ui, Segoe UI, sans-serif'; g.textAlign = 'center'; g.textBaseline = 'middle';
  g.fillText(String(label || '').slice(0, 18), 128, sub ? 38 : 48);
  if (sub) { g.fillStyle = '#8ef5c0'; g.font = '16px monospace'; g.fillText(String(sub).slice(0, 22), 128, 70); }
  const t = new THREE.CanvasTexture(c); t.encoding = THREE.sRGBEncoding; t.anisotropy = 4; return t;
}

const tbRay = new THREE.Raycaster(), tbTmpQ = new Q4(), tbTmpD = new V3(), tbTmpW = new V3();
function tryTriggerXRToolbar(hand) {
  if (!xrToolbarState.visible) return false;
  if (hand.isDesk) {
    if (xrToolbarState.minimized && pointer.ray.intersectObject(xrMiniOrb, false).length) { xrToolbarState.minimized = false; xrPanel.visible = true; xrMiniOrb.visible = false; sfx.play('click'); return true; }
    const tbHit = pointer.ray.intersectObjects(xrToolbarState.buttons.map(b => b.mesh), false)[0];
    if (tbHit?.object?.userData?.xrBtn) { sfx.play('click'); tbHit.object.userData.xrBtn.onClick(); return true; }
    return false;
  }
  const now = NOW();
  if (xrToolbarState.minimized) {
    const orbPos = xrMiniOrb.getWorldPosition(tbTmpW);
    if (hand.tipWp.distanceTo(orbPos) < 0.14 || hand.wp.distanceTo(orbPos) < 0.16) { xrToolbarState.minimized = false; xrPanel.visible = true; xrMiniOrb.visible = false; sfx.play('click'); return true; }
    return false;
  }
  if (!hand.handTrackingActive) {
    hand.c.getWorldPosition(tbTmpW); hand.c.getWorldQuaternion(tbTmpQ);
    tbTmpD.set(0, 0, -1).applyQuaternion(tbTmpQ); tbRay.set(tbTmpW, tbTmpD);
    const hit = tbRay.intersectObjects([tbDragBar, ...xrToolbarState.buttons.map(b => b.mesh)], false)[0];
    if (hit && hit.distance < 2.5) {
      if (hit.object.userData.isXRDragBar) { xrToolbarState.draggingBy = hand; xrToolbarState.dragOffset.copy(xrToolbarGroup.position).sub(hand.wp); sfx.play('click'); return true; }
      if (hit.object.userData.xrBtn) { xrToolbarState.lastPokeAt = now; sfx.play('click'); hit.object.userData.xrBtn.onClick(); return true; }
    }
  }
  return false;
}

stepHooks.set('toolbar_ui', (dt, t, dt0) => {
  if (!xrToolbarState.visible) return;
  if (xrToolbarState.minimized) {
    xrMiniOrb.rotation.y += dt0 * 1.4; xrMiniOrb.rotation.x += dt0 * 0.7;
    for (const h of hands) if (!h.isDesk && h.tipWp.distanceTo(xrMiniOrb.getWorldPosition(tbTmpW)) < 0.08 && NOW() - xrToolbarState.lastPokeAt > 0.5) {
      xrToolbarState.lastPokeAt = NOW(); xrToolbarState.minimized = false; xrPanel.visible = true; xrMiniOrb.visible = false; sfx.play('click'); break;
    }
    return;
  }
  if (xrToolbarState.draggingBy) { xrToolbarGroup.position.copy(xrToolbarState.draggingBy.wp).add(xrToolbarState.dragOffset); xrToolbarGroup.lookAt(headPos.x, headPos.y - 0.08, headPos.z); }
  for (const b of xrToolbarState.buttons) b.hovered = false;
  const now = NOW(), invTargets = xrToolbarState.buttons.map(b => b.mesh);
  for (const h of hands) {
    if (h.isDesk) continue;
    const localTip = xrPanel.worldToLocal(h.tipWp.clone());
    if (Math.abs(localTip.z) < 0.042) {
      for (const b of xrToolbarState.buttons) if (Math.abs(localTip.x - b.x) < b.w * 0.48 && Math.abs(localTip.y - b.y) < b.h * 0.48) {
        b.hovered = true;
        if (now - xrToolbarState.lastPokeAt > 0.52) { xrToolbarState.lastPokeAt = now; sfx.play('click'); b.onClick(); }
      }
    }
    if (!h.handTrackingActive && h.laser?.visible) {
      h.c.getWorldPosition(tbTmpW); h.c.getWorldQuaternion(tbTmpQ); tbTmpD.set(0, 0, -1).applyQuaternion(tbTmpQ); tbRay.set(tbTmpW, tbTmpD);
      const hit = tbRay.intersectObjects(invTargets, false)[0];
      if (hit && hit.distance < 2.5 && hit.object.userData.xrBtn) { hit.object.userData.xrBtn.hovered = true; h.laser.scale.z = hit.distance / 4.0; }
    }
  }
  for (const b of xrToolbarState.buttons) { b.mesh.scale.setScalar(b.hovered ? 1.07 : 1.0); b.mesh.material.color.setHex(b.hovered ? 0xaaffdd : 0xffffff); }
});

APP.toolbar = {
  list() { const cfg = getToolbarConfig(); return { visibleItems: getEffectiveToolbarItems().map(i => ({ id: i.id, label: i.label, sub: i.sub, isCustom: i.isCustom })), customButtons: cfg.customButtons || [], hiddenIds: cfg.hiddenIds || [] }; },
  add({ id, label, sub = '', command = '', code = '', accent = '#00e5ff', bg = '#082026' }) {
    const cfg = getToolbarConfig(); cfg.customButtons = cfg.customButtons || [];
    const cleanId = String(id || ('btn_' + normalizeText(label || command || 'custom').replace(/\s+/g, '_') + '_' + (cfg.customButtons.length + 1))).trim();
    const entry = { id: cleanId, label: String(label || command || cleanId), sub: String(sub || ''), command: String(command || ''), code: String(code || ''), accent, bg };
    const idx = cfg.customButtons.findIndex(b => b.id.toLowerCase() === cleanId.toLowerCase());
    if (idx >= 0) cfg.customButtons[idx] = entry; else cfg.customButtons.push(entry);
    cfg.hiddenIds = (cfg.hiddenIds || []).filter(h => h.toLowerCase() !== cleanId.toLowerCase());
    renderAbilityChips(); rebuildXRToolbar(); persistToolbarConfig(true); return { ok: true, added: entry };
  },
  remove(query) {
    const q = String(query || '').toLowerCase().trim(), cfg = getToolbarConfig();
    cfg.customButtons = cfg.customButtons || []; cfg.hiddenIds = cfg.hiddenIds || [];
    const before = cfg.customButtons.length;
    cfg.customButtons = cfg.customButtons.filter(b => b.id.toLowerCase() !== q && !b.label.toLowerCase().includes(q) && !b.command.toLowerCase().includes(q));
    for (const c of CORE_BASIC_BUTTONS) if (c.id.toLowerCase() === q || c.label.toLowerCase().includes(q)) if (!cfg.hiddenIds.includes(c.id)) cfg.hiddenIds.push(c.id);
    for (const ab of (APP.parsedAbilities || [])) if (ab.id.toLowerCase() === q || ab.id.toLowerCase().includes(q) || ab.matches.some(m => m.toLowerCase().includes(q))) if (!cfg.hiddenIds.includes(ab.id)) cfg.hiddenIds.push(ab.id);
    renderAbilityChips(); rebuildXRToolbar(); persistToolbarConfig(true); return { ok: true, removedCustomCount: before - cfg.customButtons.length };
  },
  show(query) { const q = String(query || '').toLowerCase().trim(), cfg = getToolbarConfig(); cfg.hiddenIds = q === 'all' ? [] : (cfg.hiddenIds || []).filter(id => !id.toLowerCase().includes(q)); renderAbilityChips(); rebuildXRToolbar(); persistToolbarConfig(true); return { ok: true }; },
  rename(query, newLabel, newSub) { const q = String(query || '').toLowerCase().trim(), cfg = getToolbarConfig(); cfg.renamed = cfg.renamed || {}; const target = getEffectiveToolbarItems().find(i => i.id.toLowerCase() === q || i.label.toLowerCase().includes(q)); if (!target) return { ok: false, error: 'Not found' }; cfg.renamed[target.id] = { label: newLabel || target.label, sub: newSub !== undefined ? newSub : target.sub }; renderAbilityChips(); rebuildXRToolbar(); persistToolbarConfig(true); return { ok: true }; },
  clearCustom() { const cfg = getToolbarConfig(); cfg.customButtons = []; cfg.hiddenIds = []; cfg.renamed = {}; renderAbilityChips(); rebuildXRToolbar(); persistToolbarConfig(true); return { ok: true }; },
  openXR() { setXRToolbarVisible(true); return { ok: true }; },
  closeXR() { setXRToolbarVisible(false); return { ok: true }; },
  renderAll() { renderAbilityChips(); rebuildXRToolbar(); }
};
