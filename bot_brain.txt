function uint8ToB64(u8) { let bin = ''; for (let i = 0; i < u8.length; i += 0x8000) bin += String.fromCharCode.apply(null, u8.subarray(i, i + 0x8000)); return btoa(bin); }
async function blobToB64(blob) { return uint8ToB64(new Uint8Array(await blob.arrayBuffer())); }
function b64Pcm16ToFloat32(b64) {
  const bin = atob(b64), len = bin.length, bytes = new Uint8Array(len);
  for (let i = 0; i < len; i++) bytes[i] = bin.charCodeAt(i);
  const view = new DataView(bytes.buffer), samples = Math.floor(len / 2), f32 = new Float32Array(samples);
  for (let i = 0; i < samples; i++) f32[i] = view.getInt16(i * 2, true) / 32768;
  return f32;
}
function downsampleTo16kInt16(float32, inRate) {
  if (inRate === 16000) { const out = new Int16Array(float32.length); for (let i = 0; i < float32.length; i++) { const s = clamp(float32[i], -1, 1); out[i] = s < 0 ? s * 0x8000 : s * 0x7fff; } return out; }
  const ratio = inRate / 16000, newLen = Math.max(1, Math.floor(float32.length / ratio)), out = new Int16Array(newLen);
  for (let i = 0; i < newLen; i++) { const pos = i * ratio, idx = Math.floor(pos), frac = pos - idx, a = float32[idx] || 0, b = float32[Math.min(float32.length - 1, idx + 1)] || a, s = clamp(a + (b - a) * frac, -1, 1); out[i] = s < 0 ? s * 0x8000 : s * 0x7fff; }
  return out;
}

  async startAudioInput() {

    if ($('vuWrap')) $('vuWrap').style.display = 'block';
    const pushSamples = float32 => {
      if (!this.enabled) return;
      let sum = 0; for (let i = 0; i < float32.length; i++) sum += float32[i] * float32[i];

      if ($('vuBar')) $('vuBar').style.width = Math.min(100, Math.round(rms * 1500)) + '%';

      if (now < this.vadCalibUntil) { this.vadNoise = Math.min(0.015, this.vadNoise * 0.85 + rms * 0.15); return; }

      if (rms > vadThr && this.connected && now > this.aiSpeakingUntil && now > this.stateFlashUntil) { if (this.state !== 'user_speaking') this.setState('user_speaking'); }
      else if (rms <= vadThr && this.state === 'user_speaking' && now > this.stateFlashUntil) this.setState('watching');
      if (!this.connected) return;
      if (!this.multimodalTurnMode && this.ws?.readyState === WebSocket.OPEN) {
        const pcm16 = downsampleTo16kInt16(float32, this.inCtx.sampleRate); this.inPcmAccum.push(pcm16); this.inPcmLen += pcm16.length;

        }

  }

        const next = args.action === 'patch' ? base.replace(args.search, args.replace ?? '') : String(args.content ?? '');
        if (/\.txt$/i.test(fn)) { loadedTxtFiles.set(fn, next); parseAbilitiesTxt(getMergedAbilitiesText()); saveFilesCache(); }
        if (aiState.githubToken && args.action !== 'delete') await github.commitDirectToGitHub([{ path: fn, content: next }], `Bot: ${args.action} ${fn}`);
        return { ok: true, file: fn };
      }
      if (name === 'manage_toolbar') {
        const act = String(args.action || '').toLowerCase();
        if (act === 'add' || act === 'add_button') return toolbar.add(args);
        if (act === 'remove' || act === 'remove_button' || act === 'hide_button') return toolbar.remove(args.target || args.id || args.label);
        if (act === 'open_xr') return toolbar.openXR();
        if (act === 'close_xr') return toolbar.closeXR();
        return toolbar.list();
      }
      if (name === 'create_procedural_object') { const r = await createProcedural(args); if (r.ok) this.flashState('learning', '✨ صُنع مجسم!', r.name, 2.5); return r; }
      if (name === 'attach_entity_behavior') return await attachEntityScript(args.target || 'last', args.code || '');
      if (name === 'apply_or_generate_texture') return await applyTexture(args.target || 'last', args);
      if (name === 'switch_ai_model') { $('liveBotModel').value = args.model; this.modelName = args.model; localStorage.setItem('ai_live_model', args.model); if (this.enabled) this.startLiveSession().catch(() => {}); return { ok: true, model: args.model }; }
      if (name === 'move_bot_navigation') { if (args.mode === 'come_here') { const tp = headPos.clone().addScaledVector(headFwd, 0.95); this.setNavMode('stay', tp); this.flyTo(tp, 6.5); setTimeout(() => this.clearFlight(), 900); } else this.setNavMode(args.mode); return { ok: true, mode: this.navMode }; }
      if (name === 'log_user_feedback') { this.memory.learnedRules.push(args.preference_summary); this.saveLocalMemory(); this.scheduleGitHubMemorySync(); return { ok: true }; }
      if (name === 'apply_simulation_adaptation') { if (args.worldG !== undefined) APP.worldG = Number(args.worldG); if (args.worldTS !== undefined) APP.worldTS = Number(args.worldTS); if (args.ragdollMode !== undefined) APP.setRagdollMode?.(Boolean(args.ragdollMode)); if (entities.length) modify({ target: args.target || 'last', ...args }); return { ok: true }; }
      if (name === 'evolve_project_instructions') { this.memory.learnedRules.push(args.memory_rule); this.saveLocalMemory(); this.scheduleGitHubMemorySync(1000); return { ok: true }; }
      if (name === 'inspect_simulation_state') return this.execInspectState();
      if (this.dynamicTools.has(name)) return await this.dynamicTools.get(name).handler(args, this);
      return { ok: false, error: 'Unknown tool' };
    } catch (e) { return { ok: false, error: e.message }; }
  }
  async handleToolCalls(calls) { const resps = []; for (const fc of calls) resps.push({ id: fc.id, name: fc.name, response: await this.executeToolCall(fc.name, fc.args || {}) }); this.sendWs({ toolResponse: { functionResponses: resps } }); }

  execInspectState() {
    return { ok: true, execMode: APP.execMode, model: this.modelName, navMode: this.navMode, importedImages: [...importedImages.keys()], importedGLBs: [...importedGLBs.keys()], activeEntities: entities.map(e => ({ id: e.id, name: e.name, type: e.type })), physics: { worldG: APP.worldG, worldTS: APP.worldTS, ragdollMode: APP.ragdollMode } };
  }
  recordTelemetry() {}

  captureVisionFrameIfDue(activeCam) {
    if (!this.enabled || !this.connected) return;
    const nowMs = performance.now(); if (nowMs - this.lastFrameAt < this.frameIntervalMs) return; this.lastFrameAt = nowMs; this.shutterPulse = 1.0;
    try {
      if (!renderer.xr.isPresenting) { this.visionCtx.drawImage(renderer.domElement, 0, 0, 512, 288); }
      const b64 = this.visionCanvas.toDataURL('image/jpeg', 0.55).split(',')[1];
      if (b64) { this.latestVisionB64 = b64; if (!this.multimodalTurnMode) this.sendWs({ realtimeInput: { video: { mimeType: 'image/jpeg', data: b64 } } }); }
    } catch {}
  }

  step(dt0, t, activeCam) {
    const now = NOW();
    if (this.dissolveTime !== null) {
      const p = clamp(1 - (now - this.dissolveTime) / 0.85, 0, 1); this.rootGroup.scale.setScalar(Math.max(0.01, p));
      if (p <= 0) { this.dissolveTime = null; this.rootGroup.visible = false; this.rootGroup.scale.setScalar(1); }
    } else if (this.enabled) {
      const p = clamp((now - this.birthTime) / 0.65, 0, 1); this.rootGroup.scale.setScalar(Math.max(0.05, 1 - Math.pow(1 - p, 3)));
    }
    if (!this.rootGroup.visible) return;
    const right = new V3().crossVectors(headFwd, new V3(0, 1, 0)).normalize();
    const toBot = this.pos.clone().sub(headPos), botDist = toBot.length(), gazeDot = botDist > 1e-4 ? toBot.clone().divideScalar(botDist).dot(headFwd) : 0, lookingAtBot = gazeDot > 0.66;
    let desiredPos = this.anchorPos.clone(), followSpeed = 3.8;
    if (this.flightTarget) { desiredPos.copy(this.flightTarget.pos); followSpeed = this.flightTarget.speed || 7.2; this.anchorPos.copy(this.pos); }
    else if (this.navMode === 'stay') { desiredPos.copy(this.anchorPos); desiredPos.y += Math.sin(t * 2) * 0.03; }
    else {
      if (!lookingAtBot && (botDist > 2.85 || botDist < 0.48)) { const side = renderer.xr.isPresenting ? 0.6 : 0.42; this.anchorPos.lerp(headPos.clone().addScaledVector(headFwd, 1.28).addScaledVector(right, side), 1 - Math.exp(-2.2 * dt0)); }
      desiredPos.copy(this.anchorPos); desiredPos.y = clamp(desiredPos.y + Math.sin(t * 2.0) * 0.036, 0.65, 2.35);
    }
    const prevPos = this.pos.clone(); this.pos.lerp(desiredPos, 1 - Math.exp(-followSpeed * dt0)); this.pos.addScaledVector(this.recoilVel, dt0); this.recoilVel.multiplyScalar(Math.exp(-6.5 * dt0));
    if (dt0 > 1e-4) this.vel.copy(this.pos).sub(prevPos).multiplyScalar(1 / dt0);
    this.rootGroup.position.copy(this.pos); this.shutterPulse = Math.max(0, this.shutterPulse - dt0 * 3.8);
    if (this.controller?.onStep) { try { this.controller.onStep(dt0, t, { pos: this.pos, vel: this.vel, right, headPos, headFwd, lookingAtBot, navMode: this.navMode, state: this.state, mood: this.mood, inRms: this.inRms, outRms: this.outRms, audioEnergy: Math.max(this.outRms * 5, this.inRms * 4), shutterPulse: this.shutterPulse, inspectPoint: this.inspectPoint, inspectUntil: this.inspectUntil, hands, activeCam }); } catch {} }
    else if (this.fallbackRig.group.visible) { this.fallbackRig.ring.rotation.z += dt0 * 1.4; this.fallbackRig.ring.rotation.y += dt0 * 0.8; }
    this.captureVisionFrameIfDue(activeCam);
  }
}
