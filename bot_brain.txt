function uint8ToB64(u8) { let bin = ''; for (let i = 0; i < u8.length; i += 0x8000) bin += String.fromCharCode.apply(null, u8.subarray(i, i + 0x8000)); return btoa(bin); }
async function blobToB64(blob) { return uint8ToB64(new Uint8Array(await blob.arrayBuffer())); }
function b64Pcm16ToFloat32(b64) {
  const bin = atob(b64), len = bin.length, bytes = new Uint8Array(len);
  for (let i = 0; i < len; i++) bytes[i] = bin.charCodeAt(i);
  const view = new DataView(bytes.buffer), samples = Math.floor(len / 2), f32 = new Float32Array(samples);
  for (let i = 0; i < samples; i++) f32[i] = view.getInt16(i * 2, true) / 32768;
  return f32;
}
function downsampleTo16kInt16(float32, inRate) {
  if (inRate === 16000) { const out = new Int16Array(float32.length); for (let i = 0; i < float32.length; i++) { const s = clamp(float32[i], -1, 1); out[i] = s < 0 ? s * 0x8000 : s * 0x7fff; } return out; }
  const ratio = inRate / 16000, newLen = Math.max(1, Math.floor(float32.length / ratio)), out = new Int16Array(newLen);
  for (let i = 0; i < newLen; i++) { const pos = i * ratio, idx = Math.floor(pos), frac = pos - idx, a = float32[idx] || 0, b = float32[Math.min(float32.length - 1, idx + 1)] || a, s = clamp(a + (b - a) * frac, -1, 1); out[i] = s < 0 ? s * 0x8000 : s * 0x7fff; }
  return out;
}

// Smart Robotic Sphere + Articulated 12-Segment Tentacle Rig
function createSmartRobotRig() {
  const root = new THREE.Group();
  const shellGroup = new THREE.Group();
  root.add(shellGroup);

  const hullMat = new THREE.MeshPhysicalMaterial({ color: 0x1a222a, roughness: 0.24, metalness: 0.86, clearcoat: 0.65 });
  const trimMat = new THREE.MeshPhysicalMaterial({ color: 0x8fa4b8, roughness: 0.16, metalness: 0.94, clearcoat: 0.8 });
  const visorMat = new THREE.MeshPhysicalMaterial({ color: 0x050b10, roughness: 0.06, metalness: 0.2, clearcoat: 1.0 });
  const coreMat = new THREE.MeshStandardMaterial({ color: 0x00ffaa, emissive: 0x00ffaa, emissiveIntensity: 1.9, roughness: 0.2 });
  const ringMat = new THREE.MeshBasicMaterial({ color: 0x00e5ff });

  // Main armored sphere + cheek fins + glowing optic eye
  const hull = new THREE.Mesh(new THREE.SphereGeometry(0.115, 28, 22), hullMat);
  const visor = new THREE.Mesh(new THREE.SphereGeometry(0.092, 24, 18, 0, Math.PI * 2, 0.35, 1.3), visorMat);
  visor.rotation.x = Math.PI * 0.5;
  visor.position.z = 0.036;
  const eyeCore = new THREE.Mesh(new THREE.SphereGeometry(0.032, 18, 14), coreMat);
  eyeCore.position.set(0, 0.005, 0.102);
  const pupil = new THREE.Mesh(new THREE.SphereGeometry(0.015, 12, 10), new THREE.MeshBasicMaterial({ color: 0xffffff }));
  pupil.position.set(0, 0.005, 0.123);

  const halo = new THREE.Mesh(new THREE.TorusGeometry(0.155, 0.0045, 10, 48), ringMat);
  halo.rotation.x = 1.15;
  const collar = new THREE.Mesh(new THREE.TorusGeometry(0.045, 0.011, 10, 24), trimMat);
  collar.rotation.x = Math.PI * 0.5;
  collar.position.y = -0.108;

  for (const sd of [-1, 1]) {
    const ear = new THREE.Mesh(new THREE.CylinderGeometry(0.022, 0.028, 0.026, 16), trimMat);
    ear.rotation.z = Math.PI * 0.5;
    ear.position.set(sd * 0.115, 0.01, 0);
    shellGroup.add(ear);
  }
  shellGroup.add(hull, visor, eyeCore, pupil, halo, collar);

  // Articulated Tentacle (lives in rootGroup local space, independent of shell rotation)
  const tentacleGroup = new THREE.Group();
  root.add(tentacleGroup);
  const N_SEG = 12;
  const segs = [];
  const segDarkMat = new THREE.MeshPhysicalMaterial({ color: 0x1e2830, roughness: 0.32, metalness: 0.85, clearcoat: 0.4 });
  const jointGlowMat = new THREE.MeshStandardMaterial({ color: 0x00e5ff, emissive: 0x00ccff, emissiveIntensity: 1.4, roughness: 0.25 });

  for (let i = 0; i < N_SEG; i++) {
    const k = i / (N_SEG - 1);
    const r = 0.027 * (1 - k * 0.56);
    const node = new THREE.Group();
    const link = new THREE.Mesh(new THREE.CylinderGeometry(r * 0.85, r, 0.068, 12), segDarkMat);
    link.rotation.x = Math.PI * 0.5;
    const ring = new THREE.Mesh(new THREE.SphereGeometry(r * 1.04, 12, 10), i % 2 === 0 ? jointGlowMat : trimMat);
    node.add(link, ring);
    tentacleGroup.add(node);
    segs.push(node);
  }

  // 3-prong mechanical claw at tentacle tip + holographic show laser
  const clawGroup = new THREE.Group();
  tentacleGroup.add(clawGroup);
  const palm = new THREE.Mesh(new THREE.CylinderGeometry(0.016, 0.022, 0.032, 12), trimMat);
  palm.rotation.x = Math.PI * 0.5;
  clawGroup.add(palm);

  const claws = [];
  for (let c = 0; c < 3; c++) {
    const ang = (c / 3) * Math.PI * 2;
    const pivot = new THREE.Group();
    pivot.rotation.z = ang;
    pivot.position.set(Math.cos(ang) * 0.014, Math.sin(ang) * 0.014, 0.015);
    const finger = new THREE.Mesh(new THREE.BoxGeometry(0.008, 0.01, 0.052), jointGlowMat);
    finger.position.set(0, 0.008, 0.026);
    pivot.add(finger);
    clawGroup.add(pivot);
    claws.push(pivot);
  }

  const beamMat = new THREE.MeshBasicMaterial({ color: 0x00ffcc, transparent: true, opacity: 0.45, blending: THREE.AdditiveBlending, depthWrite: false });
  const showBeam = new THREE.Mesh(new THREE.CylinderGeometry(0.003, 0.012, 1.0, 10, 1, true), beamMat);
  showBeam.rotation.x = Math.PI * 0.5;
  showBeam.visible = false;
  tentacleGroup.add(showBeam);

  const hitProxy = new THREE.Mesh(new THREE.SphereGeometry(0.24, 12, 10), new THREE.MeshBasicMaterial({ visible: false }));
  root.add(hitProxy);

  return {
    group: root, shellGroup, tentacleGroup, segs, clawGroup, claws, showBeam,
    orb: hull, ring: halo, coreMat, ringMat, jointGlowMat, hitProxy,
    tipLocal: new V3(0.09, -0.22, 0.11),
    clawOpen: 0.65
  };
}

  async startAudioInput() {

      if (!this.enabled) return;

      let sum = 0; for (let i = 0; i < float32.length; i++) sum += float32[i] * float32[i];

      if (rms > vadThr && this.connected && now > this.stateFlashUntil) { if (this.state !== 'user_speaking') this.setState('user_speaking'); }

  updateTentacleKinematics(dt0, t) {
    const rig = this.rig;
    if (!rig || !rig.group.visible) return;
    const now = NOW();

    // Auto-clear dissolved or removed targets
    if (this.heldEnt && (!this.heldEnt.root.parent || this.heldEnt.dissolveStart)) this.releaseTentacle(false);
    if (this.tentacleTargetEnt && (!this.tentacleTargetEnt.root.parent || this.tentacleTargetEnt.dissolveStart)) {
      this.tentacleTargetEnt = null;
      if (this.tentacleMode !== 'idle') this.tentacleMode = 'idle';
    }
    if (this.tentacleMode === 'show' && now > this.tentacleShowUntil) {
      this.tentacleMode = 'idle';
      this.tentacleTargetEnt = null;
      this.setNavMode('smart');
    }

    // Compute desired tentacle tip in WORLD space, then convert to rootGroup local space
    const baseLocal = new V3(0, -0.11, 0.02);
    let desiredLocalTip = new V3(0.08, -0.22 + Math.sin(t * 2.4) * 0.025, 0.11 + Math.cos(t * 1.9) * 0.025);
    let desiredClawOpen = 0.65;
    let showBeamActive = false;

    if (this.tentacleMode === 'show' && this.tentacleTargetEnt) {
      const targetCenter = computeBox(this.tentacleTargetEnt.root).getCenter(new V3());
      this.tentacleTargetPt.copy(targetCenter);
      const dir = targetCenter.clone().sub(this.pos);
      const dist = dir.length();
      if (dist > 1e-4) {
        const reach = Math.min(0.68, Math.max(0.28, dist * 0.55));
        const worldTip = this.pos.clone().addScaledVector(dir.divideScalar(dist), reach);
        desiredLocalTip.copy(worldTip).sub(this.pos);
        desiredClawOpen = 0.95;
        showBeamActive = true;
      }
    } else if (this.tentacleMode === 'grabbing' && this.tentacleTargetEnt) {
      const targetCenter = computeBox(this.tentacleTargetEnt.root).getCenter(new V3());
      const toObj = targetCenter.clone().sub(this.pos);
      const dist = toObj.length();
      const clampedWorldTip = dist > 0.92 ? this.pos.clone().addScaledVector(toObj.normalize(), 0.92) : targetCenter.clone();
      desiredLocalTip.copy(clampedWorldTip).sub(this.pos);
      desiredClawOpen = dist < 0.85 ? 0.85 : 0.6;

      const tipWorld = this.pos.clone().add(rig.tipLocal);
      if (tipWorld.distanceTo(targetCenter) < 0.22 || dist < 0.72) {
        this.heldEnt = this.tentacleTargetEnt;
        this.heldEnt.heldBy = 'bot_tentacle';
        if (APP._phys) APP._phys.remove(this.heldEnt);
        this.heldLocalOffset.copy(this.heldEnt.root.position).sub(targetCenter);
        if (this._afterGrabHand) {
          this._afterGrabHand = false;
          this.tentacleMode = 'handing';
          this.flyToUserHandPos();
        } else {
          this.tentacleMode = 'holding';
          this.setNavMode('smart');
          this.clearFlight();
        }
        sfx.play('click');
        say(`🤖 أصبحت الكرة تمسك بـ (${this.heldEnt.name})`);
      }
    } else if (this.tentacleMode === 'holding' && this.heldEnt) {
      const right = new V3().crossVectors(headFwd, new V3(0, 1, 0)).normalize();
      desiredLocalTip.copy(right).multiplyScalar(-0.22).add(new V3(0, -0.26, 0)).addScaledVector(headFwd, 0.24);
      desiredClawOpen = 0.14;
    } else if (this.tentacleMode === 'handing' && this.heldEnt) {
      // Extend tentacle toward user's right hand if tracked, otherwise right in front of chest
      const rightHand = hands.find(h => !h.isDesk && h.side === 'right' && (h.handTrackingActive || h.wp.lengthSq() > 0.01));
      const offerWorld = rightHand
        ? rightHand.wp.clone().add(new V3(0, 0.06, 0))
        : headPos.clone().addScaledVector(headFwd, 0.48).add(new V3(0, -0.18, 0));
      const off = offerWorld.clone().sub(this.pos);
      if (off.length() > 0.78) off.setLength(0.78);
      desiredLocalTip.copy(off);
      desiredClawOpen = 0.22;

      // Auto-hand off if user reaches their VR hand close to the offered item
      for (const h of hands) {
        if (!h.isDesk && h.tipWp.distanceTo(this.pos.clone().add(rig.tipLocal)) < 0.14 && h.sqz) {
          const item = this.heldEnt;
          this.releaseTentacle(true);
          if (APP._grabObj) APP._grabObj(h, item);
          break;
        }
      }
      if (this.handAutoRelease && this.pos.distanceTo(headPos) < 1.15) {
        this.handAutoRelease = false;
        setTimeout(() => { if (this.tentacleMode === 'handing') this.releaseTentacle(true); }, 950);
      }
    }

    // Smoothly interpolate tentacle tip and claw angle
    rig.tipLocal.lerp(desiredLocalTip, 1 - Math.exp(-10.5 * dt0));
    rig.clawOpen += (desiredClawOpen - rig.clawOpen) * (1 - Math.exp(-12 * dt0));

    // Cubic Bezier spine from base socket -> natural droop control -> outward approach -> tip
    const p0 = baseLocal;
    const p3 = rig.tipLocal;
    const p1 = baseLocal.clone().add(new V3(0, -0.14, 0.04));
    const mid = p0.clone().lerp(p3, 0.55);
    const wave = (this.tentacleMode === 'idle') ? Math.sin(t * 2.8) * 0.03 : Math.sin(t * 4.5) * 0.01;
    const p2 = mid.add(new V3(wave, -0.06, 0.02));

    const pts = [];
    const nSeg = rig.segs.length;
    for (let i = 0; i <= nSeg; i++) {
      const u = i / nSeg, inv = 1 - u;
      const pt = new V3()
        .addScaledVector(p0, inv * inv * inv)
        .addScaledVector(p1, 3 * inv * inv * u)
        .addScaledVector(p2, 3 * inv * u * u)
        .addScaledVector(p3, u * u * u);
      pts.push(pt);
    }

    for (let i = 0; i < nSeg; i++) {
      const node = rig.segs[i];
      const a = pts[i], b = pts[i + 1];
      node.position.copy(a).lerp(b, 0.5);
      node.lookAt(this.pos.clone().add(b));
    }

    rig.clawGroup.position.copy(p3);
    const tipLook = this.tentacleMode === 'show' && this.tentacleTargetEnt
      ? this.tentacleTargetPt
      : this.pos.clone().add(p3).add(p3.clone().sub(pts[nSeg - 1]));
    rig.clawGroup.lookAt(tipLook);

    for (const pivot of rig.claws) {
      pivot.children[0].rotation.x = -0.55 + rig.clawOpen * 0.72;
    }

    // Holographic show beam when presenting an object
    if (showBeamActive && this.tentacleTargetEnt) {
      const tipWorld = this.pos.clone().add(p3);
      const beamVec = this.tentacleTargetPt.clone().sub(tipWorld);
      const beamLen = Math.max(0.05, beamVec.length());
      rig.showBeam.visible = true;
      rig.showBeam.position.copy(p3).addScaledVector(beamVec.clone().normalize(), beamLen * 0.5);
      rig.showBeam.scale.set(1, 1, beamLen);
      rig.showBeam.lookAt(this.tentacleTargetPt);
      rig.showBeam.material.opacity = 0.35 + Math.sin(t * 10) * 0.14;
    } else {
      rig.showBeam.visible = false;
    }

    // Move held entity with tentacle tip
    if (this.heldEnt && this.heldEnt.heldBy === 'bot_tentacle') {
      const tipWorld = this.pos.clone().add(p3);
      this.heldEnt.root.position.copy(tipWorld).add(this.heldLocalOffset);
      this.heldEnt.root.rotation.y += dt0 * 0.55;
    }
  }

  step(dt0, t) {
    const now = NOW();
    if (this.dissolveTime !== null) {
      const p = clamp(1 - (now - this.dissolveTime) / 0.85, 0, 1); this.rootGroup.scale.setScalar(Math.max(0.01, p));
      if (p <= 0) { this.dissolveTime = null; this.rootGroup.visible = false; this.rootGroup.scale.setScalar(1); }
    } else if (this.enabled) {
      const p = clamp((now - this.birthTime) / 0.65, 0, 1); this.rootGroup.scale.setScalar(Math.max(0.05, 1 - Math.pow(1 - p, 3)));
    }
    if (!this.rootGroup.visible) return;

    const right = new V3().crossVectors(headFwd, new V3(0, 1, 0)).normalize();
    const toBot = this.pos.clone().sub(headPos), botDist = toBot.length(), gazeDot = botDist > 1e-4 ? toBot.clone().divideScalar(botDist).dot(headFwd) : 0, lookingAtBot = gazeDot > 0.66;
    let desiredPos = this.anchorPos.clone(), followSpeed = 3.8;

    if (this.flightTarget) {
      desiredPos.copy(this.flightTarget.pos); followSpeed = this.flightTarget.speed || 7.2;
      if (this.pos.distanceTo(desiredPos) < 0.08) { this.anchorPos.copy(desiredPos); this.clearFlight(); }
    } else if (this.navMode === 'stay') {
      desiredPos.copy(this.anchorPos); desiredPos.y += Math.sin(t * 2) * 0.028;
    } else {
      if (!lookingAtBot && (botDist > 2.85 || botDist < 0.48)) {
        const side = renderer.xr.isPresenting ? 0.58 : 0.42;
        this.anchorPos.lerp(headPos.clone().addScaledVector(headFwd, 1.28).addScaledVector(right, side), 1 - Math.exp(-2.2 * dt0));
      }
      desiredPos.copy(this.anchorPos); desiredPos.y = clamp(desiredPos.y + Math.sin(t * 2.0) * 0.036, 0.65, 2.35);
    }

    const prevPos = this.pos.clone();
    this.pos.lerp(desiredPos, 1 - Math.exp(-followSpeed * dt0));
    this.pos.addScaledVector(this.recoilVel, dt0);
    this.recoilVel.multiplyScalar(Math.exp(-6.5 * dt0));
    if (dt0 > 1e-4) this.vel.copy(this.pos).sub(prevPos).multiplyScalar(1 / dt0);
    this.rootGroup.position.copy(this.pos);
    this.shutterPulse = Math.max(0, this.shutterPulse - dt0 * 3.8);

    if (this.controller?.onStep) {
      try { this.controller.onStep(dt0, t, { pos: this.pos, vel: this.vel, right, headPos, headFwd, lookingAtBot, navMode: this.navMode, state: this.state, mood: this.mood, inRms: this.inRms, outRms: this.outRms, audioEnergy: Math.max(this.outRms * 5, this.inRms * 4), shutterPulse: this.shutterPulse, inspectPoint: this.inspectPoint, inspectUntil: this.inspectUntil, hands }); } catch {}
    } else if (this.rig.group.visible) {
      // Smoothly orient the robotic orb's visor/eye toward target or user
      const lookTarget = (now < this.inspectUntil && this.tentacleTargetEnt) ? this.tentacleTargetPt : headPos;
      this.rig.shellGroup.lookAt(lookTarget);
      this.rig.ring.rotation.z += dt0 * 1.6;
      this.rig.ring.rotation.y += dt0 * 0.9;
      const pulse = 1 + Math.max(this.outRms * 4, this.inRms * 3) * 0.25 + Math.sin(t * 4) * 0.05;
      this.rig.coreMat.emissiveIntensity = 1.6 * pulse;
      this.updateTentacleKinematics(dt0, t);
    }
    this.captureVisionFrameIfDue();
  }
}
