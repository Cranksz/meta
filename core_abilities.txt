function isHtmlEditPrompt(rawText) {
  const s = String(rawText || '');
  return /(\bhtml\b|say-it-see-it\.html|الـ?\s*html|تعديل\s+(?:الصفحة|الواجهة|الكود|التصميم|الشريط)|عد[ّ]?ل\s+(?:الـ?\s*html|الصفحة|الواجهة|الشريط|زر|لون)|أضف\s+زر\s+(?:في|لـ)\s*(?:الصفحة|الواجهة|الشريط)|edit\s+html|patch\s+html|modify\s+(?:the\s+)?(?:html|page|ui|css))/i.test(s);
}

function getRepoInspirations() {
  const chunks = [];
  for (const [fname, content] of loadedTxtFiles.entries()) {
    if (/^(craft_3d|ai_models|toolbar_ui|ai_studio|bot_brain|voice_stt|core_abilities)\.txt$/i.test(fname)) continue;
    // Strip comment banner lines so real craft rules and abilities fit cleanly
    const clean = String(content || '')
      .replace(/^[ \t]*#[ \t]*=+[\s\S]*?\n(?=@ability|\w)/m, '')
      .replace(/^[ \t]*# ?/gm, '')
      .trim();
    if (clean) chunks.push(`--- ${fname} ---\n${clean.slice(0, 2200)}`);
  }
  return chunks.join('\n\n').slice(0, 7500);
}

function matchBestAbility(norm, includeFallback = false) {
  let best = null, bestScore = 0;
  for (const ab of (APP.parsedAbilities || [])) {
    if (ab.autorun) continue;
    if (!includeFallback && ab.fallback) continue;
    if (includeFallback && !ab.fallback) continue;
    for (const pat of ab.matches) {
      const terms = pat.split('&').map(t => normalizeText(t)).filter(Boolean);
      if (terms.length && terms.every(t => norm.includes(t))) {
        const sc = terms.length * 100 + terms.reduce((s, t) => s + t.length, 0);
        if (sc > bestScore) { bestScore = sc; best = ab; }
      }
    }
  }
  return best;
}

APP.runPrescriptByText = function (rawText) {
  const norm = normalizeText(rawText);
  if (!norm) return { ok: false, error: 'Empty command' };
  const ab = matchBestAbility(norm, false) || matchBestAbility(norm, true);
  if (ab) { runAbility(ab, rawText); return { ok: true, ability: ab.id }; }
  return { ok: false, error: 'لم يطابق أي تعليمة مكتوبة في الملفات' };
};

APP.router = function (rawText) {
  const norm = normalizeText(rawText);
  if (!norm) return false;
  const mode = APP.execMode || 'both';

  // MODE 1: INSTRUCTIONS ONLY
  if (mode === 'instructions') {
    const direct = APP.runPrescriptByText(rawText);
    if (direct.ok) return true;
    if (APP.liveCompanion?.enabled) {
      APP.liveCompanion.sendTextFeedback(rawText);
      return true;
    }
    say('📜 وضع التعليمات فقط: لم يطابق أي تعليمة مكتوبة');
    return false;
  }

  // HTML live edit in 'ai' and 'both' modes
  if (isHtmlEditPrompt(rawText)) {
    APP.editHtmlFromPrompt?.(rawText, { forceHtml: true, fromMainTape: true, autoCommitIfToken: Boolean(aiState.githubToken) });
    return true;
  }

  // Imported GLB explicit check
  if (shouldUseGLB(rawText)) {
    const file = resolveGLB(rawText), base = file.replace(/\.(glb|gltf)$/i, '');
    if (importedGLBs.has(file) || importedGLBs.has(base + '.glb')) {
      say('📦 ' + file);
      spawnGLB({ file, name: base, height: 1.4 });
      return true;
    }
  }

  // MODE 2: AI ONLY
  if (mode === 'ai') {
    if (APP.liveCompanion?.enabled) { APP.liveCompanion.sendTextFeedback(rawText); return true; }
    creativeBuilder(rawText, 'ai');
    return true;
  }

  // MODE 3: BOTH — Run matching repo abilities first, otherwise create with AI + instructions
  const matchedAb = matchBestAbility(norm, false);
  if (matchedAb) {
    runAbility(matchedAb, rawText);
    return true;
  }
  if (APP.liveCompanion?.enabled) { APP.liveCompanion.sendTextFeedback(rawText); return true; }
  creativeBuilder(rawText, 'both');
  return true;
};

if (aiState.githubToken && !window.__memPulledOnce) {

}
