function b64ToUtf8(b64) {
  const bin = atob((b64 || '').replace(/\s+/g, '')), bytes = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
  return new TextDecoder().decode(bytes);
}
function utf8ToB64(str) {
  const bytes = new TextEncoder().encode(str); let bin = '';
  for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
  return btoa(bin);
}

async function ghFetchDirect(path, opts = {}) {
  ai.saveCreds();
  if (!aiState.githubToken) throw new Error('أدخل GitHub Token أولاً في الإعدادات (⚙️)');
  const { owner, repo } = getRepoTarget();
  const res = await fetch(`https://api.github.com/repos/${owner}/${repo}${path}`, {
    ...opts, headers: { 'Authorization': 'Bearer ' + aiState.githubToken, 'Accept': 'application/vnd.github+json', ...(opts.body ? { 'Content-Type': 'application/json' } : {}) }
  });
  const txt = await res.text(); let data = {}; try { data = txt ? JSON.parse(txt) : {}; } catch { data = { raw: txt }; }
  if (!res.ok) throw new Error(`GitHub ${res.status}: ${data.message || txt.slice(0, 180)}`);
  return data;
}

async function getGitHubSnapshot() {
  let { branch } = getRepoTarget(), items;
  try { items = await ghFetchDirect(`/contents?ref=${encodeURIComponent(branch)}&t=${Date.now()}`); }
  catch (err) { if (/404/.test(err.message)) { branch = await ensureValidBranch(); items = await ghFetchDirect(`/contents?ref=${encodeURIComponent(branch)}&t=${Date.now()}`); } else throw err; }
  const filesMap = {}, targets = (Array.isArray(items) ? items : []).filter(it => it.type === 'file' && /\.(txt|html?|json|js|css|md)$/i.test(it.name));
  await Promise.all(targets.map(async it => {
    const f = await ghFetchDirect(`/contents/${encodeURIComponent(it.name)}?ref=${encodeURIComponent(branch)}&t=${Date.now()}`);
    filesMap[it.name] = { sha: f.sha, content: b64ToUtf8(f.content) };
  }));
  return filesMap;
}
async function commitDirectToGitHub(filesToWrite, commitMessage, filesToDelete = []) {
  let { branch } = getRepoTarget(), ref;
  try { ref = await ghFetchDirect(`/git/ref/heads/${encodeURIComponent(branch)}?t=${Date.now()}`); }
  catch (err) { if (/404/.test(err.message)) { branch = await ensureValidBranch(); ref = await ghFetchDirect(`/git/ref/heads/${encodeURIComponent(branch)}?t=${Date.now()}`); } else throw err; }
  const headSha = ref.object.sha, headCommit = await ghFetchDirect(`/git/commits/${headSha}`), treeEntries = [];
  for (const f of filesToWrite) {
    const blob = await ghFetchDirect(`/git/blobs`, { method: 'POST', body: JSON.stringify({ content: utf8ToB64(f.content), encoding: 'base64' }) });
    treeEntries.push({ path: f.path, mode: '100644', type: 'blob', sha: blob.sha });
  }
  for (const delPath of filesToDelete) treeEntries.push({ path: delPath, mode: '100644', type: 'blob', sha: null });
  const newTree = await ghFetchDirect(`/git/trees`, { method: 'POST', body: JSON.stringify({ base_tree: headCommit.tree.sha, tree: treeEntries }) });
  const newCommit = await ghFetchDirect(`/git/commits`, { method: 'POST', body: JSON.stringify({ message: commitMessage || 'AI Update', tree: newTree.sha, parents: [headSha] }) });
  await ghFetchDirect(`/git/refs/heads/${encodeURIComponent(branch)}`, { method: 'PATCH', body: JSON.stringify({ sha: newCommit.sha, force: false }) });
  return { sha: newCommit.sha, branch };
}

function applyAbilityEdit(fileText, edit) {
  if (edit.action === 'replace_file' || edit.action === 'create_file') return edit.fullContent ?? '';
  if (edit.action === 'patch_html' || edit.action === 'patch_file') {
    if (!edit.search || !fileText.includes(edit.search)) throw new Error('لم أجد المقطع المطلوب تعديله في ' + edit.file);
    return fileText.replace(edit.search, edit.replace ?? '');
  }
  const { preamble, blocks } = splitAbilityBlocks(fileText), targetId = String(edit.abilityId || '').trim().toLowerCase();
  if (edit.action === 'delete_ability') {
    const filtered = blocks.filter(b => b.id.toLowerCase() !== targetId);
    return [preamble, ...filtered.map(b => b.lines.join('\n').trim())].filter(Boolean).join('\n\n') + '\n';
  }
  if (edit.action === 'upsert_ability') {
    const incoming = splitAbilityBlocks(edit.abilityBlock || '');
    if (!incoming.blocks.length) {
      const newLines = String(edit.abilityBlock || '').replace(/\r\n/g, '\n').trim().split('\n');
      const idx = blocks.findIndex(b => b.id.toLowerCase() === targetId);
      if (idx >= 0) blocks[idx].lines = newLines; else blocks.push({ id: edit.abilityId || 'ab_new', lines: newLines });
    } else {
      for (const nb of incoming.blocks) {
        const key = (nb.id || edit.abilityId || '').trim().toLowerCase(), idx = blocks.findIndex(b => b.id.toLowerCase() === key);
        if (idx >= 0) blocks[idx] = nb; else blocks.push(nb);
      }
    }
    return [preamble, ...blocks.map(b => b.lines.join('\n').trim())].filter(Boolean).join('\n\n') + '\n';
  }
  throw new Error('عملية تعديل غير معروفة: ' + edit.action);
}

function parseAiTeacherOutput(rawText) {
  const txt = String(rawText || '').trim();
  if (/<<<(?:REPLY|EDIT|META|DOM_PATCH)>>>/i.test(txt) || /<<<EDIT\s+/i.test(txt)) {
    const replyM = txt.match(/<<<REPLY>>>([\s\S]*?)<<<END_REPLY>>>/i);
    const metaM = txt.match(/<<<META>>>([\s\S]*?)<<<END_META>>>/i);
    const domM = txt.match(/<<<DOM_PATCH>>>([\s\S]*?)<<<END_DOM_PATCH>>>/i);
    let meta = {}; if (metaM) { try { meta = ai.parseJsonSafe(metaM[1]); } catch {} }
    const edits = [], editRe = /<<<EDIT\s+([^>]+)>>>([\s\S]*?)<<<END_EDIT>>>/gi;
    let m;
    while ((m = editRe.exec(txt)) !== null) {
      const hdr = m[1], body = m[2].replace(/^\r?\n/, '').replace(/\r?\n$/, '');
      const attr = k => { const am = hdr.match(new RegExp(k + '\\s*=\\s*["\']([^"\']+)["\']', 'i')); return am ? am[1].trim() : ''; };
      const file = attr('file').replace(/^\.\//, ''), action = attr('action') || 'upsert_ability';
      let abilityId = attr('abilityId');
      if (action === 'patch_html' || action === 'patch_file') {
        const sM = body.match(/<<<SEARCH>>>\r?\n?([\s\S]*?)\r?\n?<<<END_SEARCH>>>/i);
        const rM = body.match(/<<<REPLACE>>>\r?\n?([\s\S]*?)\r?\n?<<<END_REPLACE>>>/i);
        edits.push({ file, action, search: sM ? sM[1] : '', replace: rM ? rM[1] : '' });
      } else if (action === 'upsert_ability') {
        if (!abilityId) { const idM = body.match(/^@ability\s*:\s*(\S+)/im); if (idM) abilityId = idM[1].trim(); }
        edits.push({ file, action, abilityId, abilityBlock: body });
      } else if (action === 'delete_ability') {
        edits.push({ file, action, abilityId });
      } else {
        edits.push({ file, action, fullContent: body });
      }
    }
    return { reply: (replyM ? replyM[1].trim() : '') || (edits.length ? 'تم تجهيز التعديل بنجاح.' : txt.slice(0, 600)), edits, domPatch: domM ? domM[1].trim() : '', testCommands: Array.isArray(meta.testCommands) ? meta.testCommands : [], commitMessage: meta.commitMessage || '', learnedRule: meta.learnedRule || '' };
  }
  return ai.parseJsonSafe(txt);
}

      if (plan.learnedRule?.trim()) { memory.learnedRules = memory.learnedRules || []; if (!memory.learnedRules.includes(plan.learnedRule.trim())) memory.learnedRules.push(plan.learnedRule.trim()); }
      if (!snap['ai_memory.json'] || plan.learnedRule?.trim()) files.push({ path: 'ai_memory.json', content: JSON.stringify(memory, null, 2) + '\n' });
      const res = await commitDirectToGitHub(files, plan.commitMessage || 'AI Studio update');
      for (const f of files) if (/\.txt$/i.test(f.path)) loadedTxtFiles.set(f.path, f.content);
      saveFilesCache(); if (APP.liveCompanion) APP.liveCompanion.setMemory(memory);
      aiState.sandboxActive = false; aiState.rollbackSnapshot = null; aiState.pendingPlan = null;
      $('aiRollbackBtn').style.display = 'none'; $('aiSandboxBadge').style.display = 'none'; $('aiActionRow').style.display = 'none';
      window.__recentRuntimeErrors = []; $('aiErrCount').textContent = '0';
      ai.appendLog(`✅ تم الحفظ في GitHub! (${res.sha.slice(0, 7)})`, 'bot'); say(`✅ تم الحفظ على GitHub (${res.sha.slice(0, 7)})`);
      await refreshGitHubCommits();
    } catch (e) { ai.appendLog('❌ فشل الحفظ في GitHub: ' + e.message, 'err'); }
    finally { $('aiApproveBtn').disabled = false; $('aiApproveBtn').textContent = '✅ موافقة وحفظ في GitHub'; }
  };
  $('aiCommitSelect').onchange = () => { $('aiRevertCommitBtn').disabled = !$('aiCommitSelect').value; };
  $('aiRefreshCommitsBtn').onclick = refreshGitHubCommits;
  $('aiRevertCommitBtn').onclick = async () => {
    const sha = $('aiCommitSelect').value; if (!sha) return;
    try {
      $('aiRevertCommitBtn').disabled = true; const { branch } = getRepoTarget();
      const targetCommit = await ghFetchDirect(`/commits/${encodeURIComponent(sha)}`);
      const parentSha = targetCommit.parents?.[0]?.sha; if (!parentSha) throw new Error('لا يمكن استرجاع أول التزام');
      const parentCommit = await ghFetchDirect(`/commits/${encodeURIComponent(parentSha)}`);
      const ref = await ghFetchDirect(`/git/ref/heads/${encodeURIComponent(branch)}?t=${Date.now()}`);
      const rev = await ghFetchDirect(`/git/commits`, { method: 'POST', body: JSON.stringify({ message: `Revert ${sha.slice(0, 7)}`, tree: parentCommit.tree.sha, parents: [ref.object.sha] }) });
      await ghFetchDirect(`/git/refs/heads/${encodeURIComponent(branch)}`, { method: 'PATCH', body: JSON.stringify({ sha: rev.sha, force: false }) });
      ai.appendLog(`⏪ تم استرجاع الحالة قبل ${sha.slice(0, 7)}.`, 'bot');
      await fetchAllTxtFromUrl($('txtUrl').value); if (APP.liveCompanion) await APP.liveCompanion.pullMemoryFromGitHub(); await refreshGitHubCommits();
    } catch (e) { ai.appendLog('❌ فشل الاسترجاع: ' + e.message, 'err'); }
  };
}
