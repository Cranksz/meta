function encodeStandaloneWAV(chunks, inRate, outRate = 16000) {
  let total = 0; for (const c of chunks) total += c.length;
  const merged = new Float32Array(total); let off = 0; for (const c of chunks) { merged.set(c, off); off += c.length; }
  let peak = 0; for (let i = 0; i < total; i++) { const a = Math.abs(merged[i]); if (a > peak) peak = a; }
  if (peak < 0.0025) return null;
  const gain = Math.min(20.0, 0.85 / peak), ratio = inRate / outRate, newLen = Math.max(1600, Math.round(total / ratio));
  const buf = new ArrayBuffer(44 + newLen * 2), view = new DataView(buf);
  const ws = (o, s) => { for (let i = 0; i < s.length; i++) view.setUint8(o + i, s.charCodeAt(i)); };
  ws(0, 'RIFF'); view.setUint32(4, 36 + newLen * 2, true); ws(8, 'WAVE'); ws(12, 'fmt ');
  view.setUint32(16, 16, true); view.setUint16(20, 1, true); view.setUint16(22, 1, true);
  view.setUint32(24, outRate, true); view.setUint32(28, outRate * 2, true); view.setUint16(32, 2, true); view.setUint16(34, 16, true);
  ws(36, 'data'); view.setUint32(40, newLen * 2, true);
  for (let i = 0; i < newLen; i++) { const s = clamp(merged[Math.min(total - 1, Math.floor(i * ratio))] * gain, -1, 1); view.setInt16(44 + i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true); }
  return new Blob([buf], { type: 'audio/wav' });
}

function handleSpokenTranscript(text) {
  if (voiceTarget === 'ai') { if ($('aiInput')) $('aiInput').value = text; APP.editHtmlFromPrompt?.(text); }
  else if (APP.liveCompanion?.enabled) { APP.liveCompanion.sendTextFeedback(text); }
  else { if ($('txt')) $('txt').value = text; say('🗣 سمعت: ' + text); APP.matchAndRun(text); }
}

async function transcribeWavBlob(wavBlob) {
  if (!wavBlob) return;
  const key = ($('freeSttKey')?.value || $('aiGroqKey')?.value || '').trim();
  let text = '';
  if (key.startsWith('gsk_')) {
    const lc = $('sttLang')?.value, fd = new FormData();
    fd.append('file', wavBlob, 'cmd.wav'); fd.append('model', 'whisper-large-v3-turbo'); fd.append('response_format', 'json');
    if (lc === 'ar' || lc === 'en') fd.append('language', lc);
    say('🎧 جاري فهم كلامك...');
    const response = await fetch('https://api.groq.com/openai/v1/audio/transcriptions', { method: 'POST', headers: { Authorization: 'Bearer ' + key }, body: fd });
    let data = {}; try { data = await response.json(); } catch {}
    if (!response.ok) throw new Error(data.error?.message || ('Groq HTTP ' + response.status));
    text = (data.text || '').replace(/\[.*?\]|\(.*?\)/g, '').trim();
  } else if (aiState.geminiKey) {
    say('🎧 جاري فهم كلامك...');
    if (APP.liveCompanion?.resolveVoiceInputText) text = await APP.liveCompanion.resolveVoiceInputText(wavBlob);
  } else {
    say('⚠️ استخدم Chrome/Edge للصوت أو أضف مفتاح Groq/Gemini'); return;
  }
  if (!text) { say('🎙 لم أسمع أمراً واضحاً'); return; }
  const now = Date.now(); if (text === lastVoiceText && now - lastVoiceAt < 2500) return;
  lastVoiceText = text; lastVoiceAt = now; handleSpokenTranscript(text);
}

  window._pinAudioNodes = null;
  if (!APP.liveCompanion?.enabled && $('vuWrap')) $('vuWrap').style.display = 'none';
  if ($('mic')) $('mic').textContent = '🎤 صوت';
  $('aiVoiceBtn')?.classList.remove('active');
  APP.rebuildToolbar?.();
}

    micOn = true; if ($('mic')) $('mic').textContent = '⏹ إيقاف'; $('aiVoiceBtn')?.classList.toggle('active', voiceTarget === 'ai'); if ($('vuWrap')) $('vuWrap').style.display = 'block';

stepHooks.set('voice_stt', () => {
  if (!micOn || !analyser || !timeData) return;
  analyser.getFloatTimeDomainData(timeData);
  let sum = 0; for (let i = 0; i < timeData.length; i++) sum += timeData[i] * timeData[i];
  const rms = Math.sqrt(sum / timeData.length), now = NOW();
  if ($('vuBar')) $('vuBar').style.width = Math.min(100, Math.round(rms * 1500)) + '%';
  if (now < vadCalibUntil) { vadNoise = Math.min(0.018, vadNoise * 0.85 + rms * 0.15); return; }
  const thr = Math.max(0.007, vadNoise * 1.9);
  if (rms > thr) { if (!vadSpeaking) { vadSpeaking = true; vadSpeakStart = now; speechChunks = [...preRoll]; say('🎙 يسمع صوتك...'); } vadLastLoud = now; }
  else if (vadSpeaking && (now - vadLastLoud > 0.55 || now - vadSpeakStart > 7.5)) {
    const dur = vadLastLoud - vadSpeakStart; vadSpeaking = false; const chunks = speechChunks; speechChunks = [];
    if (dur >= 0.16 && chunks.length && audioCtx) sttQueue = sttQueue.then(() => transcribeWavBlob(encodeStandaloneWAV(chunks, audioCtx.sampleRate, 16000))).catch(e => say('❌ ' + e.message));
  }
});
