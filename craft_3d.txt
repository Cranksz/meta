function synthesizeProceduralTexture(prompt = 'stone') {
  const p = normalizeText(prompt);
  const c = document.createElement('canvas'); c.width = c.height = 512;
  const g = c.getContext('2d');

  if (/marble|رخام/.test(p)) {
    g.fillStyle = /black|اسود/.test(p) ? '#1c2026' : (/blue|ازرق/.test(p) ? '#cfe6f7' : (/gold|ذهب/.test(p) ? '#f6efe0' : '#eceae4'));
    g.fillRect(0, 0, 512, 512);
    for (let i = 0; i < 42; i++) {
      g.strokeStyle = /gold|ذهب/.test(p) ? 'rgba(212,175,55,0.48)' : (/blue|ازرق/.test(p) ? 'rgba(35,95,155,0.34)' : 'rgba(90,95,105,0.28)');
      g.lineWidth = 1 + Math.random() * 4;
      g.beginPath(); let x = Math.random() * 512, y = 0; g.moveTo(x, y);
      for (let s = 0; s < 12; s++) { x += (Math.random() - 0.5) * 90; y += 512 / 12; g.lineTo(x, y); }
      g.stroke();
    }
  } else if (/water|ocean|sea|aqua|ماء|بحر/.test(p)) {
    const gr = g.createLinearGradient(0, 0, 512, 512);
    gr.addColorStop(0, '#0a4f70'); gr.addColorStop(0.5, '#1986ad'); gr.addColorStop(1, '#0c5c82');
    g.fillStyle = gr; g.fillRect(0, 0, 512, 512);
    for (let i = 0; i < 160; i++) {
      g.strokeStyle = `rgba(170,245,255,${0.12 + Math.random() * 0.25})`; g.lineWidth = 1.5 + Math.random() * 2.5;
      g.beginPath(); g.arc(Math.random() * 512, Math.random() * 512, 12 + Math.random() * 44, 0, Math.PI * 2); g.stroke();
    }
  } else if (/lava|fire|magma|نار|بركان|حمم/.test(p)) {
    g.fillStyle = '#1b0804'; g.fillRect(0, 0, 512, 512);
    for (let i = 0; i < 240; i++) {
      const rx = Math.random() * 512, ry = Math.random() * 512, r = 10 + Math.random() * 34;
      const rg = g.createRadialGradient(rx, ry, 2, rx, ry, r);
      rg.addColorStop(0, 'rgba(255,245,140,0.92)'); rg.addColorStop(0.45, 'rgba(255,90,10,0.75)'); rg.addColorStop(1, 'rgba(40,5,0,0)');
      g.fillStyle = rg; g.beginPath(); g.arc(rx, ry, r, 0, Math.PI * 2); g.fill();
    }
  } else if (/galaxy|space|cosmos|nebula|star|فضاء|مجرة|نجوم/.test(p)) {
    const rg = g.createRadialGradient(256, 256, 15, 256, 256, 300);
    rg.addColorStop(0, '#441a78'); rg.addColorStop(0.5, '#12184a'); rg.addColorStop(1, '#04050e');
    g.fillStyle = rg; g.fillRect(0, 0, 512, 512);
    for (let i = 0; i < 380; i++) {
      g.fillStyle = i % 7 === 0 ? '#77eeff' : (i % 5 === 0 ? '#ff99ee' : '#ffffff');
      g.globalAlpha = 0.3 + Math.random() * 0.7;
      g.beginPath(); g.arc(Math.random() * 512, Math.random() * 512, 0.7 + Math.random() * 2.2, 0, Math.PI * 2); g.fill();
    }
    g.globalAlpha = 1;
  } else if (/wood|bark|plank|خشب/.test(p)) {
    g.drawImage(tex.wood('#6b3f20', '#2f1606', 1).image, 0, 0);
  } else if (/brick|mosaic|tile|طوب|بلاط|فسيفساء/.test(p)) {
    g.fillStyle = '#2d2926'; g.fillRect(0, 0, 512, 512);
    const cols = ['#8f3e2e', '#a44b37', '#7a3426', '#9e533c'], bw = 64, bh = 32;
    for (let y = 0; y < 512; y += bh) {
      const off = ((y / bh) % 2) * (bw / 2);
      for (let x = -bw; x < 512 + bw; x += bw) {
        g.fillStyle = /mosaic|فسيفساء/.test(p) ? `hsl(${Math.floor(Math.random() * 360)},62%,48%)` : cols[Math.floor(Math.random() * cols.length)];
        g.fillRect(x + off + 2, y + 2, bw - 4, bh - 4);
      }
    }
  } else if (/gold|bronze|copper|ذهب|نحاس/.test(p)) {
    const gr = g.createLinearGradient(0, 0, 512, 512);
    gr.addColorStop(0, '#c8962e'); gr.addColorStop(0.5, '#ffe793'); gr.addColorStop(1, '#9d6e18');
    g.fillStyle = gr; g.fillRect(0, 0, 512, 512);
    for (let i = 0; i < 120; i++) { g.strokeStyle = 'rgba(255,255,255,0.14)'; g.beginPath(); g.moveTo(0, Math.random() * 512); g.lineTo(512, Math.random() * 512); g.stroke(); }

      g.beginPath(); g.moveTo(x, y); x += (Math.random() > 0.5 ? 64 : -64); g.lineTo(x, y); y += (Math.random() > 0.5 ? 64 : -64); g.lineTo(x, y); g.stroke();

function buildBumpFromCanvas(srcCanvas) {
  const b = document.createElement('canvas'); b.width = b.height = 256;
  b.getContext('2d').drawImage(srcCanvas, 0, 0, 256, 256);
  const t = new THREE.CanvasTexture(b); t.wrapS = t.wrapT = THREE.RepeatWrapping; return t;
}

async function generateAiTexture(prompt, opts = {}) {
  const cleanPrompt = String(prompt || 'stone texture').trim();
  const slug = (normalizeText(cleanPrompt).replace(/\s+/g, '_').slice(0, 28) || 'gen') + '_' + Math.floor(Math.random() * 999);
  const fileName = `ai_${slug}.png`;

  const imgToCanvasTex = (imgEl, srcTag) => {
    const cv = document.createElement('canvas'); cv.width = cv.height = 512;
    cv.getContext('2d').drawImage(imgEl, 0, 0, 512, 512);
    const t = new THREE.CanvasTexture(cv); t.encoding = THREE.sRGBEncoding; t.wrapS = t.wrapT = THREE.RepeatWrapping; t.anisotropy = 8;
    const bumpTex = buildBumpFromCanvas(cv), url = cv.toDataURL('image/png');
    importedImages.set(fileName.toLowerCase(), { name: fileName, url, tex: t }); renderTape();
    return { ok: true, name: fileName, tex: t, bumpTex, url, source: srcTag || 'ai_image' };
  };

  const gKey = APP.ai?.state?.geminiKey || localStorage.getItem('ai_gemini_key') || '';
  if (gKey && opts.useGemini !== false) {
    try {
      const res = await fetch('https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-image:generateContent', {
        method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': gKey },
        body: JSON.stringify({ contents: [{ role: 'user', parts: [{ text: `Create a clean seamless tileable surface texture of: ${cleanPrompt}` }] }], generationConfig: { responseModalities: ['IMAGE'] } })
      });
      if (res.ok) {
        const data = await res.json();
        const imgPart = (data.candidates?.[0]?.content?.parts || []).find(p => p.inlineData?.data);
        if (imgPart) {
          const dataUrl = `data:${imgPart.inlineData.mimeType || 'image/png'};base64,${imgPart.inlineData.data}`;
          const img = await new Promise((ok, bad) => { const im = new Image(); im.onload = () => ok(im); im.onerror = bad; im.src = dataUrl; });
          return imgToCanvasTex(img, 'gemini_image');
        }
      }
    } catch {}
  }

  const cv = synthesizeProceduralTexture(cleanPrompt);
  const t = new THREE.CanvasTexture(cv); t.encoding = THREE.sRGBEncoding; t.wrapS = t.wrapT = THREE.RepeatWrapping; t.anisotropy = 8;
  const bumpTex = buildBumpFromCanvas(cv), url = cv.toDataURL('image/png');
  importedImages.set(fileName.toLowerCase(), { name: fileName, url, tex: t }); renderTape();
  return { ok: true, name: fileName, tex: t, bumpTex, url, source: 'procedural_texture' };
}

function ensurePlanarUVs(geom) {
  if (!geom || geom.attributes.uv || !geom.attributes.position) return;
  const pos = geom.attributes.position, uv = new Float32Array(pos.count * 2); geom.computeBoundingBox();
  const bb = geom.boundingBox, sz = bb.getSize(new V3()), dx = Math.max(0.001, sz.x), dy = Math.max(0.001, sz.y);
  for (let i = 0; i < pos.count; i++) { uv[i * 2] = (pos.getX(i) - bb.min.x) / dx; uv[i * 2 + 1] = (pos.getY(i) - bb.min.y) / dy; }
  geom.setAttribute('uv', new THREE.BufferAttribute(uv, 2));
}

async function applyTextureToEntity(targetSpec, opts = {}) {
  const ents = findEntities(targetSpec || 'last');
  if (!ents.length) return { ok: false, error: 'لا يوجد مجسم لتطبيق الخامة عليه' };
  let t = null, bumpTex = null, texName = '', source = 'imported';
  if (opts.image_name) { t = findImage(opts.image_name); texName = opts.image_name; }
  if (!t && opts.prompt) { const gen = await generateAiTexture(opts.prompt, opts); t = gen.tex; bumpTex = gen.bumpTex; texName = gen.name; source = gen.source; }
  if (!t) { t = findImage('last'); if (t) texName = 'last_imported_image'; }
  if (!t) { const gen = await generateAiTexture(opts.prompt || 'ornate stone texture'); t = gen.tex; bumpTex = gen.bumpTex; texName = gen.name; source = gen.source; }

  const rx = Number(opts.repeatX || opts.repeat || 1), ry = Number(opts.repeatY || opts.repeat || 1);
  const partRe = opts.part_filter ? new RegExp(opts.part_filter, 'i') : null;
  let updatedMeshes = 0;

  for (const ent of ents) {
    ent.root.traverse(o => {
      if (!o.isMesh || o.isInstancedMesh) return;
      if (partRe && !partRe.test(o.name || '')) return;
      ensurePlanarUVs(o.geometry);
      const tUse = (rx !== 1 || ry !== 1) ? t.clone() : t;
      tUse.wrapS = tUse.wrapT = THREE.RepeatWrapping; tUse.repeat.set(rx, ry); tUse.needsUpdate = true;
      const mats = Array.isArray(o.material) ? o.material : [o.material];
      const newMats = mats.map(m => {
        const cloned = m?.clone ? m.clone() : new THREE.MeshStandardMaterial({ roughness: 0.45, metalness: 0.1 });
        cloned.map = tUse; if (cloned.color) cloned.color.setHex(0xffffff);
        if (bumpTex && 'bumpMap' in cloned) { cloned.bumpMap = bumpTex; cloned.bumpScale = Number(opts.bumpScale ?? 0.015); }
        if (opts.roughness !== undefined && 'roughness' in cloned) cloned.roughness = Number(opts.roughness);
        if (opts.metalness !== undefined && 'metalness' in cloned) cloned.metalness = Number(opts.metalness);
        cloned.userData._mxPatched = false; cloned.needsUpdate = true; return cloned;
      });
      o.material = Array.isArray(o.material) ? newMats : newMats[0];
      updatedMeshes++;
    });
    fx.spawnBox(computeBox(ent.root), 'glow', 20);
  }
  sfx.play('magic');
  return { ok: true, appliedTo: ents.map(e => e.name), texture: texName, source, updatedMeshes };
}

const craft = {
  mat(opts = {}) {
    const tKind = String(opts.tex || '').toLowerCase();
    let map = null;
    if (tKind === 'wood') map = tex.wood(opts.color || '#6b3f20', '#2b1407', opts.repeat || 2);
    else if (tKind === 'metal' || tKind === 'steel' || tKind === 'chrome') map = tex.metal(opts.color || '#8c96a2', opts.repeat || 2);
    else if (tKind === 'stone' || tKind === 'rock' || tKind === 'concrete') map = tex.stone(opts.color || '#6a7478', opts.repeat || 2);
    else if (tKind === 'carbon') map = tex.carbon(opts.repeat || 4);
    else if (tKind === 'ice' || tKind === 'crystal') map = tex.ice(opts.repeat || 2);
    else if (/marble|gold|water|lava|bark|brick|mosaic|galaxy|cyber|neon/.test(tKind)) {
      const cv = synthesizeProceduralTexture(tKind + ' ' + (opts.color || ''));
      map = new THREE.CanvasTexture(cv); map.encoding = THREE.sRGBEncoding;
      map.wrapS = map.wrapT = THREE.RepeatWrapping; map.repeat.set(opts.repeat || 1, opts.repeat || 1); map.anisotropy = 8;
    }
    const bumpMap = (opts.bump !== false) ? tex.bumpNoise(opts.repeat || 2, opts.bumpContrast || 26) : null;
    const isGlass = (opts.opacity !== undefined && opts.opacity < 1) || opts.transmission > 0;
    const m = new THREE.MeshPhysicalMaterial({
      color: new THREE.Color(map ? '#ffffff' : (opts.color || '#cccccc')),
      map, bumpMap, bumpScale: bumpMap ? (opts.bumpScale ?? 0.015) : 0,
      roughness: opts.rough ?? (/metal|gold|steel|chrome/.test(tKind) ? 0.22 : (tKind === 'stone' ? 0.84 : 0.44)),
      metalness: opts.metal ?? (/metal|gold|steel|chrome/.test(tKind) ? 0.86 : 0.08),
      clearcoat: opts.clearcoat ?? (isGlass || tKind === 'marble' || tKind === 'water' ? 0.92 : 0.16),
      clearcoatRoughness: opts.clearcoatRough ?? 0.1,
      ior: opts.ior ?? 1.48, reflectivity: opts.reflectivity ?? 0.85,
      envMapIntensity: opts.env ?? 1.55,
      emissive: opts.emissive ? new THREE.Color(opts.emissive) : new THREE.Color(0x000000),
      emissiveIntensity: opts.emissiveIntensity ?? (opts.emissive ? 1.4 : 0),
      transparent: Boolean(isGlass || opts.additive), opacity: opts.opacity ?? 1,
      wireframe: Boolean(opts.wireframe), side: opts.doubleSide ? THREE.DoubleSide : THREE.FrontSide
    });
    if (map && opts.tint) m.color.set(opts.tint);
    if (opts.additive) { m.blending = THREE.AdditiveBlending; m.depthWrite = false; }
    return m;
  },
  organicSphere(radius = 0.5, segs = 36, noiseAmp = 0.12, freq = 3.2, mat = null, squash = [1, 1, 1]) {
    const g = new THREE.SphereGeometry(radius, segs, Math.round(segs * 0.75));
    const pos = g.attributes.position, v = new V3();
    for (let i = 0; i < pos.count; i++) {
      v.fromBufferAttribute(pos, i);
      const n = Math.sin(v.x * freq * 2.1 + v.y * freq) * Math.cos(v.z * freq * 1.7 - v.x * 1.3) + 0.5 * Math.sin((v.y + v.z) * freq * 2.9);
      const s = 1 + n * noiseAmp;
      v.set(v.x * s * squash[0], v.y * s * squash[1], v.z * s * squash[2]);
      pos.setXYZ(i, v.x, v.y, v.z);
    }
    g.computeVertexNormals();
    return new THREE.Mesh(g, mat || craft.mat({ tex: 'stone', rough: 0.78 }));
  },
  lathe(points2D, segs = 40, mat = null) {
    const pts = points2D.map(([x, y]) => new THREE.Vector2(Math.max(0.0001, x), y));
    const g = new THREE.LatheGeometry(pts, segs); g.computeVertexNormals();
    return new THREE.Mesh(g, mat || craft.mat({ tex: 'marble', rough: 0.25 }));
  },
  tube(points3D, radius = 0.04, segs = 32, mat = null, closed = false) {
    const curve = new THREE.CatmullRomCurve3(points3D.map(p => new V3(p[0], p[1], p[2])), closed);
    return new THREE.Mesh(new THREE.TubeGeometry(curve, segs, radius, 12, closed), mat || craft.mat({ tex: 'metal' }));
  },
  bevelBox(w = 1, h = 0.2, d = 1, bevel = 0.025, mat = null) {
    const s = new THREE.Shape(), hw = w / 2 - bevel, hd = d / 2 - bevel;
    s.moveTo(-hw, -hd); s.lineTo(hw, -hd); s.lineTo(hw, hd); s.lineTo(-hw, hd); s.closePath();
    const g = new THREE.ExtrudeGeometry(s, { depth: Math.max(0.01, h - bevel * 2), bevelEnabled: true, bevelThickness: bevel, bevelSize: bevel, bevelSegments: 3 });
    g.rotateX(-Math.PI / 2); g.center();
    return new THREE.Mesh(g, mat || craft.mat({ tex: 'metal', rough: 0.35 }));
  },
  waterJet(group, origin = [0, 0.8, 0], count = 90, spread = 0.42, speed = 2.6, floorY = 0.08, tickFn = onTick) {
    const mat = craft.mat({ color: '#99eeff', emissive: '#116688', emissiveIntensity: 0.6, rough: 0.08, metal: 0.1, clearcoat: 1, opacity: 0.85 });
    const inst = new THREE.InstancedMesh(new THREE.SphereGeometry(0.016, 8, 6), mat, count);
    inst.userData.skipMatrixShader = true; group.add(inst);
    const drops = Array.from({ length: count }, (_, i) => ({
      p: new V3(origin[0], origin[1], origin[2]),
      v: new V3((Math.random() - 0.5) * spread, speed * (0.75 + Math.random() * 0.45), (Math.random() - 0.5) * spread),
      age: (i / count) * 1.4
    }));
    const dummy = new THREE.Object3D();
    const tid = 'jet_' + Math.random().toString(36).slice(2, 7);
    tickFn(tid, 0, (dt) => {
      for (let i = 0; i < count; i++) {
        const d = drops[i]; d.age += dt;
        d.v.y += APP.worldG * 0.55 * dt; d.p.addScaledVector(d.v, dt);
        if (d.p.y < floorY || d.age > 1.6) {
          d.p.set(origin[0], origin[1], origin[2]);
          d.v.set((Math.random() - 0.5) * spread, speed * (0.75 + Math.random() * 0.45), (Math.random() - 0.5) * spread);
          d.age = 0;
        }
        dummy.position.copy(d.p);
        dummy.scale.set(0.85, 1.35, 0.85);
        dummy.updateMatrix(); inst.setMatrixAt(i, dummy.matrix);
      }
      inst.instanceMatrix.needsUpdate = true;
    });
    return { inst, tid };
  }
};
