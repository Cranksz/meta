const THREE = APP.THREE;
const bodies = [];

APP._phys = {
  bodies,
  register(ent, mode = "rigid", extra = {}) {
    let b = bodies.find(x => x.ent === ent);
    if (b) { b.mode = mode; Object.assign(b, extra); return b; }
    const box = APP.computeBox(ent.root);
    const r = Math.max(0.14, box.getSize(new THREE.Vector3()).length() * 0.34);
    b = Object.assign({
      ent, mode, vel: new THREE.Vector3(), angVel: new THREE.Vector3(),
      radius: r, mass: Math.max(0.5, r * 4), restitution: 0.35,
      onImpact: null, onPoke: null, onFloorSquash: null
    }, extra);
    ent._physBody = b;
    bodies.push(b);
    return b;
  },
  remove(ent) {
    const idx = bodies.findIndex(b => b.ent === ent);
    if (idx >= 0) bodies.splice(idx, 1);
  },
  applyImpulse(ent, vec) {
    const b = bodies.find(x => x.ent === ent) || APP._phys.register(ent, "rigid");
    b.vel.add(vec);
  }
};

// Bridge controller laser pointer trigger & hand touch pokes directly into physics bodies
APP.on("poke", ev => {
  if (!ev || !ev.ent) return;
  const b = bodies.find(x => x.ent === ev.ent);
  if (b) {
    if (b.onPoke) b.onPoke(ev.point, ev.impulse, true);
    else if (ev.ent.grabbable && !ev.ent.lockedInChamber && !ev.ent.heldBy) {
      b.vel.addScaledVector(ev.impulse, 0.72);
      b.angVel.add(new THREE.Vector3((Math.random()-0.5)*2, (Math.random()-0.5)*2, (Math.random()-0.5)*2));
    }
  }
});
APP.on("pokehold", ev => {
  if (!ev || !ev.ent) return;
  const b = bodies.find(x => x.ent === ev.ent);
  if (b && b.onPoke) b.onPoke(ev.point, ev.impulse.clone().multiplyScalar(0.35), false);
});

// Coupled Simulation Loop: Rigid <-> Softbody <-> Voronoi <-> Snow <-> Scanned Room <-> Water
APP.onTick("mod_physics_world", 0, (dt) => {
  if (dt <= 0) return;
  const h = Math.min(dt, 0.032);
  const planes = typeof APP.roomPlanes === "function" ? APP.roomPlanes() : (APP.roomPlanes || []);
  const walls = planes.filter(p => p.kind === "wall");
  const bounds = typeof APP.roomBounds === "function" ? APP.roomBounds() : (APP.roomBounds || []);
  let minX = Infinity, maxX = -Infinity, minZ = Infinity, maxZ = -Infinity;
  for (const p of bounds) {
    if (p.x < minX) minX = p.x; if (p.x > maxX) maxX = p.x;
    if (p.z < minZ) minZ = p.z; if (p.z > maxZ) maxZ = p.z;
  }
  const tank = APP.entities.find(e => e.type === "tank" && !e.dissolveStart);
  const tk = tank ? tank.tankData : null;

  for (let i = bodies.length - 1; i >= 0; i--) {
    if (!bodies[i].ent.root.parent || bodies[i].ent.dissolveStart) bodies.splice(i, 1);
  }

  for (let i = 0; i < bodies.length; i++) {
    const b = bodies[i], ent = b.ent;
    if (ent.heldBy || ent.lockedInChamber || ent._ragdollActive) continue;

    // 1. Gravity + Water Tank Buoyancy
    b.vel.y += APP.worldG * h;
    if (tk && Math.abs(ent.root.position.x - tk.cx) < tk.halfW - 0.06 && Math.abs(ent.root.position.z - tk.cz) < tk.halfD - 0.06) {
      const box = APP.computeBox(ent.root);
      if (box.min.y < tk.waterY) {
        const sub = Math.min(1, (tk.waterY - box.min.y) / Math.max(0.1, box.max.y - box.min.y));
        b.vel.y += (-APP.worldG * 1.45 * sub) * h;
        b.vel.multiplyScalar(1 - 3.5 * h * sub);
        if (!b._inWater) {
          b._inWater = true;
          if (APP._splashWater) APP._splashWater(ent.root.position.x, ent.root.position.z, Math.min(0.06, 0.015 + b.vel.length() * 0.012), true);
          APP.sfx.play("splash");
        }
        if (box.min.y < tk.baseY + 0.02) { ent.root.position.y += (tk.baseY + 0.02 - box.min.y); b.vel.y = Math.max(0, b.vel.y); }
      } else b._inWater = false;
    } else b._inWater = false;

    ent.root.position.addScaledVector(b.vel, h);
    if (b.angVel.lengthSq() > 1e-4) {
      ent.root.rotation.x += b.angVel.x * h;
      ent.root.rotation.y += b.angVel.y * h;
      ent.root.rotation.z += b.angVel.z * h;
      b.angVel.multiplyScalar(0.97);
    }

    // 2. Floor & Ceiling Collisions
    const box = APP.computeBox(ent.root);
    if (box.min.y < 0) {
      const spd = Math.abs(b.vel.y);
      ent.root.position.y += -box.min.y;
      if (b.onImpact && spd > 1.25) b.onImpact(spd, new THREE.Vector3(0, 1, 0), box.getCenter(new THREE.Vector3()));
      b.vel.y = Math.abs(b.vel.y) > 0.35 ? -b.vel.y * b.restitution : 0;
      b.vel.x *= 0.86; b.vel.z *= 0.86;
      if (b.onFloorSquash && spd > 0.45) b.onFloorSquash(spd);
    }

    // 3. Scanned Room Wall Collisions
    const c = box.getCenter(new THREE.Vector3());
    if (walls.length) {
      for (const w of walls) {
        const n = w.normal.clone();
        if (n.dot(APP.headPos.clone().sub(w.pos)) < 0) n.negate();
        const local = c.clone().sub(w.pos).applyQuaternion(w.quat.clone().invert());
        if (Math.abs(local.x) < w.w * 0.58 && Math.abs(local.z) < w.h * 0.58) {
          const signed = c.clone().sub(w.pos).dot(n);
          if (signed < b.radius && signed > -0.7) {
            ent.root.position.addScaledVector(n, b.radius - signed);
            const vn = b.vel.dot(n);
            if (vn < 0) {
              if (b.onImpact && Math.abs(vn) > 1.25) b.onImpact(Math.abs(vn), n, c);
              b.vel.addScaledVector(n, -(1 + b.restitution) * vn);
            }
          }
        }
      }
    } else if (isFinite(minX)) {
      if (box.min.x < minX) { ent.root.position.x += minX - box.min.x; b.vel.x = Math.abs(b.vel.x) * b.restitution; }
      if (box.max.x > maxX) { ent.root.position.x -= box.max.x - maxX; b.vel.x = -Math.abs(b.vel.x) * b.restitution; }
      if (box.min.z < minZ) { ent.root.position.z += minZ - box.min.z; b.vel.z = Math.abs(b.vel.z) * b.restitution; }
      if (box.max.z > maxZ) { ent.root.position.z -= box.max.z - maxZ; b.vel.z = -Math.abs(b.vel.z) * b.restitution; }
    }

    // 4. Direct VR Hand Contact / Strike
    for (const hd of APP.hands) {
      if (hd.isDesk) continue;
      const dist = hd.tipWp.distanceTo(c);
      if (dist < b.radius + 0.08 && hd.vel.length() > 0.3) {
        const push = c.clone().sub(hd.tipWp).normalize();
        const imp = hd.vel.length() * 0.65;
        b.vel.addScaledVector(push, imp);
        if (b.onImpact && imp > 1.3) b.onImpact(imp, push, hd.tipWp);
        if (b.onPoke) b.onPoke(hd.tipWp, hd.vel, false);
      }
    }

    // 5. Object-to-Object Cross-Collisions
    for (let j = i + 1; j < bodies.length; j++) {
      const b2 = bodies[j], ent2 = b2.ent;
      if (ent2.heldBy || ent2.lockedInChamber) continue;
      const c2 = APP.computeBox(ent2.root).getCenter(new THREE.Vector3());
      const diff = c.clone().sub(c2), d = diff.length(), minD = (b.radius + b2.radius) * 0.85;
      if (d < minD && d > 1e-5) {
        diff.divideScalar(d);
        const pen = (minD - d) * 0.5;
        ent.root.position.addScaledVector(diff, pen);
        ent2.root.position.addScaledVector(diff, -pen);
        const relV = b.vel.clone().sub(b2.vel), vn = relV.dot(diff);
        if (vn < 0) {
          const imp = -vn * 0.7;
          b.vel.addScaledVector(diff, imp); b2.vel.addScaledVector(diff, -imp);
          const contact = c.clone().lerp(c2, 0.5);
          if (b.onImpact && Math.abs(vn) > 1.35) b.onImpact(Math.abs(vn), diff, contact);
          if (b2.onImpact && Math.abs(vn) > 1.35) b2.onImpact(Math.abs(vn), diff.clone().negate(), contact);
          if (b.onPoke) b.onPoke(contact, b2.vel, false);
          if (b2.onPoke) b2.onPoke(contact, b.vel, false);
        }
      }
    }
  }
}, null, true);

const THREE = APP.THREE;
const activeRagdolls = new Map();

APP._activateRagdoll = (ent, hand) => {
  if (!ent || !ent.bones || ent.bones.length < 3) return;
  ent._ragdollActive = true;
  if (ent.mixer) ent.mixer.timeScale = 0;

  if (hand && hand.grip) hand.grip.attach(ent.root);

  const jointSims = [];
  ent.root.updateMatrixWorld(true);
  for (const bone of ent.bones) {
    const childBones = bone.children.filter(c => c.isBone);
    if (!childBones.length) continue;

    const localAxis = new THREE.Vector3();
    for (const cb of childBones) localAxis.add(cb.position);
    if (localAxis.lengthSq() < 1e-6) continue;
    localAxis.normalize();

    const restQuat = (bone.userData.restQuat || bone.quaternion).clone();
    const wp = new THREE.Vector3(), cp = new THREE.Vector3();
    bone.getWorldPosition(wp);
    childBones[0].getWorldPosition(cp);
    const len = Math.max(0.04, wp.distanceTo(cp));

    jointSims.push({
      bone, localAxis, restQuat, len,
      tipP: cp.clone(), tipQ: cp.clone(),
      maxAngle: childBones.length > 1 ? 0.42 : 0.72
    });
  }

  activeRagdolls.set(ent, { ent, jointSims, releasedAt: null });
  APP.say("🦴 تم تفعيل Ragdoll للمجسم");
};

APP._releaseRagdoll = (ent) => {
  const rd = activeRagdolls.get(ent);
  if (rd) rd.releasedAt = performance.now() / 1000;
};

APP._stopRagdoll = (ent) => {
  ent._ragdollActive = false;
  activeRagdolls.delete(ent);
};

APP.onTick("mod_ragdoll_sim", 0, (dt) => {
  if (dt <= 0 || !activeRagdolls.size) return;
  const h = Math.min(dt, 0.03), now = performance.now() / 1000;
  const grav = APP.worldG * h * h;
  const wp = new THREE.Vector3(), parentWorldQ = new THREE.Quaternion(), invParentQ = new THREE.Quaternion();
  const desiredWorldDir = new THREE.Vector3(), desiredLocalDir = new THREE.Vector3();
  const deltaQ = new THREE.Quaternion(), targetQ = new THREE.Quaternion();

  for (const [ent, rd] of activeRagdolls.entries()) {
    if (!ent.root.parent || ent.dissolveStart) { activeRagdolls.delete(ent); continue; }
    if (!ent.heldBy && rd.releasedAt && now - rd.releasedAt > 2.2) {
      ent._ragdollActive = false;
      activeRagdolls.delete(ent);
      if (APP._phys) APP._phys.register(ent, "rigid");
      continue;
    }

    ent.root.updateMatrixWorld(true);
    for (const j of rd.jointSims) {
      j.bone.getWorldPosition(wp);

      const vx = (j.tipP.x - j.tipQ.x) * 0.90, vy = (j.tipP.y - j.tipQ.y) * 0.90, vz = (j.tipP.z - j.tipQ.z) * 0.90;
      j.tipQ.copy(j.tipP);
      j.tipP.x += vx; j.tipP.y += vy + grav; j.tipP.z += vz;
      if (j.tipP.y < 0.03) j.tipP.y = 0.03;

      desiredWorldDir.copy(j.tipP).sub(wp);
      const d = desiredWorldDir.length() || 1e-5;
      desiredWorldDir.divideScalar(d);
      j.tipP.copy(wp).addScaledVector(desiredWorldDir, j.len);

      if (j.bone.parent) {
        j.bone.parent.getWorldQuaternion(parentWorldQ);
        invParentQ.copy(parentWorldQ).invert();
        desiredLocalDir.copy(desiredWorldDir).applyQuaternion(invParentQ).normalize();

        const restDirInParent = j.localAxis.clone().applyQuaternion(j.restQuat).normalize();
        deltaQ.setFromUnitVectors(restDirInParent, desiredLocalDir);
        targetQ.copy(deltaQ).multiply(j.restQuat);

        const ang = j.restQuat.angleTo(targetQ);
        if (ang > j.maxAngle) {
          targetQ.copy(j.restQuat).slerp(targetQ, j.maxAngle / ang);
        }
        j.bone.quaternion.slerp(targetQ, Math.min(1, dt * 16));
      }
    }
  }
}, null, true);

const THREE = APP.THREE;

const pointToSeg = (p, a, b, out) => {
  const ab = b.clone().sub(a), l2 = ab.lengthSq();
  const t = l2 > 1e-8 ? Math.min(1, Math.max(0, p.clone().sub(a).dot(ab) / l2)) : 0;
  out.copy(a).addScaledVector(ab, t);
  return { dist: p.distanceTo(out), t };
};

APP._grabObj = (hand, ent) => {
  if (!ent || ent.dissolveStart) return;
  if (ent.heldBy && ent.heldBy !== hand) APP._releaseObj(ent.heldBy);
  hand.held = ent; hand.heldType = ent.type; ent.heldBy = hand;
  ent.lockedInChamber = false;

  if (APP.ragdollMode && (ent.isRigged || (ent.bones && ent.bones.length >= 3)) && APP._activateRagdoll) {
    APP._activateRagdoll(ent, hand);
    APP.emit("entitygrabbed", { hand, ent });
    return;
  }

  if (ent.type === "pen") {
    hand.grip.add(ent.root);
    ent.root.position.set(hand.isDesk ? 0 : -0.008 * (hand.side === "right" ? 1 : -1), hand.isDesk ? 0 : -0.025, hand.isDesk ? -0.04 : -0.055);
    ent.root.rotation.set(hand.isDesk ? -0.2 : -0.55, 0, 0);
  } else {
    hand.grip.attach(ent.root);
  }
  APP.emit("entitygrabbed", { hand, ent });
};

APP._releaseObj = (hand) => {
  if (!hand.held) return;
  const ent = hand.held;
  if (ent._ragdollActive && APP._releaseRagdoll) {
    APP._releaseRagdoll(ent, hand);
  }
  if (ent.root.parent === hand.grip) APP.scene.attach(ent.root);
  ent.heldBy = null; hand.held = null; hand.heldType = null;

  if (ent.type === "pen") {
    ent.root.rotation.x = ent.root.rotation.z = 0;
    ent.baseY = ent.root.position.y;
  } else if (ent.type === "fish" && ent.fishState) {
    ent.fishState.baseHoverY = ent.root.position.y;
    const tank = APP.entities.find(e => e.type === "tank" && !e.dissolveStart);
    if (tank && tank.tankData) {
      const tk = tank.tankData;
      if (Math.abs(ent.root.position.x - tk.cx) < tk.halfW - 0.06 && Math.abs(ent.root.position.z - tk.cz) < tk.halfD - 0.06 && ent.root.position.y < tk.waterY + 0.35) {
        ent.fishState.inWater = true;
        if (APP._splashWater) APP._splashWater(ent.root.position.x, ent.root.position.z, 0.055, true);
        APP.sfx.play("splash");
      }
    }
  } else if (APP._phys && ent.type !== "machine" && ent.type !== "piano" && ent.type !== "door" && ent.type !== "window") {
    const b = APP._phys.register(ent, ent._physBody ? ent._physBody.mode : "rigid");
    if (hand.vel.length() > 0.25) b.vel.copy(hand.vel).multiplyScalar(0.95);
  }
  APP.emit("entityreleased", { hand, ent });
};

APP.on("ragdollchanged", (active) => {
  if (!active) {
    for (const e of APP.entities) {
      if (e._ragdollActive && APP._stopRagdoll) APP._stopRagdoll(e);
