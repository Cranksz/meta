// Reusable butterfly spawner attached to APP
APP._spawnButterfly = (pos, scale = 0.24) => {
  const c = document.createElement("canvas"); c.width = 256; c.height = 256;
  const g = c.getContext("2d");
  const grad = g.createRadialGradient(40, 128, 10, 140, 128, 180);
  grad.addColorStop(0, "#fff078"); grad.addColorStop(0.28, "#ff7b1c"); grad.addColorStop(0.65, "#00d4ff"); grad.addColorStop(1, "#111018");
  g.fillStyle = grad;
  g.beginPath(); g.moveTo(0, 128);
  g.bezierCurveTo(60, 15, 225, 10, 245, 78);
  g.bezierCurveTo(250, 118, 170, 130, 145, 135);
  g.bezierCurveTo(215, 165, 195, 245, 105, 240);
  g.bezierCurveTo(45, 232, 12, 175, 0, 135);
  g.closePath(); g.fill();
  g.strokeStyle = "#111"; g.lineWidth = 5; g.stroke();
  g.fillStyle = "rgba(255,255,255,0.85)";
  for (let i = 0; i < 8; i++) { g.beginPath(); g.arc(195 - i * 8, 55 + i * 22, 4, 0, Math.PI * 2); g.fill(); }
  const tex = new THREE.CanvasTexture(c); tex.encoding = THREE.sRGBEncoding;

  const bRoot = new THREE.Group();
  bRoot.position.copy(pos);
  const bodyMat = new THREE.MeshStandardMaterial({ color: 0x16161c, roughness: 0.45 });
  const body = new THREE.Mesh(new THREE.CapsuleGeometry ? new THREE.CapsuleGeometry(scale * 0.055, scale * 0.36, 6, 10) : new THREE.CylinderGeometry(scale * 0.05, scale * 0.04, scale * 0.42, 10), bodyMat);
  body.rotation.x = Math.PI / 2; bRoot.add(body);
  for (const sd of [-1, 1]) {
    const ant = new THREE.Mesh(new THREE.CylinderGeometry(0.002, 0.003, scale * 0.32, 6), bodyMat);
    ant.position.set(sd * scale * 0.06, scale * 0.1, scale * 0.3); ant.rotation.set(0.7, 0, -sd * 0.3); bRoot.add(ant);
  }
  const wingMat = new THREE.MeshPhysicalMaterial({ map: tex, transparent: true, alphaTest: 0.1, roughness: 0.35, iridescence: 0.8, side: THREE.DoubleSide });
  const wGeoR = new THREE.PlaneGeometry(scale, scale); wGeoR.rotateX(-Math.PI / 2); wGeoR.translate(scale * 0.5, 0, 0);
  const wGeoL = new THREE.PlaneGeometry(scale, scale); wGeoL.rotateX(-Math.PI / 2); wGeoL.scale(-1, 1, 1); wGeoL.translate(-scale * 0.5, 0, 0);
  const wingR = new THREE.Mesh(wGeoR, wingMat), wingL = new THREE.Mesh(wGeoL, wingMat);
  bRoot.add(wingR, wingL);
  APP.scene.add(bRoot);
  const bb = APP.computeBox(bRoot);
  const mxU = APP.applyDissolveShader(bRoot, bb.min.y, bb.max.y, false);

  let phase = Math.random() * 6.28, ang = Math.random() * 6.28, perched = false;
  const home = pos.clone(), vel = new THREE.Vector3();
  const ent = APP.registerEntity({
    name: "butterfly", type: "butterfly", root: bRoot, box: bb, matrixUniforms: mxU, buildDur: 0.7, grabbable: true,
    upd: (dt, t) => {
      if (ent.dissolveStart || ent.heldBy || dt <= 0) return;
      // Check if user makes index pointing gesture (non-desk VR hand with tipWp extended or steady)
      let perchTarget = null;
      for (const h of APP.hands) {
        if (h.isDesk || !h.model || !h.model.root.visible) continue;
        const dist = h.tipWp.distanceTo(bRoot.position);
        const speed = h.vel.length();
        const isPointing = h.tipWp.distanceTo(h.wp) > 0.06 && !h.act;
        if (isPointing && speed < 0.35 && dist < 2.2) { perchTarget = h.tipWp; break; }
      }
      if (perchTarget) {
        const d = bRoot.position.distanceTo(perchTarget);
        if (d < 0.045) {
          perched = true;
          bRoot.position.lerp(perchTarget.clone().add(new THREE.Vector3(0, 0.02, 0)), Math.min(1, dt * 12));
          phase += dt * 3.2;
          const flap = 0.25 + Math.sin(phase) * 0.22;
          wingR.rotation.z = flap; wingL.rotation.z = -flap;
          return;
        } else {
          perched = false;
          const dir = perchTarget.clone().sub(bRoot.position).normalize();
          vel.lerp(dir.multiplyScalar(0.75), dt * 4);
          bRoot.position.addScaledVector(vel, dt);
          bRoot.lookAt(bRoot.position.clone().add(vel));
        }
      } else {
        perched = false;
        ang += dt * (0.7 + Math.sin(t * 0.9) * 0.6);
        const tx = home.x + Math.sin(ang) * 0.85 + Math.cos(t * 1.3) * 0.35;
        const ty = Math.max(0.65, home.y + Math.sin(t * 1.7) * 0.3);
        const tz = home.z + Math.cos(ang * 0.8) * 0.85;
        const desired = new THREE.Vector3(tx, ty, tz).sub(bRoot.position).multiplyScalar(1.6);
        vel.lerp(desired, dt * 2.5);
        bRoot.position.addScaledVector(vel, dt);
        if (vel.lengthSq() > 0.002) bRoot.lookAt(bRoot.position.clone().add(vel));
      }
      phase += dt * (perched ? 3.0 : 18.0);
      const flap = Math.sin(phase) * 0.95;
      wingR.rotation.z = flap; wingL.rotation.z = -flap;
    }
  });
  return ent;
};

// Smart geometric classifier for magic pen strokes
const classifyStroke = (pts) => {
  const right = new THREE.Vector3(1, 0, 0).applyQuaternion(APP.camera.quaternion).setY(0).normalize();
  if (right.lengthSq() < 0.01) right.set(1, 0, 0);
  const up = new THREE.Vector3(0, 1, 0);
  const proj = pts.map(p => ({ x: p.dot(right), y: p.dot(up) }));
  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const p of proj) { if (p.x < minX) minX = p.x; if (p.x > maxX) maxX = p.x; if (p.y < minY) minY = p.y; if (p.y > maxY) maxY = p.y; }
  const W = Math.max(0.05, maxX - minX), H = Math.max(0.05, maxY - minY), aspect = W / H;
  const norm = proj.map(p => ({ x: (p.x - minX) / W, y: (p.y - minY) / H }));
  const closeDist = Math.hypot(norm[0].x - norm[norm.length - 1].x, norm[0].y - norm[norm.length - 1].y);

  // Count center crossings & radial lobes
  let centerCross = 0, topMidDip = 1, botMidMin = 1, topEdgeMax = 0;
  let topLeftMax = 0, topRightMax = 0, midTopMin = 1;
  const rVals = norm.map(p => Math.hypot(p.x - 0.5, p.y - 0.5));
  const rMean = rVals.reduce((a, b) => a + b, 0) / rVals.length;
  let radialPeaks = 0;
  for (let i = 2; i < norm.length - 2; i++) {
    const p = norm[i];
    if (Math.abs(p.x - 0.5) < 0.1 && Math.abs(p.y - 0.5) < 0.25 && Math.sign(norm[i - 1].x - 0.5) !== Math.sign(norm[i + 1].x - 0.5)) centerCross++;
    if (p.y > 0.5 && Math.abs(p.x - 0.5) < 0.14) topMidDip = Math.min(topMidDip, p.y);
    if (p.y > 0.5 && Math.abs(p.x - 0.5) > 0.2) topEdgeMax = Math.max(topEdgeMax, p.y);
    if (Math.abs(p.x - 0.5) < 0.18) botMidMin = Math.min(botMidMin, p.y);
    if (p.x < 0.38) topLeftMax = Math.max(topLeftMax, p.y);
    if (p.x > 0.62) topRightMax = Math.max(topRightMax, p.y);
    if (p.x > 0.38 && p.x < 0.62 && p.y > 0.55) midTopMin = Math.min(midTopMin, p.y);
    if (rVals[i] > rMean * 1.08 && rVals[i] > rVals[i - 1] && rVals[i] > rVals[i + 1]) radialPeaks++;
  }
  const topW = norm.filter(p => p.y > 0.6).reduce((a, p) => Math.max(a, Math.abs(p.x - 0.5)), 0);
  const botW = norm.filter(p => p.y < 0.35).reduce((a, p) => Math.max(a, Math.abs(p.x - 0.5)), 0);

  // 1. Heart: wider at top, clear top-center cleft, pointed bottom
  if (topEdgeMax - topMidDip > 0.12 && topW > botW * 1.2 && botMidMin < 0.18 && closeDist < 0.45) return "heart";
  // 2. Butterfly: multiple center crossings or symmetric wide figure-8
  if (centerCross >= 2 && aspect > 0.95 && radialPeaks >= 3) return "butterfly";
  // 3. Flower: many radial petal peaks around center
  if (radialPeaks >= 4 && aspect > 0.65 && aspect < 1.55) return "flower";
