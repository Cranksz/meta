APP._spawnButterfly = (pos, scale = 0.2) => {
  const c = document.createElement("canvas"); c.width = 256; c.height = 256;
  const g = c.getContext("2d");
  const grad = g.createRadialGradient(40, 128, 10, 140, 128, 180);
  grad.addColorStop(0, "#ffe47a"); grad.addColorStop(0.28, "#eb6b15"); grad.addColorStop(0.68, "#1895c7"); grad.addColorStop(1, "#101014");
  g.fillStyle = grad;
  g.beginPath(); g.moveTo(0, 128);
  g.bezierCurveTo(60, 15, 225, 10, 245, 78); g.bezierCurveTo(250, 118, 170, 130, 145, 135);
  g.bezierCurveTo(215, 165, 195, 245, 105, 240); g.bezierCurveTo(45, 232, 12, 175, 0, 135); g.closePath(); g.fill();
  g.strokeStyle = "#111"; g.lineWidth = 4; g.stroke();
  const tex = new THREE.CanvasTexture(c); tex.encoding = THREE.sRGBEncoding;

  const bRoot = new THREE.Group(); bRoot.position.copy(pos);
  const bodyMat = new THREE.MeshStandardMaterial({ color: 0x15151a, roughness: 0.55 });
  const body = new THREE.Mesh(new THREE.CylinderGeometry(scale * 0.04, scale * 0.032, scale * 0.42, 10), bodyMat);
  body.rotation.x = Math.PI / 2; bRoot.add(body);
  const wingMat = new THREE.MeshPhysicalMaterial({ map: tex, transparent: true, alphaTest: 0.1, roughness: 0.3, iridescence: 0.85, side: THREE.DoubleSide });
  const wingR = new THREE.Mesh(new THREE.PlaneGeometry(scale, scale).rotateX(-Math.PI / 2).translate(scale * 0.5, 0, 0), wingMat);
  const wingL = new THREE.Mesh(new THREE.PlaneGeometry(scale, scale).rotateX(-Math.PI / 2).scale(-1, 1, 1).translate(-scale * 0.5, 0, 0), wingMat);
  bRoot.add(wingR, wingL);
  APP.scene.add(bRoot);
  const bb = APP.computeBox(bRoot), mxU = APP.applyDissolveShader(bRoot, bb.min.y, bb.max.y, false);
  let phase = 0, ang = Math.random() * 6.28, perched = false;
  const home = pos.clone(), vel = new THREE.Vector3();
  const ent = APP.registerEntity({
    name: "butterfly", type: "butterfly", root: bRoot, box: bb, matrixUniforms: mxU, buildDur: 0.7, grabbable: true,
    upd: (dt, t) => {
      if (ent.dissolveStart || ent.heldBy || dt <= 0) return;
      let perchTarget = null;
      for (const h of APP.hands) {
        if (!h.isDesk && h.isPointing && h.vel.length() < 0.35 && h.tipWp.distanceTo(bRoot.position) < 2.2) { perchTarget = h.tipWp; break; }
      }
      if (perchTarget && bRoot.position.distanceTo(perchTarget) < 0.045) {
        perched = true;
        bRoot.position.lerp(perchTarget.clone().add(new THREE.Vector3(0, 0.015, 0)), Math.min(1, dt * 12));
        phase += dt * 3.0;
        wingR.rotation.z = 0.25 + Math.sin(phase) * 0.2; wingL.rotation.z = -wingR.rotation.z;
        return;
      }
      perched = false;
      const target = perchTarget || new THREE.Vector3(home.x + Math.sin(ang += dt * 0.85) * 0.8, Math.max(0.7, home.y + Math.sin(t * 1.6) * 0.25), home.z + Math.cos(ang * 0.75) * 0.8);
      vel.lerp(target.clone().sub(bRoot.position).multiplyScalar(1.6), dt * 3);
      bRoot.position.addScaledVector(vel, dt);
      if (vel.lengthSq() > 0.002) bRoot.lookAt(bRoot.position.clone().add(vel));
      phase += dt * 18.0;
      wingR.rotation.z = Math.sin(phase) * 0.9; wingL.rotation.z = -wingR.rotation.z;
    }
  });
  return ent;
};

const classifyStroke = (pts) => {
  const right = new THREE.Vector3(1, 0, 0).applyQuaternion(APP.camera.quaternion).setY(0).normalize();
  if (right.lengthSq() < 0.01) right.set(1, 0, 0);
  const up = new THREE.Vector3(0, 1, 0);
  const proj = pts.map(p => ({ x: p.dot(right), y: p.dot(up) }));
  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const p of proj) { if (p.x < minX) minX = p.x; if (p.x > maxX) maxX = p.x; if (p.y < minY) minY = p.y; if (p.y > maxY) maxY = p.y; }
  const W = Math.max(0.05, maxX - minX), H = Math.max(0.05, maxY - minY), aspect = W / H;
  const norm = proj.map(p => ({ x: (p.x - minX) / W, y: (p.y - minY) / H }));
  const closeDist = Math.hypot(norm[0].x - norm[norm.length - 1].x, norm[0].y - norm[norm.length - 1].y);

  let centerCross = 0, topMidDip = 1, botMidMin = 1, topEdgeMax = 0, topLeftMax = 0, topRightMax = 0, midTopMin = 1, radialPeaks = 0;
  const rVals = norm.map(p => Math.hypot(p.x - 0.5, p.y - 0.5)), rMean = rVals.reduce((a, b) => a + b, 0) / rVals.length;
  for (let i = 2; i < norm.length - 2; i++) {
    const p = norm[i];
    if (Math.abs(p.x - 0.5) < 0.1 && Math.abs(p.y - 0.5) < 0.25 && Math.sign(norm[i - 1].x - 0.5) !== Math.sign(norm[i + 1].x - 0.5)) centerCross++;
    if (p.y > 0.5 && Math.abs(p.x - 0.5) < 0.14) topMidDip = Math.min(topMidDip, p.y);
    if (p.y > 0.5 && Math.abs(p.x - 0.5) > 0.2) topEdgeMax = Math.max(topEdgeMax, p.y);
    if (Math.abs(p.x - 0.5) < 0.18) botMidMin = Math.min(botMidMin, p.y);
    if (p.x < 0.38) topLeftMax = Math.max(topLeftMax, p.y);
    if (p.x > 0.62) topRightMax = Math.max(topRightMax, p.y);
    if (p.x > 0.38 && p.x < 0.62 && p.y > 0.55) midTopMin = Math.min(midTopMin, p.y);
    if (rVals[i] > rMean * 1.08 && rVals[i] > rVals[i - 1] && rVals[i] > rVals[i + 1]) radialPeaks++;
  }
  const topW = norm.filter(p => p.y > 0.6).reduce((a, p) => Math.max(a, Math.abs(p.x - 0.5)), 0);
  const botW = norm.filter(p => p.y < 0.35).reduce((a, p) => Math.max(a, Math.abs(p.x - 0.5)), 0);

  if (topEdgeMax - topMidDip > 0.12 && topW > botW * 1.2 && botMidMin < 0.18 && closeDist < 0.45) return "heart";
  if (centerCross >= 2 && aspect > 0.95 && radialPeaks >= 3) return "butterfly";
  if (radialPeaks >= 4 && aspect > 0.65 && aspect < 1.55) return "flower";
  if (topLeftMax > 0.78 && topRightMax > 0.78 && midTopMin < Math.min(topLeftMax, topRightMax) - 0.12) return "cat";
  if (topMidDip > 0.85 && botW > 0.32 && aspect > 0.65 && aspect < 1.5) return "house";
  return "fish";
};

const distortedGlassMat = new THREE.ShaderMaterial({
  uniforms: glassUniforms,
  transparent: true,
  depthWrite: false,
  side: THREE.FrontSide,
  vertexShader: `
    varying vec3 vNormalW;
    varying vec3 vPosW;
    void main() {
      vec4 wp = modelMatrix * vec4(position, 1.0);
      vPosW = wp.xyz;
      vNormalW = normalize(mat3(modelMatrix) * normal);
      gl_Position = projectionMatrix * viewMatrix * wp;
    }
  `,
  fragmentShader: `
    uniform samplerCube uEnv;
    uniform float uTime;
    varying vec3 vNormalW;
    varying vec3 vPosW;
    void main() {
      vec3 V = normalize(vPosW - cameraPosition);
      // Subtle organic glass surface ripple distortion
      vec3 ripple = vec3(
        sin(vPosW.y * 18.0 + vPosW.x * 9.0 + uTime * 0.6),
        cos(vPosW.z * 16.0 - vPosW.y * 11.0),
        sin(vPosW.x * 17.0 + vPosW.z * 10.0)
