const glassUniforms = { uEnv: { value: envCube }, uTime: { value: 0 } };
const glassMat = new THREE.ShaderMaterial({
  uniforms: glassUniforms, transparent: true, depthWrite: false, side: THREE.FrontSide,
  vertexShader: `
    varying vec3 vNormalW; varying vec3 vPosW;
    void main() {
      vec4 wp = modelMatrix * vec4(position, 1.0);
      vPosW = wp.xyz; vNormalW = normalize(mat3(modelMatrix) * normal);
      gl_Position = projectionMatrix * viewMatrix * wp;
    }`,
  fragmentShader: `
    uniform samplerCube uEnv; uniform float uTime;
    varying vec3 vNormalW; varying vec3 vPosW;
    void main() {
      vec3 V = normalize(vPosW - cameraPosition);
      vec3 ripple = vec3(
        sin(vPosW.y * 24.0 + vPosW.x * 11.0),
        cos(vPosW.z * 21.0 - vPosW.y * 13.0),
        sin(vPosW.x * 22.0 + vPosW.z * 12.0)
      ) * 0.09;
      vec3 N = normalize(vNormalW + ripple);
      float fresnel = pow(1.0 - max(0.0, dot(-V, N)), 2.8);
      vec3 reflCol = textureCube(uEnv, reflect(V, N)).rgb;
      vec3 refrCol = vec3(
        textureCube(uEnv, refract(V, N, 0.64)).r,
        textureCube(uEnv, refract(V, N, 0.665)).g,
        textureCube(uEnv, refract(V, N, 0.69)).b
      );
      vec3 col = mix(refrCol * 0.92 + vec3(0.02, 0.05, 0.07), reflCol * 1.35, clamp(fresnel * 1.15 + 0.16, 0.0, 1.0));
      gl_FragColor = vec4(col, clamp(0.22 + fresnel * 0.74, 0.2, 0.95));
    }`
});
ent.root.traverse(o => { if (o.isMesh) { o.material = glassMat; o.userData._mxPatched = true; } });

  if (!ent.root.parent || ent.dissolveStart) { APP.stopTick("glass_env_" + ent.id); return; }
  glassUniforms.uTime.value = t;
});
