APP._applyCinematicMaterials = (ent) => {
  if (!ent || !ent.root || ent.dissolveStart) return false;
  const center = APP.computeBox(ent.root).getCenter(new THREE.Vector3());
  const env = APP.updateReflection(center);
  ent.root.traverse(o => {
    if (!o.isMesh || !o.geometry) return;
    o.castShadow = true; o.receiveShadow = true;
    const upgrade = (src) => {
      if (!src) return src;
      const m = src.clone();
      // Preserve authored albedo, UV maps, normal maps and material identity.
      if (!m.isMeshPhysicalMaterial) {
        const physical = new THREE.MeshPhysicalMaterial();
        physical.copy(m);
        physical.clearcoat = 0.18;
        physical.clearcoatRoughness = 0.32;
        return physical;
      }
      m.clearcoat = Math.max(m.clearcoat || 0, 0.12);
      m.clearcoatRoughness = Math.max(m.clearcoatRoughness || 0, 0.22);
      return m;
    };
    o.material = Array.isArray(o.material) ? o.material.map(upgrade) : upgrade(o.material);
    const mats = Array.isArray(o.material) ? o.material : [o.material];
    for (const m of mats) {
      if (!m) continue;
      if ('envMap' in m && env) { m.envMap = env; m.envMapIntensity = Math.min(1.35, Math.max(0.35, m.envMapIntensity || 0.8)); }
      for (const key of ['map','normalMap','roughnessMap','metalnessMap','aoMap']) {
        const tx = m[key];
        if (tx) { tx.anisotropy = Math.max(tx.anisotropy || 1, 8); tx.needsUpdate = true; }
      }
      // Avoid turning organic surfaces into chrome; retain deliberate metal.
      if (m.metalness !== undefined && m.metalness < 0.35) m.roughness = Math.max(0.28, m.roughness || 0.5);
      m.needsUpdate = true;
    }
  });
  ent.root.updateMatrixWorld(true);
  return true;
};

const pointToSeg = (p, a, b, out) => {
  const ab = b.clone().sub(a), l2 = ab.lengthSq();
  const t = l2 > 1e-8 ? Math.min(1, Math.max(0, p.clone().sub(a).dot(ab) / l2)) : 0;
  out.copy(a).addScaledVector(ab, t);
  return { dist: p.distanceTo(out), t };
};

APP._grabObj = (hand, ent) => {
  if (!ent || ent.dissolveStart) return;
  if (ent.heldBy && ent.heldBy !== hand) APP._releaseObj(ent.heldBy);
  hand.held = ent; hand.heldType = ent.type; ent.heldBy = hand;
  if (ent.type === "pen") {
    hand.grip.add(ent.root);
    ent.root.position.set(hand.isDesk ? 0 : -0.008 * (hand.side === "right" ? 1 : -1), hand.isDesk ? 0 : -0.025, hand.isDesk ? -0.04 : -0.055);
    ent.root.rotation.set(hand.isDesk ? -0.2 : -0.55, 0, 0);
  } else {
    hand.grip.attach(ent.root);
  }
  APP.emit("entitygrabbed", { hand, ent });
};

APP._releaseObj = (hand) => {
  if (!hand.held) return;
  const ent = hand.held;
  if (ent.root.parent === hand.grip) APP.scene.attach(ent.root);
  ent.heldBy = null; hand.held = null; hand.heldType = null;
  if (ent.type === "pen") {
    ent.root.rotation.x = ent.root.rotation.z = 0;
    ent.baseY = ent.root.position.y;
  } else if (ent.type === "fish" && ent.fishState) {
    ent.fishState.baseHoverY = ent.root.position.y;
    const tank = APP.entities.find(e => e.type === "tank" && !e.dissolveStart);
    if (tank && tank.tankData) {
      const tk = tank.tankData;
      if (Math.abs(ent.root.position.x - tk.cx) < tk.halfW - 0.06 && Math.abs(ent.root.position.z - tk.cz) < tk.halfD - 0.06 && ent.root.position.y < tk.waterY + 0.35) {

        if (APP._splashWater) APP._splashWater(ent.root.position.x, ent.root.position.z, 0.055, true);
        APP.sfx.play("splash");

  }
  APP.emit("entityreleased", { hand, ent });

const releaseCloth = (hand) => {
  if (hand.clothGrab && hand.clothGrab.flag && hand.clothGrab.flag.clothData) {
    const { P, Q } = hand.clothGrab.flag.clothData;
    for (const gi of hand.clothGrab.nodes) {
      const k = gi.idx * 3;
      Q[k] = P[k] - hand.vel.x * 0.012;
      Q[k + 1] = P[k + 1] - hand.vel.y * 0.012;
      Q[k + 2] = P[k + 2] - hand.vel.z * 0.012;
    }
  }
  hand.clothGrab = null;
};

const tryGrab = (hand) => {
  if (hand.held) return;
  const hp = hand.wp, tp = hand.tipWp, tmpC = new THREE.Vector3();
  let best = null, bestD = 0.34;
  for (const ent of APP.entities) {
    if (!ent.grabbable || ent.dissolveStart) continue;
    let d = Infinity;
    if (ent.type === "stroke" && ent.strokeObj) {
      const WP = ent.strokeObj.worldPts;
      for (let i = 0; i < WP.length - 1; i++) d = Math.min(d, pointToSeg(hp, WP[i], WP[i + 1], tmpC).dist, pointToSeg(tp, WP[i], WP[i + 1], tmpC).dist);
    } else {
      const box = APP.computeBox(ent.root);
      d = Math.min(box.distanceToPoint(hp), box.distanceToPoint(tp));
    }
    if (d < bestD) { bestD = d; best = ent; }
  }
  if (best) { APP._grabObj(hand, best); return; }

  const clothEnt = APP.entities.find(e => e.type === "cloth" && !e.dissolveStart);
  if (clothEnt && clothEnt.clothData) {
    const { P, N } = clothEnt.clothData;
    let minD = Infinity, bestIdx = -1, anchor = hp, useTip = false;
    for (let i = 0; i < N; i++) {
      const k = i * 3;
      const dP = Math.hypot(P[k] - hp.x, P[k + 1] - hp.y, P[k + 2] - hp.z);
      const dT = Math.hypot(P[k] - tp.x, P[k + 1] - tp.y, P[k + 2] - tp.z);
      if (dP < minD) { minD = dP; bestIdx = i; anchor = hp; useTip = false; }
      if (dT < minD) { minD = dT; bestIdx = i; anchor = tp; useTip = true; }
    }
    if (minD < 0.42 && bestIdx >= 0) {
      const nodes = [], bx = P[bestIdx * 3], by = P[bestIdx * 3 + 1], bz = P[bestIdx * 3 + 2];
      for (let i = 0; i < N; i++) {
        const k = i * 3, dd = Math.hypot(P[k] - bx, P[k + 1] - by, P[k + 2] - bz);
        if (dd < 0.34) nodes.push({ idx: i, ox: P[k] - anchor.x, oy: P[k + 1] - anchor.y, oz: P[k + 2] - anchor.z, w: i === bestIdx ? 1 : Math.pow(1 - dd / 0.34, 0.85) });
      }
      if (nodes.length) hand.clothGrab = { flag: clothEnt, nodes, useTip };
    }
  }
};

});

const buildSphereCloud = (root, voxelSize = 0.085) => {
  root.updateMatrixWorld(true);
  const buckets = new Map(), v = new THREE.Vector3();
  const addPt = pt => {
    const k = Math.round(pt.x / voxelSize) + "," + Math.round(pt.y / voxelSize) + "," + Math.round(pt.z / voxelSize);
    let b = buckets.get(k); if (!b) { b = { sx: 0, sy: 0, sz: 0, n: 0 }; buckets.set(k, b); }
    b.sx += pt.x; b.sy += pt.y; b.sz += pt.z; b.n++;
  };
  root.traverse(o => {
    if (!o.isMesh || !o.geometry) return;
    const pos = o.geometry.attributes.position; if (!pos) return;
    const step = Math.max(1, Math.floor(pos.count / 1200));
    for (let i = 0; i < pos.count; i += step) {
      v.fromBufferAttribute(pos, i);
      if (o.isSkinnedMesh && o.skeleton && typeof o.boneTransform === "function") o.boneTransform(i, v);
      v.applyMatrix4(o.matrixWorld); addPt(v);
    }
  });
  return [...buckets.values()].map(b => ({ x: b.sx / b.n, y: b.sy / b.n, z: b.sz / b.n, r: voxelSize * 0.92 }));
};

const extractRotation = (A, q, iters = 5) => {

  for (let i = 0; i < iters; i++) {

const closestSegSeg = (p1, q1, p2, q2, o1, o2) => {
  const d1 = q1.clone().sub(p1), d2 = q2.clone().sub(p2), r = p1.clone().sub(p2), a = d1.dot(d1), e = d2.dot(d2), f = d2.dot(r);
  let s = 0, t = 0;
  if (a > 1e-8 && e > 1e-8) {
    const c = d1.dot(r), b = d1.dot(d2), den = a * e - b * b;
    s = den !== 0 ? clamp((b * f - c * e) / den, 0, 1) : 0; t = (b * s + f) / e;
    if (t < 0) { t = 0; s = clamp(-c / a, 0, 1); } else if (t > 1) { t = 1; s = clamp((b - c) / a, 0, 1); }
  }
  o1.copy(p1).addScaledVector(d1, s); o2.copy(p2).addScaledVector(d2, t); return { s, t };
};

};

// Fusion design target: one coherent creature/object, never two complete models
// pasted together. The current generic mesh pipeline can blend PBR appearance
// and scale, but it cannot infer anatomy or retarget arbitrary skeletons.
// Keep this constraint explicit for future semantic-part / rig-aware fusion.
// Helper to build a believable, integrated 3D Hybrid from two source entities
const buildHybridEntity = (entA, entB, spawnWorldPos) => {
  const hybridRoot = new THREE.Group();
  hybridRoot.position.copy(spawnWorldPos);

  const collectMeshes = (ent) => {
    const list = [];
    ent.root.updateMatrixWorld(true);
    ent.root.traverse(o => {
      if (!o.isMesh || !o.geometry) return;
      const g = o.geometry.clone();
      g.applyMatrix4(o.matrixWorld);
      list.push({ geo: g, mat: o.material, name: String(o.name || "").toLowerCase() });
    });
    return list;
  };
  const meshesA = collectMeshes(entA), meshesB = collectMeshes(entB);
  // Only transfer anatomically meaningful appendages from source B. Never paste
  // its complete silhouette over source A (the old behavior made two models overlap).
  const featurePattern = /wing|feather|tail|beak|horn|fin|crest|ear|tusk|antler|claw|spike|mane|whisker|shell|scale|tentacle|petal|leaf/i;
  const featureMeshesB = meshesB.filter(item => featurePattern.test(item.name));
  const bA = APP.computeBox(entA.root), bB = APP.computeBox(entB.root);
  const cA = bA.getCenter(new THREE.Vector3()), szA = bA.getSize(new THREE.Vector3());
  const cB = bB.getCenter(new THREE.Vector3()), szB = bB.getSize(new THREE.Vector3());

  // Extract averaged & fused PBR material characteristics (color, metalness, roughness, clearcoat, texture)
  const getMatProps = (m) => {
    const mat = Array.isArray(m) ? m[0] : m;
    return {
      color: mat && mat.color ? mat.color.clone() : new THREE.Color(0x99aabb),
      metalness: mat && mat.metalness !== undefined ? mat.metalness : 0.25,
      roughness: mat && mat.roughness !== undefined ? mat.roughness : 0.45,
      map: mat && mat.map ? mat.map : null
    };
  };
  const pA = getMatProps(meshesA[0]?.mat), pB = getMatProps(meshesB[0]?.mat);
  const fusedColorA = pA.color.clone().lerp(pB.color, 0.38);
  const fusedColorB = pB.color.clone().lerp(pA.color, 0.38);
  const fusedMetal = (pA.metalness + pB.metalness) * 0.5;
  const fusedRough = Math.max(0.08, (pA.roughness + pB.roughness) * 0.5);

  const normFactorA = 0.52 / Math.max(0.05, szA.length());
  const normFactorB = 0.48 / Math.max(0.05, szB.length());

  // Core chassis from Object A with blended PBR material + bio-fusion sheen
  const bodyGrp = new THREE.Group();
  for (let i = 0; i < meshesA.length; i++) {
    const item = meshesA[i];
    item.geo.translate(-cA.x, -cA.y, -cA.z);
    item.geo.scale(normFactorA, normFactorA, normFactorA);
    const origM = Array.isArray(item.mat) ? item.mat[0] : item.mat;
    const m = new THREE.MeshPhysicalMaterial({
      color: origM && origM.color ? origM.color.clone().lerp(fusedColorB, 0.35) : fusedColorA,
      map: (origM && origM.map) || pB.map || null,
      metalness: fusedMetal, roughness: fusedRough, clearcoat: 0.55,
      emissive: new THREE.Color(0x004433), emissiveIntensity: 0.22
    });
    const mesh = new THREE.Mesh(item.geo, m);
    mesh.castShadow = mesh.receiveShadow = true;
    bodyGrp.add(mesh);
  }
  hybridRoot.add(bodyGrp);

  // Integrated features/appendages from Object B socketed symmetrically & dorsally onto Core A
  const featGrp = new THREE.Group();
  for (let i = 0; i < featureMeshesB.length; i++) {
    const item = featureMeshesB[i];
    item.geo.translate(-cB.x, -cB.y, -cB.z);
    item.geo.scale(normFactorB, normFactorB, normFactorB);
    const origM = Array.isArray(item.mat) ? item.mat[0] : item.mat;
    const m = new THREE.MeshPhysicalMaterial({
      color: origM && origM.color ? origM.color.clone().lerp(fusedColorA, 0.35) : fusedColorB,
      map: (origM && origM.map) || pA.map || null,
      metalness: fusedMetal, roughness: fusedRough, clearcoat: 0.6,
      emissive: new THREE.Color(0x003344), emissiveIntensity: 0.25
    });
    const mesh = new THREE.Mesh(item.geo, m);
    mesh.castShadow = mesh.receiveShadow = true;
    featGrp.add(mesh);
  }
  // Attach named appendages to the host silhouette, rather than retaining B's body.
  // Model-specific names are the strongest available semantic signal in generic GLBs.
  featGrp.position.set(0, szA.y * normFactorA * 0.28, -szA.z * normFactorA * 0.08);
  featGrp.rotation.x = 0.08;
  hybridRoot.add(featGrp);

  // Fallback for common bird + quadruped fusion when the bird GLB is a single
  // unnamed mesh: synthesize a paired feathered wing structure instead of
  // overlaying the entire bird. The source bird's material color is retained.
  const sourceBName = String(entB.name || "").toLowerCase();
  if (!featureMeshesB.length && /bird|eagle|hawk|raven|crow|parrot|owl|falcon|فراش|طائر|نسر|صقر|بومة/.test(sourceBName)) {
    const birdMat = getMatProps(meshesB[0]?.mat);
    const featherMat = new THREE.MeshPhysicalMaterial({
      color: birdMat.color, map: birdMat.map, metalness: Math.min(0.18, birdMat.metalness),
      roughness: Math.max(0.38, birdMat.roughness), sheen: 0.55, clearcoat: 0.12
    });
    for (const side of [-1, 1]) {
      const wing = new THREE.Group();
      wing.position.set(side * 0.16, szA.y * normFactorA * 0.18, -szA.z * normFactorA * 0.02);
      wing.rotation.z = side * -0.18;
      for (let f = 0; f < 7; f++) {
        const feather = new THREE.Mesh(new THREE.SphereGeometry(0.075, 14, 10), featherMat);
        feather.scale.set(0.42, 1.65 - f * 0.09, 0.20);
        feather.position.set(side * (0.055 + f * 0.047), 0.025 + f * 0.012, -f * 0.018);
        feather.rotation.z = side * (0.35 + f * 0.105);
        feather.castShadow = feather.receiveShadow = true;
        wing.add(feather);
      }
      hybridRoot.add(wing);
    }
    // A small integrated feather fan at the rear completes the avian silhouette.
    const tail = new THREE.Group();
    tail.position.set(0, szA.y * normFactorA * 0.02, -szA.z * normFactorA * 0.43);
    for (let f = -2; f <= 2; f++) {
      const plume = new THREE.Mesh(new THREE.SphereGeometry(0.055, 12, 8), featherMat);
      plume.scale.set(0.38, 1.7, 0.2);
      plume.position.set(f * 0.045, Math.abs(f) * 0.012, -Math.abs(f) * 0.018);
      plume.rotation.z = f * 0.18;
      plume.castShadow = plume.receiveShadow = true;
      tail.add(plume);
    }
    hybridRoot.add(tail);
  }

  // Add glowing energy fusion seam core uniting both geometries
  const seamMat = new THREE.MeshPhysicalMaterial({ color: 0x00ffaa, emissive: 0x00cc77, emissiveIntensity: 0.9, roughness: 0.15, transparent: true, opacity: 0.72 });
  const seam = new THREE.Mesh(new THREE.TorusGeometry(0.11, 0.012, 10, 32).rotateX(Math.PI / 2), seamMat);
  seam.position.y = szA.y * normFactorA * 0.12;
  hybridRoot.add(seam);

  APP.scene.add(hybridRoot);
  const hb = APP.computeBox(hybridRoot);
  const mxU = APP.applyDissolveShader(hybridRoot, hb.min.y, hb.max.y, false);
  return APP.registerEntity({
    name: entA.name + " + " + entB.name, type: "hybrid", root: hybridRoot, box: hb,
    matrixUniforms: mxU, buildDur: 1.1, grabbable: true, ambientFx: "glow"
  });
};

const triggerFusion = () => {
  if (fusing) return;
  if (!leftObj || !rightObj) { APP.say("⚠️ ضع غرضاً في الغرفة اليسرى وآخر في اليمنى أولاً"); return; }
  fusing = true; fuseT = 0;
  APP.sfx.play("magic");
  APP.say("🧬 جاري المسح والدمج الجزيئي...");

  // Sci-fi reactor sweep audio
  const c2 = APP.sfx.ctx;
  if (c2) {
    const t = c2.currentTime, o1 = c2.createOscillator(), o2 = c2.createOscillator(), g = c2.createGain();
    o1.type = "sawtooth"; o1.frequency.setValueAtTime(90, t); o1.frequency.exponentialRampToValueAtTime(680, t + 3.2);
    o2.type = "sine"; o2.frequency.setValueAtTime(180, t); o2.frequency.exponentialRampToValueAtTime(1120, t + 3.2);
    g.gain.setValueAtTime(0.001, t); g.gain.linearRampToValueAtTime(0.09, t + 0.4); g.gain.exponentialRampToValueAtTime(0.001, t + 3.4);
    o1.connect(g); o2.connect(g); g.connect(c2.destination);
    o1.start(t); o2.start(t); o1.stop(t + 3.45); o2.stop(t + 3.45);
  }
};

      }

    // Detect & magnetically snap objects placed inside Left & Right Chambers
    if (!fusing) {
      for (const e of APP.entities) {
        if (e === machEnt || e.dissolveStart || !e.grabbable || e.heldBy || e.type === "hybrid") continue;
        const ec = APP.computeBox(e.root).getCenter(new THREE.Vector3());
        if (!leftObj && e !== rightObj && ec.distanceTo(lWorld) < 0.45) {
          leftObj = e; e.lockedInChamber = true;
          const b0 = APP.computeBox(e.root), sz = b0.getSize(new THREE.Vector3());
          if (Math.max(sz.x, sz.y, sz.z) > 0.42) e.root.scale.multiplyScalar(0.38 / Math.max(sz.x, sz.y, sz.z));
          e.root.position.set(lWorld.x, lWorld.y - 0.16, lWorld.z);
          APP.sfx.play("magic"); APP.say("📥 تم تثبيت الجسم الأول في الحجرة اليسرى");
        } else if (!rightObj && e !== leftObj && ec.distanceTo(rWorld) < 0.45) {
          rightObj = e; e.lockedInChamber = true;
          const b0 = APP.computeBox(e.root), sz = b0.getSize(new THREE.Vector3());
          if (Math.max(sz.x, sz.y, sz.z) > 0.42) e.root.scale.multiplyScalar(0.38 / Math.max(sz.x, sz.y, sz.z));
          e.root.position.set(rWorld.x, rWorld.y - 0.16, rWorld.z);
          APP.sfx.play("magic"); APP.say("📥 تم تثبيت الجسم الثاني في الحجرة اليمنى — اضغط الزر الأخضر!");
        }
      }
      leftCham.glowRing.material.emissive.setHex(leftObj ? 0x00ff88 : 0x00aacc);
      rightCham.glowRing.material.emissive.setHex(rightObj ? 0x00ff88 : 0x00aacc);
    } else {
      fuseT += dt;
      // 1. Scan & Energize Phase (0 -> 3.2s)
      if (fuseT < 3.2) {
        const k = fuseT / 3.2;
        leftCham.scanRing.material.opacity = rightCham.scanRing.material.opacity = 0.95;
        centerCham.scanRing.material.opacity = 0.95;
        leftCham.scanRing.position.y = rightCham.scanRing.position.y = 0.12 + (0.5 + 0.5 * Math.sin(fuseT * 8)) * 0.62;
        centerCham.scanRing.position.y = 0.14 + (0.5 + 0.5 * Math.cos(fuseT * 10)) * 0.72;
        coreLight.intensity = 1.5 + k * 5.0 + Math.random() * 1.5;

        if (leftObj) { leftObj.root.rotation.y += dt * (2 + k * 8); if (leftObj.matrixUniforms) leftObj.matrixUniforms.uProg.value = 1 - k * 0.85; }
        if (rightObj) { rightObj.root.rotation.y -= dt * (2 + k * 8); if (rightObj.matrixUniforms) rightObj.matrixUniforms.uProg.value = 1 - k * 0.85; }

        if (Math.random() < 0.9) {
          APP.fx.spawnPoint("glow", lWorld.x + (cWorld.x - lWorld.x) * Math.random(), lWorld.y + 0.25, lWorld.z, 0.8, 0x00f0ff);
          APP.fx.spawnPoint("glow", rWorld.x + (cWorld.x - rWorld.x) * Math.random(), rWorld.y + 0.25, rWorld.z, 0.8, 0x00ff88);
          APP.fx.spawnPoint("sparks", cWorld.x, cWorld.y, cWorld.z, 0.9, 0x44ffaa);
        }
      } else {
        // 2. Complete Fusion & Eject Hybrid Creation!
        leftCham.scanRing.material.opacity = rightCham.scanRing.material.opacity = centerCham.scanRing.material.opacity = 0;
        coreLight.intensity = 0.8;

        const srcA = leftObj, srcB = rightObj;
        leftObj = null; rightObj = null; fusing = false;

        if (srcA && srcB) {
          const hybridEnt = buildHybridEntity(srcA, srcB, cWorld);
          APP.removeEntity(srcA); APP.removeEntity(srcB);
          APP.sfx.play("magic");
          APP.fx.groundPulse(cWorld.x, cWorld.z, 2.5, 0x00ffaa);
          for (let i = 0; i < 45; i++) APP.fx.spawnPoint("sparks", cWorld.x, cWorld.y, cWorld.z, 1.2, 0x00ff99);

          // Eject fused creation out of central chamber toward user
          const ejectDir = new THREE.Vector3(0, 0, 1).applyQuaternion(root.quaternion).normalize();
          const ejectTarget = cWorld.clone().addScaledVector(ejectDir, 0.72).setY(1.22);
          let ejT = 0;
          APP.onTick("eject_" + hybridEnt.id, 1.1, (edt) => {
            ejT += edt;
            if (!hybridEnt.heldBy) {
              hybridEnt.root.position.lerp(ejectTarget, Math.min(1, edt * 4.5));
              hybridEnt.root.rotation.y += edt * 1.4;
            }
          });
          APP.say("✨ تم الدمج! تم إخراج الكائن الهجين: " + hybridEnt.name);
        }
      }
    }
  },
  dispose: () => offPtr()
});
