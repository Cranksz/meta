function playEmotionalChirp(mood = 'curious', intensity = 1) {
  const ctx = APP.sfx?.ctx;
  const bus = APP.sfx?.bus;
  if (!ctx || !bus || bot.muted) return;
  const t = ctx.currentTime;
  const presets = {
    curious:    [520, 740, 960],
    happy:      [660, 880, 1174, 1320],
    excited:    [587, 880, 1174, 1568],
    thoughtful: [440, 392, 494],
    alert:      [780, 520, 780],
    warm:       [494, 622, 740],
    proud:      [523, 659, 784, 1046]
  };
  const notes = presets[mood] || presets.curious;
  notes.forEach((f, idx) => {
    const osc = ctx.createOscillator();
    const g = ctx.createGain();
    osc.type = idx % 2 === 0 ? 'sine' : 'triangle';
    const st = t + idx * 0.052;
    osc.frequency.setValueAtTime(f * (0.97 + Math.random() * 0.06), st);
    osc.frequency.exponentialRampToValueAtTime(f * 1.04, st + 0.09);
    g.gain.setValueAtTime(0.001, st);
    g.gain.linearRampToValueAtTime(0.055 * intensity, st + 0.018);
    g.gain.exponentialRampToValueAtTime(0.001, st + 0.13);
    osc.connect(g).connect(bus);
    osc.start(st);
    osc.stop(st + 0.14);
  });
}

let spatialPanner = null, spatialAudioCtx = null;
bot.setOutputRouter((audioCtx, analyserNode) => {
  spatialAudioCtx = audioCtx;
  const lowWarm = audioCtx.createBiquadFilter();
  lowWarm.type = 'lowshelf';
  lowWarm.frequency.value = 190;
  lowWarm.gain.value = 3.2;

  const presence = audioCtx.createBiquadFilter();
  presence.type = 'peaking';
  presence.frequency.value = 2850;
  presence.Q.value = 1.1;
  presence.gain.value = 2.6;

  const comp = audioCtx.createDynamicsCompressor();
  comp.threshold.value = -16;
  comp.knee.value = 18;
  comp.ratio.value = 3.5;

  let tail = comp;
  try {
    const panner = audioCtx.createPanner();
    panner.panningModel = 'HRTF';
    panner.distanceModel = 'inverse';
    panner.refDistance = 1.0;
    panner.maxDistance = 18;
    panner.rolloffFactor = 0.85;
    spatialPanner = panner;
    comp.connect(panner);
    tail = panner;
  } catch {
    spatialPanner = null;
  }

  analyserNode.connect(lowWarm);
  lowWarm.connect(presence);
  presence.connect(comp);
  return tail;
});

function updateSpatialAudioPose(botPos, headPos, headFwd) {
  if (!spatialAudioCtx || !spatialPanner) return;
  try {
    const listener = spatialAudioCtx.listener;
    if (listener.positionX) {
      listener.positionX.value = headPos.x;
      listener.positionY.value = headPos.y;
      listener.positionZ.value = headPos.z;
      listener.forwardX.value = headFwd.x;
      listener.forwardY.value = 0;
      listener.forwardZ.value = headFwd.z;
      listener.upX.value = 0; listener.upY.value = 1; listener.upZ.value = 0;
    } else if (listener.setPosition) {
      listener.setPosition(headPos.x, headPos.y, headPos.z);
      listener.setOrientation(headFwd.x, 0, headFwd.z, 0, 1, 0);
    }
    if (spatialPanner.positionX) {
      spatialPanner.positionX.value = botPos.x;
      spatialPanner.positionY.value = botPos.y;
      spatialPanner.positionZ.value = botPos.z;
    } else if (spatialPanner.setPosition) {
      spatialPanner.setPosition(botPos.x, botPos.y, botPos.z);
    }
  } catch {}
}

let activeTtsSource = null;
async function speakAliveVoice(text, mood = 'warm') {
  if (bot.muted || !text) return true;
  const openAiKey = bot.getOpenAiKey ? bot.getOpenAiKey() : '';
  if (openAiKey && openAiKey.startsWith('sk-')) {
    try {
      await bot.ensureAudioOutput();
      const voiceMap = { Kore: 'nova', Aoede: 'shimmer', Zephyr: 'coral', Puck: 'echo', Charon: 'onyx', Fenrir: 'ash' };
      const chosenVoice = voiceMap[bot.voiceName] || 'nova';
      const toneMap = {
        excited: 'Speak with bright, enthusiastic, lively human warmth and natural rhythm.',
        happy: 'Speak warmly with a genuine smile and upbeat friendly expression.',
        curious: 'Speak with an inquisitive, attentive, friendly tone and expressive phrasing.',
        thoughtful: 'Speak calmly, warmly, and thoughtfully like an intelligent companion.',
        alert: 'Speak clearly, briskly, and attentively with urgent helpfulness.',
        warm: 'Speak with a rich, natural, empathetic, alive voice.'
      };
      const res = await fetch('https://api.openai.com/v1/audio/speech', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + openAiKey },
        body: JSON.stringify({
          model: 'gpt-4o-mini-tts',
          voice: chosenVoice,
          input: text,
          instructions: toneMap[mood] || toneMap.warm,
          response_format: 'mp3'
        })
      });
      if (res.ok) {
        const arr = await res.arrayBuffer();
        const audioBuf = await bot.outCtx.decodeAudioData(arr);
        if (activeTtsSource) { try { activeTtsSource.stop(); } catch {} }
        const src = bot.outCtx.createBufferSource();
        src.buffer = audioBuf;
        src.connect(bot.outGain);
        src.start();
        activeTtsSource = src;
        bot.aiSpeakingUntil = NOW() + audioBuf.duration + 0.15;
        bot.setState('ai_speaking');
        src.onended = () => {
          if (activeTtsSource === src) activeTtsSource = null;
          if (bot.connected && NOW() > bot.stateFlashUntil) bot.setState('watching');
        };
        return true;
      }
    } catch (e) {
      console.warn('OpenAI TTS fallback to WebSpeech:', e);
    }
  }

  // Expressive browser speech fallback with gentle pulse modulation
  if ('speechSynthesis' in window) {
    try {
      window.speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(text);
      u.lang = /[\u0600-\u06FF]/.test(text) ? 'ar-SA' : 'en-US';
      const pitchMap = { excited: 1.16, happy: 1.12, curious: 1.08, warm: 1.03, thoughtful: 0.96, alert: 1.18, proud: 1.10 };
      const rateMap  = { excited: 1.09, happy: 1.05, curious: 1.02, warm: 0.98, thoughtful: 0.94, alert: 1.12, proud: 1.02 };
      u.pitch = pitchMap[mood] || 1.05;
      u.rate  = rateMap[mood]  || 1.02;
      playEmotionalChirp(mood, 0.55);
      u.onstart = () => { bot.aiSpeakingUntil = NOW() + Math.max(2.0, text.length * 0.065); bot.setState('ai_speaking'); };
      u.onend = () => { bot.aiSpeakingUntil = 0; if (bot.connected && NOW() > bot.stateFlashUntil) bot.setState('watching'); };
      window.speechSynthesis.speak(u);
      return true;
    } catch {}
  }
  return false;
}

function solveAndRenderTentacle(basePos, targetTipPos, t, dt, clawOpenAmt) {
  const totalDist = basePos.distanceTo(targetTipPos);
  const segLen = Math.max(SEG_LEN, Math.min(0.28, (totalDist + 0.08) / SEG_COUNT));

  tentaclePoints[0].copy(basePos);
  if (tentaclePoints[SEG_COUNT].lengthSq() < 1e-5) {
    for (let i = 1; i <= SEG_COUNT; i++) tentaclePoints[i].copy(basePos).add(new V3(0, -i * segLen * 0.6, 0));
  }

  // Organic wave perturbation along middle segments
  const midDir = targetTipPos.clone().sub(basePos);
  const swayAmp = tentacleState.mode === 'idle' ? 0.035 : 0.014;

  // FABRIK 4 iterations for buttery smooth articulation
  for (let iter = 0; iter < 4; iter++) {
    tentaclePoints[SEG_COUNT].copy(targetTipPos);
    for (let i = SEG_COUNT - 1; i >= 1; i--) {
      const dir = tentaclePoints[i].clone().sub(tentaclePoints[i + 1]);
      if (dir.lengthSq() < 1e-6) dir.set(0, -1, 0);
      dir.normalize();
      tentaclePoints[i].copy(tentaclePoints[i + 1]).addScaledVector(dir, segLen);
    }
    tentaclePoints[0].copy(basePos);
    for (let i = 1; i <= SEG_COUNT; i++) {
      const dir = tentaclePoints[i].clone().sub(tentaclePoints[i - 1]);
      if (dir.lengthSq() < 1e-6) dir.set(0, -1, 0);
      dir.normalize();
      // Gentle biological curve
      const wave = Math.sin(t * 2.8 - i * 0.55) * swayAmp * (i / SEG_COUNT);
      dir.x += wave;
      dir.y -= 0.03 * (1 - i / SEG_COUNT);
      dir.normalize();
      tentaclePoints[i].copy(tentaclePoints[i - 1]).addScaledVector(dir, segLen);
    }
  }

  const up = new V3(0, 1, 0);
  for (let i = 0; i <= SEG_COUNT; i++) {
    tentacleJoints[i].position.copy(tentaclePoints[i]);
    if (i < SEG_COUNT) {
      const a = tentaclePoints[i], b = tentaclePoints[i + 1];
      const link = tentacleLinks[i];
      link.position.copy(a).lerp(b, 0.5);
      const diff = b.clone().sub(a);
      const len = Math.max(0.001, diff.length());
      link.scale.set(1, len, 1);
      link.quaternion.setFromUnitVectors(up, diff.divideScalar(len));
    }
  }

  const tip = tentaclePoints[SEG_COUNT];
  const prev = tentaclePoints[SEG_COUNT - 1];
  clawGroup.position.copy(tip);
  const lookEnd = tip.clone().add(tip.clone().sub(prev));
  clawGroup.lookAt(lookEnd);

  for (const finger of clawProngs) {
    finger.rotation.x = THREE.MathUtils.lerp(finger.rotation.x, 0.15 + clawOpenAmt * 0.55, 1 - Math.exp(-12 * dt));
    finger.position.y = 0.017 + clawOpenAmt * 0.016;
  }
}

function computeSemanticDestination(spec = {}) {
  if (Number.isFinite(Number(spec.x)) && Number.isFinite(Number(spec.z))) {
    return new V3(Number(spec.x), Number.isFinite(Number(spec.y)) ? Number(spec.y) : 0, Number(spec.z));
  }

  const dest = normalizeText(String(spec.destination || spec.relation || ''));
  const dist = clamp(Number(spec.distance || 1.9), 0.7, 5.0);
  const right = new V3().crossVectors(APP.headFwd, new V3(0, 1, 0)).normalize();
  const basePos = APP.headPos.clone();

  // Reference another object if specified
  if (spec.reference_object) {
    const refs = APP.findEntities(spec.reference_object);
    if (refs.length) {
      const refEnt = refs[0];
      const rb = APP.computeBox(refEnt.root);
      const rc = rb.getCenter(new V3());
      const out = rc.clone();
      if (/left|يسار|شمال/.test(dest)) out.addScaledVector(right, -0.95);
      else if (/right|يمين/.test(dest)) out.addScaledVector(right, 0.95);
      else if (/front|امام|قدام/.test(dest)) out.addScaledVector(APP.headFwd, -0.85);
      else if (/behind|خلف|وراء/.test(dest)) out.addScaledVector(APP.headFwd, 0.85);
      else if (/top|above|فوق|اعلى/.test(dest)) out.y = rb.max.y + 0.05;
      else out.addScaledVector(right, 0.85);
      if (!/top|above|فوق|اعلى/.test(dest)) out.y = Number.isFinite(Number(spec.height)) ? Number(spec.height) : refEnt.root.position.y;
      return out;
    }
  }

  // Nearest room wall
  if (/wall|حائط|جدار/.test(dest)) {
    const w = APP.findNearestWall(APP.headPos, APP.headFwd);
    if (w) return w.point.clone().addScaledVector(w.normal, 0.45);
  }

  // Relative to current entity
  const targets = APP.findEntities(spec.target || 'last');
  if (targets.length && /closer|اقرب|قرب|farther|ابعد|بعد|higher|ارفع|اعلى|lower|اخفض|انزل/.test(dest)) {
    const cur = targets[0].root.position.clone();
    if (/closer|اقرب|قرب/.test(dest)) return cur.lerp(new V3(basePos.x, cur.y, basePos.z), 0.45);
    if (/farther|ابعد|بعد/.test(dest)) return cur.addScaledVector(APP.headFwd, 1.1);
    if (/higher|ارفع|اعلى/.test(dest)) { cur.y += Number(spec.height || 0.85); return cur; }
    if (/lower|اخفض|انزل/.test(dest)) { cur.y = Math.max(0, cur.y - Number(spec.height || 0.7)); return cur; }
  }

  // Relative to user
  const out = new V3(basePos.x, Number.isFinite(Number(spec.height)) ? Number(spec.height) : 0, basePos.z);
  if (/left|يسار|شمال/.test(dest)) out.addScaledVector(APP.headFwd, dist).addScaledVector(right, -1.1);
  else if (/right|يمين/.test(dest)) out.addScaledVector(APP.headFwd, dist).addScaledVector(right, 1.1);
  else if (/behind|خلف|وراء/.test(dest)) out.addScaledVector(APP.headFwd, -dist);
  else if (/near|hand|يد|بجانبي|جنبي|قريب/.test(dest)) out.addScaledVector(APP.headFwd, 0.75).addScaledVector(right, 0.25);
  else out.addScaledVector(APP.headFwd, dist);

  if (/fly|air|هواء|طائر/.test(dest) && !Number.isFinite(Number(spec.height))) out.y = 1.25;
  return out;
}

function startTentacleMoveAndPlace(args = {}) {
  const targets = APP.findEntities(args.target || 'last');
  if (!targets.length) return { ok: false, error: 'لا يوجد مجسم في المشهد لتحريكه الآن' };

  const ent = targets[0];
  const destPos = computeSemanticDestination(args);
  const startPos = ent.root.position.clone();
  const box = APP.computeBox(ent.root);
  const grabOffset = box.isEmpty() ? new V3(0, 0.4, 0) : box.getCenter(new V3()).sub(startPos);

  tentacleState.mode = 'grab_reach';
  tentacleState.grabTask = {
    ent, startPos, destPos, grabOffset,
    phaseTime: 0, reachDur: 0.65, carryDur: 1.15, retractDur: 0.55,
    yawDeg: Number.isFinite(Number(args.yaw_deg)) ? Number(args.yaw_deg) : null
  };
  bot.inspectPoint.copy(startPos).add(grabOffset);
  bot.inspectUntil = NOW() + 3.2;
  bot.setMood('excited');
  playEmotionalChirp('excited', 0.9);

  return {
    ok: true,
    movingObject: ent.name,
    from: { x: +startPos.x.toFixed(2), y: +startPos.y.toFixed(2), z: +startPos.z.toFixed(2) },
    to: { x: +destPos.x.toFixed(2), y: +destPos.y.toFixed(2), z: +destPos.z.toFixed(2) }
  };
}

let lastTouchAt = 0, lastHoverChirpAt = 0;

function handleTactileImpact(hitPt, impulseDir, handSide) {
  const now = NOW();
  if (now - lastTouchAt < 0.28) return;
  lastTouchAt = now;
  bot.recoilVel.addScaledVector(impulseDir, 1.35);
  bot.inspectPoint.copy(hitPt);
  bot.inspectUntil = now + 2.4;
  bot.setMood('happy', 'لمسة لطيفة! ✨');
  playEmotionalChirp('happy', 0.9);
  for (let i = 0; i < 12; i++) {
    APP.fx.spawnPoint('sparks', hitPt.x, hitPt.y, hitPt.z, 0.75, 0x44ffee);
  }
  bot.recordTelemetry('companion_touched', { hand: handSide });
}

const touchRay = new THREE.Raycaster(), tmpQ = new Q4(), tmpD = new V3(), tmpW = new V3();

bot.setController({
  mount(rootGroup) {
    rootGroup.add(botRoot);
    tentacleGroup.visible = true;
    playEmotionalChirp('curious', 0.7);
  },
  unmount(rootGroup) {
    rootGroup.remove(botRoot);
    tentacleGroup.visible = false;
    pointerBeam.visible = false;
    APP.scene.remove(tentacleGroup);
    APP.scene.remove(pointerBeam);
    if (activeTtsSource) { try { activeTtsSource.stop(); } catch {} }
    for (const d of disposables) { try { d.dispose(); } catch {} }
  },
  onStateChange(state, mood, subText) {
    const pal = MOOD_PALETTE[mood] || MOOD_PALETTE.curious;
    const stateCol =
      state === 'error' ? 0xff4455 :
      state === 'coding' ? 0xaa66ff :

      state === 'synced' ? 0x44ffcc : pal.core;

    conduitMat.emissive.setHex(pal.ring);

    irisGlowMat.emissive.setHex(pal.iris);
    botLight.color.setHex(stateCol);
    drawBotTag(subText || '🤖 الكرة الذكية', `${bot.modelName} • ${mood}`, pal.hex);
  },
  onSpeak(text, mood) {
    return speakAliveVoice(text, mood);
  },
  onAction(name, payload) {
    if (name === 'hologram') return showHologram(payload);
    if (name === 'move_object') return startTentacleMoveAndPlace(payload);
    return null;
  },
  tryInteract(hand) {
    if (!botRoot.visible) return false;
    const interactiveTargets = [hitProxy];
    if (holoState.visible) holoState.tabs.forEach(t => interactiveTargets.push(t.mesh));

    let hit = null;
    if (hand.isDesk) {
      hit = APP.pointer.ray.intersectObjects(interactiveTargets, false)[0] || null;
    } else if (hand.handTrackingActive) {
      if (holoState.visible) {
        for (const t of holoState.tabs) {
          const wp = t.mesh.getWorldPosition(new V3());
          if (hand.tipWp.distanceTo(wp) < 0.11) { hit = { object: t.mesh, point: wp }; break; }
        }
      }
      if (!hit && hand.tipWp.distanceTo(bot.pos) < 0.22) {
        hit = { object: hitProxy, point: hand.tipWp.clone() };
      }
    } else if (hand.c) {
      hand.c.getWorldPosition(tmpW); hand.c.getWorldQuaternion(tmpQ);
      tmpD.set(0, 0, -1).applyQuaternion(tmpQ);
      touchRay.set(tmpW, tmpD);
      hit = touchRay.intersectObjects(interactiveTargets, false)[0] || null;
    }

    if (!hit) return false;
    if (hit.object.userData && hit.object.userData.holoTabId) {
      const tid = hit.object.userData.holoTabId;
      APP.sfx.play('click');
      if (tid === 'close') showHologram({ visible: false });
      else showHologram({ visible: true, active_tab: tid });
      return true;
    }
    const pushDir = bot.pos.clone().sub(hit.point).normalize();
    handleTactileImpact(hit.point, pushDir, hand.side);
    return true;
  },
  onStep(dt, t, ctx) {
    const now = NOW();
    const pal = MOOD_PALETTE[ctx.mood] || MOOD_PALETTE.curious;
    const energy = ctx.audioEnergy;

    updateSpatialAudioPose(ctx.pos, ctx.headPos, ctx.headFwd);

    // 1) Continuous Virtual Hand Touch & Proximity Detection
    let nearestHandDist = Infinity, nearestHandPt = null;
    for (const h of ctx.hands) {
      if (h.isDesk && !APP.pointer.down) continue;
      const pts = h.handTrackingActive ? [h.tipWp, ...h.fingerTips] : [h.tipWp, h.wp];
      for (const p of pts) {
        const d = p.distanceTo(ctx.pos);
        if (d < nearestHandDist) { nearestHandDist = d; nearestHandPt = p; }
      }
      // Physical contact with sphere!
      if (nearestHandDist < 0.16 && nearestHandPt) {
        const pushDir = ctx.pos.clone().sub(nearestHandPt).normalize();
        handleTactileImpact(nearestHandPt, pushDir, h.side);
      }
      // Virtual finger touching holographic tabs directly
      if (holoState.visible && h.handTrackingActive) {
        for (const tab of holoState.tabs) {
          const wp = tab.mesh.getWorldPosition(new V3());
          if (h.tipWp.distanceTo(wp) < 0.085 && now - lastTouchAt > 0.45) {
            lastTouchAt = now;
            APP.sfx.play('click');
            if (tab.id === 'close') showHologram({ visible: false });
            else showHologram({ visible: true, active_tab: tab.id });
          }
        }
      }
    }

    const handNear = nearestHandDist < 0.36;
    if (handNear && now - lastHoverChirpAt > 3.0) {
      lastHoverChirpAt = now;
      playEmotionalChirp('curious', 0.45);
    }

    // 2) Armor Breathing, Cheek Flare, Floating Wing Fins & Eye LookAt
    const flare = pal.flare + (handNear ? 0.18 : 0) + clamp(energy * 0.14, 0, 0.22);
    upperShell.position.y = THREE.MathUtils.lerp(upperShell.position.y, flare * 0.045, 1 - Math.exp(-8 * dt));
    lowerShell.position.y = THREE.MathUtils.lerp(lowerShell.position.y, -flare * 0.045, 1 - Math.exp(-8 * dt));
    leftArmor.position.x = -0.114 - flare * 0.045;
    rightArmor.position.x = 0.114 + flare * 0.045;
    leftArmor.rotation.z = flare * 0.38;
    rightArmor.rotation.z = -flare * 0.38;

    bodyPivot.rotation.z = clamp(-ctx.vel.dot(ctx.right) * 0.16, -0.30, 0.30);
    bodyPivot.rotation.x = clamp(ctx.vel.dot(ctx.headFwd) * 0.14, -0.25, 0.25);

    for (const f of aeroFins) {
      const r = 0.145 + flare * 0.055;
      const ang = f.angle + Math.sin(t * 1.8) * 0.06;
      f.mesh.position.set(Math.cos(ang) * r, Math.sin(t * 3.1 + f.angle) * 0.012, Math.sin(ang) * r);
      f.mesh.rotation.y = -ang + Math.PI / 2;
      f.mesh.rotation.z = flare * 0.35;
    }

    const lookTarget = (handNear && nearestHandPt)
      ? nearestHandPt
      : ((now < ctx.inspectUntil && ctx.state !== 'ai_speaking' && ctx.state !== 'user_speaking') ? ctx.inspectPoint : ctx.headPos);

    const lookObj = new THREE.Object3D();
    lookObj.position.copy(ctx.pos);
    lookObj.lookAt(lookTarget);
    eyePivot.quaternion.slerp(lookObj.quaternion, 1 - Math.exp(-8.5 * dt));

    // 6-Blade Shutter & Pupil Dilation
    const pupilScale = 0.85 + clamp(energy * 0.45, 0, 0.5) + ctx.shutterPulse * 0.35 + (handNear ? 0.25 : 0);
    eyePupil.scale.set(pupilScale, pupilScale, 1);
    for (const sb of shutterBlades) {
      sb.blade.position.x = sb.baseR + (1 - ctx.shutterPulse) * 0.009 + (handNear ? 0.006 : 0);
    }

    // Gyro Rings & Core Pulse
    const ringSpd = pal.spin * (1 + energy * 1.9 + (ctx.state === 'coding' ? 1.5 : 0));
    gyroRing1.rotation.z += dt * ringSpd;
    gyroRing1.rotation.y += dt * ringSpd * 0.45;
    gyroRing2.rotation.z -= dt * ringSpd * 0.82;
    gyroRing2.rotation.x += dt * ringSpd * 0.38;

    plasmaCore.scale.setScalar(1 + clamp(energy * 0.2, 0, 0.22) + ctx.shutterPulse * 0.05);
    plasmaCoreMat.emissiveIntensity = 1.8 + clamp(energy * 3.8, 0, 3.6) + ctx.shutterPulse * 1.5;
    botLight.intensity = 0.7 + clamp(energy * 1.4, 0, 1.4);
    ribCage.rotation.y += dt * 0.85;

    // Subtle micro-hover particles under sphere
    if (Math.random() < 0.28) {
      APP.fx.spawnPoint('glow', ctx.pos.x + (Math.random() - 0.5) * 0.06, ctx.pos.y - 0.14, ctx.pos.z + (Math.random() - 0.5) * 0.06, 0.42, pal.core);
    }

    // Keep Holographic Panel smoothly refreshed when visible
    if (holoState.visible) {
      holoGroup.lookAt(ctx.headPos.x, ctx.pos.y, ctx.headPos.z);
      if (Math.floor(t * 2) !== Math.floor((t - dt) * 2)) redrawHoloTabsAndScreen();
    }

    // 3) Articulated IK Robotic Tentacle & Object Carrying State Machine
    const basePos = ctx.pos.clone().add(new V3(0, -0.125, 0));
    let desiredTip = basePos.clone().add(new V3(-0.08, -0.34, 0.05));
    let clawOpenAmt = 0.45;
    pointerBeam.visible = false;

    if (tentacleState.mode === 'point') {
      if (now >= tentacleState.pointUntil) {
        tentacleState.mode = 'idle';
      } else {
        const dir = tentacleState.targetPos.clone().sub(basePos);
        const reachDist = Math.min(0.85, Math.max(0.35, dir.length() * 0.6));
        desiredTip = basePos.clone().add(dir.normalize().multiplyScalar(reachDist));
        clawOpenAmt = 0.08;
        const posArr = pointerBeam.geometry.attributes.position;
        posArr.setXYZ(0, desiredTip.x, desiredTip.y, desiredTip.z);
        posArr.setXYZ(1, tentacleState.targetPos.x, tentacleState.targetPos.y, tentacleState.targetPos.z);
        posArr.needsUpdate = true;
        pointerBeam.visible = true;
      }
    } else if (tentacleState.mode === 'grab_reach' && tentacleState.grabTask) {
      const task = tentacleState.grabTask;
      task.phaseTime += dt;
      const k = clamp(task.phaseTime / task.reachDur, 0, 1);
      const grabPt = task.ent.root.position.clone().add(task.grabOffset);
      desiredTip = basePos.clone().lerp(grabPt, 1 - Math.pow(1 - k, 3));
      clawOpenAmt = 0.85 * (1 - k * 0.5);
      if (k >= 1) {
        tentacleState.mode = 'grab_carry';
        task.phaseTime = 0;
        APP.sfx.play('click');
        APP.fx.spawnPoint('sparks', grabPt.x, grabPt.y, grabPt.z, 0.85, 0x00ffff);
      }
    } else if (tentacleState.mode === 'grab_carry' && tentacleState.grabTask) {
      const task = tentacleState.grabTask;
      task.phaseTime += dt;
      const k = clamp(task.phaseTime / task.carryDur, 0, 1);
      const ease = k < 0.5 ? 2 * k * k : 1 - Math.pow(-2 * k + 2, 2) / 2;
      const curObjPos = task.startPos.clone().lerp(task.destPos, ease);
      curObjPos.y += Math.sin(ease * Math.PI) * 0.28; // Graceful lifting arc
      task.ent.root.position.copy(curObjPos);
      task.ent.baseY = curObjPos.y;
      if (task.yawDeg !== null) {
        task.ent.root.rotation.y = THREE.MathUtils.lerp(task.ent.root.rotation.y, THREE.MathUtils.degToRad(task.yawDeg), ease * 0.25);
      }
      task.ent.box = APP.computeBox(task.ent.root);
      desiredTip = curObjPos.clone().add(task.grabOffset);
      clawOpenAmt = 0.12;
      bot.inspectPoint.copy(desiredTip);

      if (k >= 1) {
        task.ent.root.position.copy(task.destPos);
        task.ent.baseY = task.destPos.y;
        task.ent.box = APP.computeBox(task.ent.root);
        APP.fx.groundPulse(task.destPos.x, task.destPos.z, 1.2, 0x00ffaa);
        playEmotionalChirp('proud', 0.8);
        tentacleState.mode = 'grab_retract';
        task.phaseTime = 0;
      }
    } else if (tentacleState.mode === 'grab_retract' && tentacleState.grabTask) {
      const task = tentacleState.grabTask;
      task.phaseTime += dt;
      const k = clamp(task.phaseTime / task.retractDur, 0, 1);
      const endPos = task.destPos.clone().add(task.grabOffset);
      const restPos = basePos.clone().add(new V3(-0.08, -0.34, 0.05));
      desiredTip = endPos.lerp(restPos, k);
      clawOpenAmt = 0.65;
      if (k >= 1) {
        tentacleState.mode = 'idle';
        tentacleState.grabTask = null;
      }
    } else {
      // Idle graceful coiling behind & below sphere
      desiredTip = basePos.clone().add(new V3(
        Math.sin(t * 1.3) * 0.08 - 0.08,
        -0.36 + Math.cos(t * 1.7) * 0.04,
        Math.cos(t * 1.1) * 0.06 + 0.04
      ));
      clawOpenAmt = 0.4 + Math.sin(t * 2.2) * 0.12;
    }

    solveAndRenderTentacle(basePos, desiredTip, t, dt, clawOpenAmt);
  }
});
