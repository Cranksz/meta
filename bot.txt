function playEmotionalChirp(mood = 'curious', intensity = 1) {
  const ctx = APP.sfx?.ctx, bus = APP.sfx?.bus;
  if (!ctx || !bus || bot.muted) return;
  const t = ctx.currentTime;
  const presets = {
    curious:    [520, 740, 960],
    happy:      [660, 880, 1174, 1320],
    excited:    [587, 880, 1174, 1568],
    thoughtful: [440, 392, 494],
    alert:      [780, 520, 780],
    warm:       [494, 622, 740],
    proud:      [523, 659, 784, 1046]
  };
  (presets[mood] || presets.curious).forEach((f, idx) => {
    const osc = ctx.createOscillator(), g = ctx.createGain();
    osc.type = idx % 2 === 0 ? 'sine' : 'triangle';
    const st = t + idx * 0.052;
    osc.frequency.setValueAtTime(f * (0.97 + Math.random() * 0.06), st);
    osc.frequency.exponentialRampToValueAtTime(f * 1.04, st + 0.09);
    g.gain.setValueAtTime(0.001, st);
    g.gain.linearRampToValueAtTime(0.055 * intensity, st + 0.018);
    g.gain.exponentialRampToValueAtTime(0.001, st + 0.13);
    osc.connect(g).connect(bus); osc.start(st); osc.stop(st + 0.14);
  });
}

let spatialPanner = null, spatialAudioCtx = null;
bot.setOutputRouter((audioCtx, analyserNode) => {
  spatialAudioCtx = audioCtx;
  const lowWarm = audioCtx.createBiquadFilter(); lowWarm.type = 'lowshelf'; lowWarm.frequency.value = 190; lowWarm.gain.value = 3.2;
  const presence = audioCtx.createBiquadFilter(); presence.type = 'peaking'; presence.frequency.value = 2850; presence.Q.value = 1.1; presence.gain.value = 2.6;
  const comp = audioCtx.createDynamicsCompressor(); comp.threshold.value = -16; comp.knee.value = 18; comp.ratio.value = 3.5;
  let tail = comp;
  try {
    const panner = audioCtx.createPanner();
    panner.panningModel = 'HRTF'; panner.distanceModel = 'inverse';
    panner.refDistance = 1.0; panner.maxDistance = 18; panner.rolloffFactor = 0.85;
    spatialPanner = panner; comp.connect(panner); tail = panner;
  } catch { spatialPanner = null; }
  analyserNode.connect(lowWarm); lowWarm.connect(presence); presence.connect(comp);
  return tail;
});

function updateSpatialAudioPose(botPos, headPos, headFwd) {
  if (!spatialAudioCtx || !spatialPanner || !headPos || !headFwd || !botPos) return;
  try {
    const l = spatialAudioCtx.listener;
    if (l.positionX) {
      l.positionX.value = headPos.x; l.positionY.value = headPos.y; l.positionZ.value = headPos.z;
      l.forwardX.value = headFwd.x; l.forwardY.value = 0; l.forwardZ.value = headFwd.z;
      l.upX.value = 0; l.upY.value = 1; l.upZ.value = 0;
    }
    if (spatialPanner.positionX) {
      spatialPanner.positionX.value = botPos.x; spatialPanner.positionY.value = botPos.y; spatialPanner.positionZ.value = botPos.z;
    }
  } catch {}
}

async function speakAliveVoice(text, mood = 'warm') {
  if (bot.muted || !text) return true;
  if ('speechSynthesis' in window) {
    try {
      window.speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(text);
      u.lang = /[\u0600-\u06FF]/.test(text) ? 'ar-SA' : 'en-US';
      const pitchMap = { excited: 1.16, happy: 1.12, curious: 1.08, warm: 1.03, thoughtful: 0.96, alert: 1.18, proud: 1.10 };
      const rateMap  = { excited: 1.08, happy: 1.04, curious: 1.02, warm: 0.98, thoughtful: 0.94, alert: 1.10, proud: 1.02 };
      u.pitch = pitchMap[mood] || 1.05; u.rate = rateMap[mood] || 1.02;
      playEmotionalChirp(mood, 0.55);
      u.onstart = () => { bot.aiSpeakingUntil = NOW() + Math.max(2.0, text.length * 0.065); bot.setState('ai_speaking'); };
      u.onend = () => { bot.aiSpeakingUntil = 0; if (bot.connected && NOW() > bot.stateFlashUntil) bot.setState('watching'); };
      window.speechSynthesis.speak(u);
      return true;
    } catch {}
  }
  return false;
}

function solveAndRenderTentacle(basePos, targetTipPos, t, dt, clawOpenAmt) {
  const isAliveVisible = bot.rootGroup.visible && appearanceState.showTentacle && (bot.dissolveTime === null || bot.rootGroup.scale.x > 0.04);
  tentacleGroup.visible = isAliveVisible;
  if (!isAliveVisible) return;

  const tScale = clamp(Number(appearanceState.tentacleScale || 1) * Number(appearanceState.scale || 1) * bot.rootGroup.scale.x, 0.05, 2.5);
  const totalDist = basePos.distanceTo(targetTipPos);
  const segLen = Math.max(SEG_LEN * tScale, Math.min(0.28, (totalDist + 0.06) / SEG_COUNT));

  tentaclePoints[0].copy(basePos);
  if (tentaclePoints[SEG_COUNT].lengthSq() < 1e-5) {
    for (let i = 1; i <= SEG_COUNT; i++) tentaclePoints[i].copy(basePos).add(new V3(0, -i * segLen * 0.55, 0));
  }
  const swayAmp = tentacleState.mode === 'idle' ? 0.032 : (tentacleState.mode === 'wave' ? 0.08 : 0.012);

  for (let iter = 0; iter < 5; iter++) {
    tentaclePoints[SEG_COUNT].copy(targetTipPos);
    for (let i = SEG_COUNT - 1; i >= 1; i--) {
      const dir = tentaclePoints[i].clone().sub(tentaclePoints[i + 1]);
      if (dir.lengthSq() < 1e-6) dir.set(0, -1, 0);
      tentaclePoints[i].copy(tentaclePoints[i + 1]).addScaledVector(dir.normalize(), segLen);
    }
    tentaclePoints[0].copy(basePos);
    for (let i = 1; i <= SEG_COUNT; i++) {
      const dir = tentaclePoints[i].clone().sub(tentaclePoints[i - 1]);
      if (dir.lengthSq() < 1e-6) dir.set(0, -1, 0);
      dir.normalize();
      const wave = Math.sin(t * (tentacleState.mode === 'wave' ? 5.2 : 2.6) - i * 0.52) * swayAmp * (i / SEG_COUNT);
      dir.x += wave; dir.y -= 0.025 * (1 - i / SEG_COUNT); dir.normalize();
      tentaclePoints[i].copy(tentaclePoints[i - 1]).addScaledVector(dir, segLen);
    }
  }

  const up = new V3(0, 1, 0);
  for (let i = 0; i <= SEG_COUNT; i++) {
    tentacleJoints[i].position.copy(tentaclePoints[i]);
    tentacleJoints[i].scale.setScalar(tScale);
    if (i < SEG_COUNT) {
      const a = tentaclePoints[i], b = tentaclePoints[i + 1], link = tentacleLinks[i];
      link.position.copy(a).lerp(b, 0.5);
      const diff = b.clone().sub(a), len = Math.max(0.001, diff.length());
      link.scale.set(tScale, len, tScale);
      link.quaternion.setFromUnitVectors(up, diff.divideScalar(len));
    }
  }
  const tip = tentaclePoints[SEG_COUNT], prev = tentaclePoints[SEG_COUNT - 1];
  clawGroup.position.copy(tip); clawGroup.scale.setScalar(tScale);
  clawGroup.lookAt(tip.clone().add(tip.clone().sub(prev)));
  for (const finger of clawProngs) {
    finger.rotation.x = THREE.MathUtils.lerp(finger.rotation.x, 0.14 + clawOpenAmt * 0.56, 1 - Math.exp(-12 * dt));
    finger.position.y = 0.017 + clawOpenAmt * 0.016;
  }
}

function applyBotAppearance(opts = {}) {
  if (opts.preset && APPEARANCE_PRESETS[opts.preset]) {
    Object.assign(appearanceState, APPEARANCE_PRESETS[opts.preset], { preset: opts.preset });
  }
  ['shellColor', 'metalColor', 'coreColor', 'ringColor', 'irisColor', 'tentacleColor', 'finish', 'tagTitle'].forEach(k => {
    const snake = k.replace(/[A-Z]/g, m => '_' + m.toLowerCase());
    if (opts[k] !== undefined) appearanceState[k] = opts[k];
    else if (opts[snake] !== undefined) appearanceState[k] = opts[snake];
  });
  if (opts.scale !== undefined) appearanceState.scale = clamp(Number(opts.scale), 0.4, 2.5);
  if (opts.ring_speed !== undefined) appearanceState.ringSpeed = clamp(Number(opts.ring_speed), 0.1, 6.0);
  if (opts.show_rings !== undefined) appearanceState.showRings = Boolean(opts.show_rings);
  if (opts.show_fins !== undefined) appearanceState.showFins = Boolean(opts.show_fins);
  if (opts.show_tentacle !== undefined) appearanceState.showTentacle = Boolean(opts.show_tentacle);
  if (opts.tentacle_scale !== undefined) appearanceState.tentacleScale = clamp(Number(opts.tentacle_scale), 0.5, 2.2);

  chassisRoot.scale.setScalar(appearanceState.scale);
  tagSprite.position.y = 0.28 * appearanceState.scale;
  carbonMat.color.set(appearanceState.shellColor);
  titaniumMat.color.set(appearanceState.metalColor);

  const finMode = String(appearanceState.finish || 'carbon').toLowerCase();
  carbonMat.wireframe = (finMode === 'wireframe');
  ribMat.wireframe = true;
  if (finMode === 'glass' || finMode === 'crystal') {
    carbonMat.transparent = true; carbonMat.opacity = 0.45; carbonMat.transmission = 0.72; carbonMat.roughness = 0.06; carbonMat.metalness = 0.1;
  } else if (finMode === 'gold') {
    carbonMat.transparent = false; carbonMat.opacity = 1; carbonMat.transmission = 0; carbonMat.metalness = 0.96; carbonMat.roughness = 0.14;
    titaniumMat.metalness = 0.98; titaniumMat.roughness = 0.12;
  } else if (finMode === 'neon') {
    carbonMat.transparent = false; carbonMat.opacity = 1; carbonMat.transmission = 0; carbonMat.emissive = new THREE.Color(appearanceState.coreColor || '#00ffaa'); carbonMat.emissiveIntensity = 0.35;
  } else {
    carbonMat.transparent = false; carbonMat.opacity = 1; carbonMat.transmission = 0; carbonMat.emissive = new THREE.Color(0x000000); carbonMat.emissiveIntensity = 0;
    carbonMat.metalness = finMode === 'metallic' ? 0.92 : 0.78; carbonMat.roughness = finMode === 'metallic' ? 0.18 : 0.25;
  }
  carbonMat.needsUpdate = true; titaniumMat.needsUpdate = true;

  gyroRing1.visible = appearanceState.showRings;
  gyroRing2.visible = appearanceState.showRings;
  aeroFins.forEach(f => { f.mesh.visible = appearanceState.showFins; });

  if (appearanceState.tentacleColor) {
    tentacleConduitMat.color.set(appearanceState.tentacleColor);
    tentacleConduitMat.emissive.set(appearanceState.tentacleColor);
  }
  if (opts.mood) bot.setMood(opts.mood, appearanceState.tagTitle);
  else bot.setMood(bot.mood, appearanceState.tagTitle);

  if (opts.save_permanently && bot.memory) {
    bot.memory.botAppearance = structuredClone(appearanceState);
    bot.saveLocalMemory();
    bot.scheduleGitHubMemorySync(1200);
  }
  return { ok: true, appearance: appearanceState };
}

function computeSemanticDestination(spec = {}) {
  if (Number.isFinite(Number(spec.x)) && Number.isFinite(Number(spec.z))) {
    return new V3(Number(spec.x), Number.isFinite(Number(spec.y)) ? Number(spec.y) : 0, Number(spec.z));
  }
  const dest = APP.normalizeText(String(spec.destination || spec.relation || ''));
  const dist = clamp(Number(spec.distance || 1.85), 0.65, 5.5);
  const right = new V3().crossVectors(APP.headFwd, new V3(0, 1, 0)).normalize();
  const basePos = APP.headPos.clone();

  if (spec.reference_object) {
    const refs = APP.findEntities(spec.reference_object);
    if (refs.length) {
      const refEnt = refs[0], rb = APP.computeBox(refEnt.root), rc = rb.getCenter(new V3());
      const out = rc.clone();
      if (/left|يسار|شمال/.test(dest)) out.addScaledVector(right, -0.95);
      else if (/right|يمين/.test(dest)) out.addScaledVector(right, 0.95);
      else if (/front|امام|قدام/.test(dest)) out.addScaledVector(APP.headFwd, -0.85);
      else if (/behind|خلف|وراء/.test(dest)) out.addScaledVector(APP.headFwd, 0.85);
      else if (/top|above|فوق|اعلى/.test(dest)) out.y = rb.max.y + 0.04;
      else out.addScaledVector(right, 0.85);
      if (!/top|above|فوق|اعلى/.test(dest)) out.y = Number.isFinite(Number(spec.height)) ? Number(spec.height) : refEnt.root.position.y;
      return out;
    }
  }
  if (/wall|حائط|جدار/.test(dest)) { const w = APP.findNearestWall(APP.headPos, APP.headFwd); if (w) return w.point.clone().addScaledVector(w.normal, 0.45); }
  const targets = APP.findEntities(spec.target || 'last');
  if (targets.length && /closer|اقرب|قرب|farther|ابعد|بعد|higher|ارفع|اعلى|lower|اخفض|انزل|floor|ارض/.test(dest)) {
    const cur = targets[0].root.position.clone();
    if (/closer|اقرب|قرب/.test(dest)) return cur.lerp(new V3(basePos.x, cur.y, basePos.z), 0.48);
    if (/farther|ابعد|بعد/.test(dest)) return cur.addScaledVector(APP.headFwd, 1.15);
    if (/higher|ارفع|اعلى/.test(dest)) { cur.y += Number(spec.height || 0.85); return cur; }
    if (/lower|اخفض|انزل|floor|ارض/.test(dest)) { cur.y = Math.max(0, Number.isFinite(Number(spec.height)) ? Number(spec.height) : 0); return cur; }
  }
  const out = new V3(basePos.x, Number.isFinite(Number(spec.height)) ? Number(spec.height) : 0, basePos.z);
  if (/left|يسار|شمال/.test(dest)) out.addScaledVector(APP.headFwd, dist).addScaledVector(right, -1.05);
  else if (/right|يمين/.test(dest)) out.addScaledVector(APP.headFwd, dist).addScaledVector(right, 1.05);
  else if (/behind|خلف|وراء/.test(dest)) out.addScaledVector(APP.headFwd, -dist);
  else if (/near|hand|يد|بجانبي|جنبي|قريب|عندي|الي/.test(dest)) out.addScaledVector(APP.headFwd, 0.8).addScaledVector(right, 0.25);
  else out.addScaledVector(APP.headFwd, dist);
  if (/fly|air|هواء|طائر/.test(dest) && !Number.isFinite(Number(spec.height))) out.y = 1.25;
  return out;
}

function getApproachSpotFor(worldPt) {
  const toBot = bot.pos.clone().sub(worldPt); toBot.y = 0;
  if (toBot.lengthSq() < 1e-4) toBot.set(0.4, 0, 0.4);
  toBot.normalize();
  return worldPt.clone().addScaledVector(toBot, 0.62).add(new V3(0, 0.34, 0));
}

function startTentacleMoveAndPlace(args = {}) {
  const targets = APP.findEntities(args.target || 'last');
  if (!targets.length) return { ok: false, error: 'لا يوجد مجسم في المشهد لتحريكه الآن' };
  const ent = targets[0], destPos = computeSemanticDestination(args), startPos = ent.root.position.clone();
  const box = APP.computeBox(ent.root);
  const center = box.isEmpty() ? startPos.clone().add(new V3(0, 0.35, 0)) : box.getCenter(new V3());
  const grabOffset = center.clone().sub(startPos);

  if (tentacleState.heldEnt && tentacleState.heldEnt !== ent) tentacleState.heldEnt.heldByTentacle = false;
  tentacleState.heldEnt = null;
  ent.heldByTentacle = true;
  tentacleState.mode = 'grab_reach';
  tentacleState.grabTask = {
    ent, startPos, destPos, grabOffset, holdOnly: Boolean(args.hold_only),
    phaseTime: 0, reachDur: 0.65, carryDur: 1.15, retractDur: 0.45,
    yawDeg: Number.isFinite(Number(args.yaw_deg)) ? Number(args.yaw_deg) : null
  };
  bot.flyTo(getApproachSpotFor(center), 7.5);
  bot.inspectPoint.copy(center); bot.inspectUntil = NOW() + 3.5;
  bot.setMood('excited'); playEmotionalChirp('excited', 0.9);
  return { ok: true, movingObject: ent.name, from: startPos, to: destPos };
}

function releaseHeldObject(throwPower = 0) {
  const ent = tentacleState.heldEnt || tentacleState.grabTask?.ent;
  tentacleState.grabTask = null;
  if (!ent) { tentacleState.mode = 'idle'; bot.clearFlight(); return { ok: false, error: 'المخلب فارغ حالياً' }; }
  ent.heldByTentacle = false; tentacleState.heldEnt = null; tentacleState.mode = 'idle'; bot.clearFlight();

  if (throwPower > 0) {
    const dir = APP.headFwd.clone().multiplyScalar(throwPower).add(new V3(0, throwPower * 0.42, 0));
    const vel = dir.clone(), tid = 'throw_' + ent.id + '_' + Date.now();
    const box = APP.computeBox(ent.root), rad = Math.max(0.15, box.getSize(new V3()).y * 0.35);
    APP.sfx.play('magic');
    APP.onTick(tid, 4.5, dt => {
      if (!APP.entities.includes(ent)) { APP.stopTick(tid); return; }
      vel.y += APP.worldG * dt;
      const center = ent.root.position.clone().add(new V3(0, rad, 0));
      center.addScaledVector(vel, dt);
      APP.phys.collidePoint(center, rad, vel, 0.48, 0.88);
      ent.root.position.set(center.x, Math.max(0, center.y - rad), center.z);
      ent.baseY = ent.root.position.y; ent.box = APP.computeBox(ent.root);
      if (vel.lengthSq() < 0.04 && ent.root.position.y <= 0.03) APP.stopTick(tid);
    });
    return { ok: true, action: 'thrown', object: ent.name };
  }
  ent.baseY = ent.root.position.y; ent.box = APP.computeBox(ent.root); APP.sfx.play('click');
  return { ok: true, action: 'released', object: ent.name };
}

function handleControlTentacle(args = {}) {
  const action = String(args.action || 'wave').toLowerCase();
  if (action === 'grab_hold' || action === 'grab') return startTentacleMoveAndPlace({ target: args.target || 'last', hold_only: true });
  if (action === 'release' || action === 'drop') return releaseHeldObject(0);
  if (action === 'throw' || action === 'toss') return releaseHeldObject(clamp(Number(args.power || 5.2), 2, 12));
  if (action === 'retract' || action === 'idle') {
    if (tentacleState.heldEnt) releaseHeldObject(0);
    tentacleState.mode = 'idle'; bot.clearFlight();
    return { ok: true, action: 'retract' };
  }
  if (action === 'wave' || action === 'high_five') {
    tentacleState.mode = 'wave'; tentacleState.until = NOW() + clamp(Number(args.duration || 3.2), 1, 8);
    bot.setMood('happy'); playEmotionalChirp('happy', 0.9);
    return { ok: true, action };
  }
  if (action === 'point') {
    const ents = APP.findEntities(args.target || 'last');
    if (!ents.length) return { ok: false, error: 'No target to point at' };
    const b = APP.computeBox(ents[0].root);
    tentacleState.targetPos = b.isEmpty() ? ents[0].root.position.clone() : b.getCenter(new V3());
    tentacleState.mode = 'point'; tentacleState.until = NOW() + clamp(Number(args.duration || 4), 1, 12);
    bot.inspectPoint.copy(tentacleState.targetPos); bot.inspectUntil = tentacleState.until;
    return { ok: true, pointingAt: ents[0].name };
  }
  if (action === 'poke_push' || action === 'poke') {
    const ents = APP.findEntities(args.target || 'last');
    if (!ents.length) return { ok: false, error: 'No object to poke' };
    const ent = ents[0], b = APP.computeBox(ent.root), c = b.isEmpty() ? ent.root.position.clone() : b.getCenter(new V3());
    tentacleState.mode = 'poke_push'; tentacleState.until = NOW() + 1.4; tentacleState.targetPos.copy(c);
    bot.flyTo(getApproachSpotFor(c), 8.0);
    setTimeout(() => {
      if (!APP.entities.includes(ent)) return;
      const pushDir = c.clone().sub(APP.headPos); pushDir.y = 0; pushDir.normalize();
      ent.root.position.addScaledVector(pushDir, 0.65); ent.baseY = ent.root.position.y; ent.box = APP.computeBox(ent.root);
      APP.fx.spawnPoint('sparks', c.x, c.y, c.z, 0.9, 0x00ffff); APP.sfx.play('click');
      setTimeout(() => { if (tentacleState.mode === 'poke_push') { tentacleState.mode = 'idle'; bot.clearFlight(); } }, 500);
    }, 520);
    return { ok: true, poked: ent.name };
  }
  if (action === 'extend') {
    const dirStr = APP.normalizeText(String(args.direction || 'front'));
    const dist = clamp(Number(args.distance || 0.72), 0.25, 1.3);
    const right = new V3().crossVectors(APP.headFwd, new V3(0, 1, 0)).normalize();
    const off = new V3(0, -0.1, 0);
    if (/left|يسار|شمال/.test(dirStr)) off.addScaledVector(right, -dist);
    else if (/right|يمين/.test(dirStr)) off.addScaledVector(right, dist);
    else if (/up|اعلى|فوق/.test(dirStr)) off.y = dist;
    else if (/down|اسفل|تحت/.test(dirStr)) off.y = -dist;
    else off.addScaledVector(APP.headFwd, dist);
    tentacleState.extendOffset.copy(off);
    tentacleState.mode = 'extend'; tentacleState.until = NOW() + clamp(Number(args.duration || 4), 1, 12);
    return { ok: true, direction: dirStr, distance: dist };
  }
  return { ok: false, error: 'Unknown tentacle action: ' + action };
}

let lastTouchAt = 0, lastHoverChirpAt = 0, lastEyeChirpAt = 0;

function handleTactileImpact(hitPt, impulseDir, handSide) {
  const now = NOW();
  if (now - lastTouchAt < 0.28) return;
  lastTouchAt = now;
  bot.recoilVel.addScaledVector(impulseDir, 1.35);
  bot.inspectPoint.copy(hitPt); bot.inspectUntil = now + 2.4;
  bot.setMood('happy', 'لمسة لطيفة! ✨'); playEmotionalChirp('happy', 0.9);
  for (let i = 0; i < 12; i++) APP.fx.spawnPoint('sparks', hitPt.x, hitPt.y, hitPt.z, 0.75, 0x44ffee);
  bot.recordTelemetry('companion_touched', { hand: handSide });
}

const touchRay = new THREE.Raycaster(), tmpQ = new Q4(), tmpD = new V3(), tmpW = new V3();

bot.setController({
  mount(rootGroup) {

      if (!hit && hand.tipWp.distanceTo(bot.pos) < 0.24 * appearanceState.scale) hit = { object: hitProxy, point: hand.tipWp.clone() };
    } else if (hand.c) {
      hand.c.getWorldPosition(tmpW); hand.c.getWorldQuaternion(tmpQ); tmpD.set(0, 0, -1).applyQuaternion(tmpQ);
      touchRay.set(tmpW, tmpD); hit = touchRay.intersectObjects(targets, false)[0] || null;

  onStep(dt, t, ctx) {

    updateSpatialAudioPose(ctx.pos, ctx.headPos, ctx.headFwd);

      for (const p of pts) { const d = p.distanceTo(ctx.pos); if (d < nearestHandDist) { nearestHandDist = d; nearestHandPt = p; } }
      if (nearestHandDist < 0.16 * appearanceState.scale && nearestHandPt) handleTactileImpact(nearestHandPt, ctx.pos.clone().sub(nearestHandPt).normalize(), h.side);
      if (holoState.visible && h.handTrackingActive) {
        for (const tab of holoState.tabs) {
          if (h.tipWp.distanceTo(tab.mesh.getWorldPosition(new V3())) < 0.085 && now - lastTouchAt > 0.45) {
            lastTouchAt = now; APP.sfx.play('click');
            if (tab.id === 'close') showHologram({ visible: false }); else showHologram({ visible: true, active_tab: tab.id });
          }
        }
      }
    }
    const handNear = nearestHandDist < 0.36 * appearanceState.scale;
    if (handNear && now - lastHoverChirpAt > 3.0) { lastHoverChirpAt = now; playEmotionalChirp('curious', 0.45); }
    if (ctx.lookingAtBot && now - lastEyeChirpAt > 6.5) { lastEyeChirpAt = now; playEmotionalChirp('warm', 0.3); }

    const flare = pal.flare + (handNear ? 0.18 : 0) + (ctx.lookingAtBot ? 0.08 : 0) + clamp(energy * 0.14, 0, 0.22);
    upperShell.position.y = THREE.MathUtils.lerp(upperShell.position.y, flare * 0.045, 1 - Math.exp(-8 * dt));
    lowerShell.position.y = THREE.MathUtils.lerp(lowerShell.position.y, -flare * 0.045, 1 - Math.exp(-8 * dt));
    leftArmor.position.x = -0.114 - flare * 0.045; rightArmor.position.x = 0.114 + flare * 0.045;
    leftArmor.rotation.z = flare * 0.38; rightArmor.rotation.z = -flare * 0.38;
    bodyPivot.rotation.z = clamp(-ctx.vel.dot(ctx.right) * 0.16, -0.30, 0.30) + (ctx.lookingAtBot ? Math.sin(t * 1.4) * 0.06 : 0);
    bodyPivot.rotation.x = clamp(ctx.vel.dot(ctx.headFwd) * 0.14, -0.25, 0.25);

    for (const f of aeroFins) {
      const r = 0.145 + flare * 0.055, ang = f.angle + Math.sin(t * 1.8) * 0.06;
      f.mesh.position.set(Math.cos(ang) * r, Math.sin(t * 3.1 + f.angle) * 0.012, Math.sin(ang) * r);
      f.mesh.rotation.y = -ang + Math.PI / 2; f.mesh.rotation.z = flare * 0.35;
    }

    const lookTarget = (handNear && nearestHandPt) ? nearestHandPt : ((now < ctx.inspectUntil && !ctx.lookingAtBot && ctx.state !== 'ai_speaking') ? ctx.inspectPoint : ctx.headPos);
    const lookObj = new THREE.Object3D(); lookObj.position.copy(ctx.pos); lookObj.lookAt(lookTarget);
    eyePivot.quaternion.slerp(lookObj.quaternion, 1 - Math.exp(-9.0 * dt));

    const pupilScale = 0.85 + clamp(energy * 0.45, 0, 0.5) + ctx.shutterPulse * 0.35 + (handNear || ctx.lookingAtBot ? 0.22 : 0);
    eyePupil.scale.set(pupilScale, pupilScale, 1);
    for (const sb of shutterBlades) sb.blade.position.x = sb.baseR + (1 - ctx.shutterPulse) * 0.009 + (handNear ? 0.006 : 0);

    const ringSpd = pal.spin * appearanceState.ringSpeed * (1 + energy * 1.9 + (ctx.state === 'coding' ? 1.5 : 0));
    gyroRing1.rotation.z += dt * ringSpd; gyroRing1.rotation.y += dt * ringSpd * 0.45;
    gyroRing2.rotation.z -= dt * ringSpd * 0.82; gyroRing2.rotation.x += dt * ringSpd * 0.38;
    plasmaCore.scale.setScalar(1 + clamp(energy * 0.2, 0, 0.22) + ctx.shutterPulse * 0.05);
    plasmaCoreMat.emissiveIntensity = 1.8 + clamp(energy * 3.8, 0, 3.6) + ctx.shutterPulse * 1.5;
    botLight.intensity = 0.7 + clamp(energy * 1.4, 0, 1.4); ribCage.rotation.y += dt * 0.85;

    if (Math.random() < 0.25) APP.fx.spawnPoint('glow', ctx.pos.x + (Math.random() - 0.5) * 0.06, ctx.pos.y - 0.14 * appearanceState.scale, ctx.pos.z + (Math.random() - 0.5) * 0.06, 0.42, pal.core);
    if (holoState.visible) { holoGroup.lookAt(ctx.headPos.x, ctx.pos.y, ctx.headPos.z); if (Math.floor(t * 2) !== Math.floor((t - dt) * 2)) redrawHoloTabsAndScreen(); }

    const basePos = ctx.pos.clone().add(new V3(0, -0.125 * appearanceState.scale, 0));

        desiredTip = basePos.clone().add(dir.normalize().multiplyScalar(reachDist)); clawOpenAmt = 0.08;

        posArr.needsUpdate = true; pointerBeam.visible = true;
      }

      if (now >= tentacleState.until) tentacleState.mode = 'idle';
      else {
        desiredTip = basePos.clone().addScaledVector(ctx.right, 0.28 + Math.sin(t * 7.0) * 0.22).add(new V3(0, 0.16 + Math.cos(t * 5.0) * 0.1, 0.18));
        clawOpenAmt = 0.85;
      }
    } else if (tentacleState.mode === 'extend') {
      if (now >= tentacleState.until) tentacleState.mode = 'idle';
      else { desiredTip = basePos.clone().add(tentacleState.extendOffset); clawOpenAmt = 0.55; }
    } else if (tentacleState.mode === 'poke_push') {
      desiredTip = tentacleState.targetPos.clone(); clawOpenAmt = 0.05;
    } else if (tentacleState.mode === 'grab_reach' && tentacleState.grabTask) {
      const task = tentacleState.grabTask; task.phaseTime += dt;
      if (!APP.entities.includes(task.ent)) { tentacleState.mode = 'idle'; tentacleState.grabTask = null; bot.clearFlight(); }
      else {
        const k = clamp(task.phaseTime / task.reachDur, 0, 1), grabPt = task.ent.root.position.clone().add(task.grabOffset);
        bot.flyTo(getApproachSpotFor(grabPt), 7.5);
        desiredTip = basePos.clone().lerp(grabPt, 1 - Math.pow(1 - k, 3)); clawOpenAmt = 0.85 * (1 - k * 0.55);
        if (k >= 1) {
          APP.sfx.play('click'); APP.fx.spawnPoint('sparks', grabPt.x, grabPt.y, grabPt.z, 0.85, 0x00ffff);
          if (task.holdOnly) {
            tentacleState.heldEnt = task.ent; tentacleState.heldOffset.copy(task.grabOffset);
            tentacleState.mode = 'grab_hold'; tentacleState.grabTask = null; bot.clearFlight();
          } else {
            tentacleState.mode = 'grab_carry'; task.phaseTime = 0;
          }
        }
      }
    } else if (tentacleState.mode === 'grab_carry' && tentacleState.grabTask) {
      const task = tentacleState.grabTask; task.phaseTime += dt;
      if (!APP.entities.includes(task.ent)) { tentacleState.mode = 'idle'; tentacleState.grabTask = null; bot.clearFlight(); }
      else {
        const k = clamp(task.phaseTime / task.carryDur, 0, 1), ease = k < 0.5 ? 2 * k * k : 1 - Math.pow(-2 * k + 2, 2) / 2;
        const curObjPos = task.startPos.clone().lerp(task.destPos, ease); curObjPos.y += Math.sin(ease * Math.PI) * 0.28;
        task.ent.root.position.copy(curObjPos); task.ent.baseY = curObjPos.y;
        if (task.yawDeg !== null) task.ent.root.rotation.y = THREE.MathUtils.lerp(task.ent.root.rotation.y, THREE.MathUtils.degToRad(task.yawDeg), ease * 0.25);
        task.ent.box = APP.computeBox(task.ent.root);
        desiredTip = curObjPos.clone().add(task.grabOffset); clawOpenAmt = 0.1;
        bot.flyTo(getApproachSpotFor(desiredTip), 7.5); bot.inspectPoint.copy(desiredTip);
        if (k >= 1) {
          task.ent.root.position.copy(task.destPos); task.ent.baseY = task.destPos.y; task.ent.box = APP.computeBox(task.ent.root);
          task.ent.heldByTentacle = false;
          APP.fx.groundPulse(task.destPos.x, task.destPos.z, 1.2, 0x00ffaa); playEmotionalChirp('proud', 0.8);
          tentacleState.mode = 'grab_retract'; task.phaseTime = 0; bot.clearFlight();
        }
      }
    } else if (tentacleState.mode === 'grab_hold' && tentacleState.heldEnt) {
      const ent = tentacleState.heldEnt;
      if (!APP.entities.includes(ent)) { tentacleState.mode = 'idle'; tentacleState.heldEnt = null; }
      else {
        desiredTip = basePos.clone().addScaledVector(ctx.right, -0.14).addScaledVector(ctx.headFwd, 0.28).add(new V3(0, -0.16, 0));
        ent.root.position.copy(desiredTip).sub(tentacleState.heldOffset);
        ent.baseY = ent.root.position.y; ent.box = APP.computeBox(ent.root);
        clawOpenAmt = 0.1;
      }
    } else if (tentacleState.mode === 'grab_retract' && tentacleState.grabTask) {
      const task = tentacleState.grabTask; task.phaseTime += dt;
      const k = clamp(task.phaseTime / task.retractDur, 0, 1);
      desiredTip = task.destPos.clone().add(task.grabOffset).lerp(basePos.clone().add(new V3(-0.08, -0.34, 0.05)), k);
      clawOpenAmt = 0.65;
      if (k >= 1) { tentacleState.mode = 'idle'; tentacleState.grabTask = null; }
    } else if (handNear && nearestHandPt) {
      desiredTip = basePos.clone().lerp(nearestHandPt, 0.72); clawOpenAmt = 0.75;
    } else {
      desiredTip = basePos.clone().add(new V3(Math.sin(t * 1.3) * 0.08 - 0.08, -0.36 + Math.cos(t * 1.7) * 0.04, Math.cos(t * 1.1) * 0.06 + 0.04));
      clawOpenAmt = 0.4 + Math.sin(t * 2.2) * 0.12;
    }

    solveAndRenderTentacle(basePos, desiredTip, t, dt, clawOpenAmt);
  }
});
